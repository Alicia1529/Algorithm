**Approach:** Build one Trie of all words, then DFS the grid a single time, walking the
Trie in lockstep. Prune the moment the current grid path isn't a prefix in the Trie. Beats
running Word Search once per word (`O(W · m · n · 4ᴸ)` → shared-prefix traversal).

- **Time:** O(m · n · 4·3^(L-1)) where L = longest word, plus O(total chars) to build the Trie.
- **Space:** O(total chars in words) for the Trie + O(L) recursion depth.

```python
class TrieNode:
    def __init__(self):
        self.children = {}      # char -> TrieNode
        self.word = None        # store the full word at its terminal node


class Solution:
    def findWords(self, board: List[List[str]], words: List[str]) -> List[str]:
        # 1. Build one Trie of all target words
        root = TrieNode()
        for w in words:
            node = root
            for c in w:
                if c not in node.children:
                    node.children[c] = TrieNode()
                node = node.children[c]
            node.word = w           # mark the end with the complete word

        rows, cols = len(board), len(board[0])
        res = []

        def dfs(r, c, node):
            ch = board[r][c]
            if ch not in node.children:     # prefix not in Trie → prune this branch
                return
            nxt = node.children[ch]
            if nxt.word:                    # found a complete word
                res.append(nxt.word)
                nxt.word = None             # de-dup: don't report it again

            board[r][c] = '#'               # mark visited
            for dr, dc in ((1, 0), (-1, 0), (0, 1), (0, -1)):
                nr, nc = r + dr, c + dc
                if 0 <= nr < rows and 0 <= nc < cols and board[nr][nc] != '#':
                    dfs(nr, nc, nxt)
            board[r][c] = ch                # restore (backtrack)

        for r in range(rows):
            for c in range(cols):
                dfs(r, c, root)
        return res
```

```c++
class Solution {
public:
    int R;
    int C;
    vector<string> ans;
    bool findWordsImpl(vector<vector<char>>& board, int i, int j, string& word, int idx, vector<int>& visited) {
        if (i < 0 || i >= R || j < 0 || j >= C) return false;
        if (visited[i * C + j] || board[i][j] != word[idx]) return false;
        if (idx == word.size() - 1) {
            if (find(ans.begin(), ans.end(), word) == ans.end())
                ans.push_back(word);
            return true;
        }
        
        visited[i * C + j] = 1;
        bool ret = findWordsImpl(board, i + 1, j, word, idx + 1, visited) || findWordsImpl(board, i - 1, j, word, idx + 1, visited) || findWordsImpl(board, i, j - 1, word, idx + 1, visited) || findWordsImpl(board, i, j + 1, word, idx + 1, visited);
        if (ret) return true;
        visited[i * C + j] = 0;
        return false;
    }
    vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {
        R = board.size();
        C = board[0].size();
        vector<int> visited(R * C);
        for (int i = 0; i < R; i++) {
            for (int j = 0; j < C; j++) {
                for (int k = 0; k < words.size(); k++) {
                    fill(visited.begin(), visited.end(), 0);
                    findWordsImpl(board, i, j, words[k], 0, visited);
                }
            }
        }
        return ans;
    }
};


class Solution {
    vector<string> ans;
public:
    
    bool dfs(vector<vector<char>> &board,const string &query,int n,int index,int i,int j){
        if(i>=0&&j>=0&&i<board.size()&&j<board[i].size()){
            bool res=false;
            if(board[i][j]!='*'){
                char prev=board[i][j];
                if(board[i][j]!=query[index]){
                    return false;
                }
                if(index==n-1)return true;
                board[i][j]='*';
                if(!res)res|=dfs(board,query,n,index+1,i-1,j);
                if(!res)res|=dfs(board,query,n,index+1,i,j-1);
                if(!res)res|=dfs(board,query,n,index+1,i+1,j);
                if(!res)res|=dfs(board,query,n,index+1,i,j+1);
                board[i][j]=prev;
                
                return res;
            }
        }
        return false;
    }
    
    void solve(vector<vector<char>> &board,const string &query){
        int n=board.size(),m=board[0].size();
        
        for(int i=0;i<n;i++){
            for(int j=0;j<m;j++){
                if(board[i][j]==query[0]){
                    if(dfs(board,query,query.length(),0,i,j)){
                        ans.push_back(query);
                        return ;
                    }
                }
            }
        }
    }
    
    vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {
        int n=words.size();
        
        for(int i=0;i<n;i++){
            solve(board,words[i]);
        }
        
        return ans;
    }
};




class Node {
public:
    char c;
    bool isWord = false;
    Node* children[26];

    Node (char _c): c(_c) {
        for (int i = 0; i < 26; i++)
            children[i] = nullptr;
    }
    
    void addWord(string word) {
        Node* curr = this;
        for (int i = 0; i < word.size(); i++) {
            char c = word[i];
            if (curr->children[c - 'a'] == nullptr) {
                curr->children[c - 'a'] = new Node(c);
            }
            curr = curr->children[c - 'a'];
        }
        curr->isWord = true;
    }
};

class Solution {
public:
    unordered_set<string> ans;
    vector<vector<bool>> memo;

    void dfs(vector<vector<char>>& board, int m, int n, int r, int c, string word, Node* node) {
        if (r < 0 || r >= m || c < 0 || c >= n) return;
        if (memo[r][c] || node->children[board[r][c] - 'a'] == nullptr) return;

        memo[r][c] = true;
        node = node->children[board[r][c] - 'a'];
        word += board[r][c];
        if (node->isWord) ans.insert(word);
        
        dfs(board, m, n, r + 1, c, word, node);
        dfs(board, m, n, r, c + 1, word, node);
        dfs(board, m, n, r - 1, c, word, node);
        dfs(board, m, n, r, c - 1, word, node);
        
        memo[r][c] = false;
    }
    
    vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {
        Node* node = new Node('\0');
        for (auto& word: words) node->addWord(word);
        
        int m = board.size(), n = board[0].size();
        memo = vector<vector<bool>>(m, vector<bool>(n, false));
        
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                dfs(board, m, n, i, j, "", node);
            }
        }
        
        return vector<string>(ans.begin(), ans.end());
    }
};
```
