class Solution:
    def longestOnes(self, nums: List[int], k: int) -> int:
        longest = 0
        numZeros = 0
        n = len(nums)
        l = 0
        for r in range(n):
            if nums[r] == 0:
                numZeros += 1
            while numZeros == k + 1:
                if nums[l] == 0:
                    numZeros -= 1
                l += 1
            longest = max(longest, r - l + 1)
        return longest
