class Solution:
    def isPalindrome(self, head: Optional[ListNode]) -> bool:
        array = []
        while head:
            array.append(head.val)
            head = head.next
        i, j = 0, len(array) - 1
        while i <= j:
            if array[i] != array[j]: return False
            i += 1
            j -= 1
        return True

    def isPalindrome(self, head: Optional[ListNode]) -> bool:
        curr = head
        size = 0
        while curr:
            size += 1
            curr = curr.next
        new_head = head
        for _ in range(size // 2):
            new_head = new_head.next

        tail_next = new_head
        if size % 2:
            new_head = new_head.next
        
        # reverse first half
        prev, curr = None, head
        while curr != tail_next:
            tmp = curr.next
            curr.next = prev
            prev = curr
            curr = tmp
        while prev is not None and new_head is not None:
            if prev.val != new_head.val: return False
            prev = prev.next
            new_head = new_head.next
        return True
