    }
    
    string get(string key, int timestamp) {
        int l = 0, r = map[key].size() - 1;
        while (l <= r) {
            int m = l + (r - l) / 2;
            if (map[key][m].first <= timestamp) {
                l = m + 1;
            } else {
                r  = m - 1;
            }
        }
        if (r == -1) return "";
        return map[key][r].second;
    }
};
