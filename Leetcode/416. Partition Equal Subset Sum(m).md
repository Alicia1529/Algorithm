Approach 3: Bottom Up Dynamic Programming
```c++
class Solution {
public:
    bool canPartition(vector<int>& nums) {
        int sum = 0;
        for (auto num: nums) sum += num;
        if (sum % 2) return false;

        int n = nums.size();
        int m = sum / 2;
        bool dp[n + 1][m + 1];
        for (int i = 0; i <= n; i++) dp[i][0] = true;
        for (int j = 1; j <= m; j++) dp[0][j] = false;
        
        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (nums[i - 1] <= j) {
                    dp[i][j] = dp[i - 1][j - nums[i - 1]] || dp[i - 1][j];
                } else {
                    dp[i][j] = dp[i - 1][j];
                }
            }
        }
        return dp[n][m];
    }
};
```

Approach 4: Optimised Dynamic Programming - Using 1D Array
```c++
// Runtime: 204 ms, faster than 40.16% of C++ online submissions for Partition Equal Subset Sum.
// Memory Usage: 8.9 MB, less than 99.07% of C++ online submissions for Partition Equal Subset Sum.
class Solution {
public:
    bool canPartition(vector<int>& nums) {
        int sum = 0;
        for (auto num: nums) sum += num;
        if (sum % 2) return false;

        int n = nums.size();
        int m = sum / 2;
        vector<bool> dp(m + 1, false);
        dp[0] = true;

        for (int num: nums) {
            // take care!!!!! order here
            // if (int j = num; j <= m) then the answer is wrong
            // count num more than once
            for (int j = m; j >= num; j--) {
                dp[j] = dp[j] || dp[j - num];
            }
        }
        return dp[m];
    }
};
```
