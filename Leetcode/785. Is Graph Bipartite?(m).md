class Solution {
public:
    bool dfs(vector<vector<int>>& graph, vector<char>& group, int v) {
        for (int u: graph[v]) {
            if (group[u] == 'C') {
                group[u] = (group[v] == 'A') ? 'B': 'A';
                if (!dfs(graph, group, u)) return false;
            } else if (group[u] == group[v]){
                return false;
            }
        }
        return true;
    }
    bool isBipartite(vector<vector<int>>& graph) {
        int n = graph.size();
        vector<char> group(n, 'C');
        
        for (int i = 0; i < n; i++) {
            if (group[i] != 'C') continue;
            group[i] = 'A';
            if (!dfs(graph, group, i)) return false;
        }
        return true;
    }
};
