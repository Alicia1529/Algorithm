class Logger:

    def __init__(self):
        self.history = dict()
        

    def shouldPrintMessage(self, timestamp: int, message: str) -> bool:
        if message not in self.history:
            self.history[message] = timestamp
            return True
        if timestamp >= self.history[message] + 10:
            self.history[message] = timestamp
            return True
        return False
