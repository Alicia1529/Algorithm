**Complexity** (grid `m × n`, word length `L`):
- **Time: O(m · n · 4ᴸ)** — DFS launches from every cell (`m · n` starts), and each
  path branches up to 4 directions for up to `L` levels (`4 · 3^(L-1)` ≈ 4ᴸ per start,
  since you never step back the way you came). Pruning on letter mismatch makes the real
  runtime far lower — most branches die after 1–2 characters.
- **Space: O(L)** — recursion depth is at most `L`. Holds for both versions: the in-place
  version marks/restores the board (O(1) extra), and the not-in-place version's `explored`
  set holds at most `L` cells (the current path) at once.

```python
# not-in-place solution — uses an `explored` set instead of mutating the board
class Solution:
    def exist(self, board: List[List[str]], word: str) -> bool:
        m, n = len(board), len(board[0])
        DIRECTIONS = [[1, 0], [-1, 0], [0, 1], [0, -1]]
        self.can_found = False

        def dfs(i, j, idx, explored):
            if i < 0 or i >= m or j < 0 or j >= n:
                return
            if (i, j) in explored:
                return
            if word[idx] != board[i][j]:        # current cell must match first
                return
            if idx == len(word) - 1:            # matched the LAST char → done
                self.can_found = True
                return

            explored.add((i, j))
            for x, y in DIRECTIONS:
                dfs(i + x, j + y, idx + 1, explored)
            explored.remove((i, j))

        for i in range(m):
            for j in range(n):
                if not self.can_found:
                    explored = set()
                    dfs(i, j, 0, explored)
        return self.can_found
```

```python
            found = False
            for x, y in directions:
                if backtrack(i + x, j + y, idx + 1):
                    found = True
                    break
            board[i][j] = tmp
            return found

        for i in range(m):
            for j in range(n):
                if backtrack(i, j, 0):
                    return True
        return False
```

```python
class Solution:
    def exist(self, board: List[List[str]], word: str) -> bool:
        self.m = len(board)
        self.n = len(board[0])
        self.word = word
        self.board = board

        for i in range(self.m):
            for j in range(self.n):                
                if self.backtrack(i, j, 0):
                    return True
        return False

    def backtrack(self, i, j, idx):
        if idx == len(self.word): return True

        if i < 0 or i >= self.m or j < 0 or j >= self.n: return False

        if self.board[i][j] != self.word[idx]: return False
        
        self.board[i][j] = "*"

        ret = self.backtrack(i - 1, j, idx + 1) or self.backtrack(i, j - 1, idx + 1) or self.backtrack(i + 1, j, idx + 1) or self.backtrack(i, j + 1, idx + 1)

        if ret: return True

        self.board[i][j] = self.word[idx]
        return False
```

```c++
class Solution {
public:
    int R, C;
    bool existImpl(vector<vector<char>>& board, int i, int j, string& word, int idx, bool *visited) {
        if (idx == word.size()) return true;
        if (i < 0 || i >= R || j < 0 || j >= C) return false;
        if (visited[i * C + j]) return false;
        if (board[i][j] != word[idx])
            return false;
        visited[i * C + j] = true;
        bool ret = existImpl(board, i + 1, j, word, idx + 1, visited) || existImpl(board, i, j + 1, word, idx + 1, visited) || existImpl(board, i - 1, j, word, idx + 1, visited) || existImpl(board, i, j - 1, word, idx + 1, visited);
        if (ret) return true;
        visited[i * C + j] = false;
        return false;
    }
    bool exist(vector<vector<char>>& board, string word) {
        R = board.size();
        C = board[0].size();
        bool visited[R * C];
        memset(visited, 0, sizeof(visited));
        for (int i = 0; i < R; i++) {
            for (int j = 0; j < C; j++) {
                if (existImpl(board, i, j, word, 0, visited))
                    return true;
            }
        }
        return false;
    }
};
```
