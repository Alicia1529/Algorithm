class Solution {
public:
    bool isvalid(unordered_map<char, int>& map) {
        for (auto itor: map) {
            if (itor.second != 0) return false; 
        }
        return true;
    }
    vector<int> findAnagrams(string s, string p) {
        vector<int> ans;
        int m = s.size(), n = p.size();
        if (m < n) return ans;

        unordered_map<char, int> map;
        for (auto chr: p) map[chr]++;
        
        for (int i = 0; i < n; i++) {
            if (map.find(s[i]) != map.end()) map[s[i]]--;
        }
        if (isvalid(map)) ans.push_back(0);
        for (int i = n; i < m; i++) {
            if (map.find(s[i - n]) != map.end()) map[s[i - n]]++;
            if (map.find(s[i]) != map.end()) map[s[i]]--;
            if (isvalid(map)) ans.push_back(i - n + 1);
        }
        return ans;
    }
};
