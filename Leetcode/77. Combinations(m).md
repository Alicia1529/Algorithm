class Solution:
    def combine(self, n: int, k: int) -> List[List[int]]:
        result = []
        path = []

        def backtrack(start):
            if len(path) == k:
                result.append(path[:])
                return
            # prune: stop if not enough numbers left to fill path
            for i in range(start, n - (k - len(path)) + 2):
                path.append(i)
                backtrack(i + 1)
                path.pop()

        backtrack(1)
        return result

class Solution {
public:
    vector<vector<int>> ans;
    vector<vector<int>> combine(int n, int k) {
        if (k <= 0 || n <= 0) return ans;
        vector<int> track;
        backtrack(1, n, k, track);
        return ans;
    }
    void backtrack(int start, int end, int remaining, vector<int>& curr) {
        if (remaining == 0) {
            ans.push_back(curr);
            return;
        }
        for (int i = start; i <= end; i++) {
            curr.push_back(i);
            backtrack(i + 1, end, remaining - 1, curr);
            curr.pop_back();
        }
    }
};

        if (track.size() == k) {
            ans.push_back(track);
            return;
        }
        for (int i = idx; i <= n; i++) {
            track.push_back(i);
            backtrack(track, n, k, i+1);
            track.pop_back();
        }
        
    }
    vector<vector<int>> combine(int n, int k) {
        vector<int> track;
        backtrack(track, n, k, 1);

    }
