// Runtime: 4 ms, faster than 92.15% of C++ online submissions for Best Time to Buy and Sell Stock IV.
// Memory Usage: 11.7 MB, less than 37.67% of C++ online submissions for Best Time to Buy and Sell Stock IV.
class Solution {
public:
    int maxProfit(int k, vector<int>& prices) {
        if (prices.size() <= 1 || k == 0) return 0;
        int n = prices.size();
        // dp[days][num operations]
        int buy[n][k + 1], sell[n][k + 1];
        memset(buy, 0, sizeof(buy));
        memset(sell, 0, sizeof(sell));

        // on day 0, for all non-zero # of transaction
        for (int j = 1; j <= k; ++j) {
            buy[0][j] = -prices[0]; // we can make a purchase on day 0
        }
        
        for (int i = 1; i < n; i++) {
            for (int j = 1; j <= k; j++) {
                buy[i][j] = max(buy[i - 1][j], sell[i - 1][j - 1] - prices[i]);
                sell[i][j] = max(sell[i - 1][j], buy[i - 1][j] + prices[i]);           
            }
        }
        return sell[n - 1][k];
    }
};


// 	Time Limit Exceeded
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        int ans = 0;
        int first = 0;
        int l1 = 0;
        int n = prices.size();
        
        for (int i = 0; i < n; i++) {
            if (prices[i] < prices[l1]) {
                l1 = i;
            } else {
                first = max(first, prices[i] - prices[l1]);
            }

            int l2 = i + 1;
            int second = 0;
            for (int j = i + 1; j < n; j++) {
                if (prices[j] < prices[l2]) {
                    l2 = j;
                } else {
                    second = max(second, prices[j] - prices[l2]);
                }
            }
            if (first + second > ans) ans = first + second;
        }
        return ans;
    }
};
