class Node:
    def __init__(self, value, prev_node=None, next_node=None):
        self.value = value
        self.prev = prev_node
        self.next = next_node 

class MyLinkedList:
    def __init__(self):
        self.size = 0
        self.tail = Node(-1)
        self.head = Node(-1)
        self.head.next = self.tail
        self.tail.prev = self.head

    def get(self, index: int) -> int:
        if index < 0 or index >= self.size:
            return -1
        # choose the fastest way: to move from the head
        # or to move from the tail
        if 2 * index <= self.size:
            curr = self.head
            for _ in range(index + 1):
                curr = curr.next
        else:
            curr = self.tail
            for _ in range(self.size - index):
                curr = curr.prev
        return curr.value
        

    def addAtHead(self, val: int) -> None:
        node = Node(val, self.head, self.head.next)
        self.head.next.prev = node
        self.head.next = node
        self.size+=1

    def addAtTail(self, val: int) -> None:
        node = Node(val, self.tail.prev, self.tail)
        self.tail.prev.next = node
        self.tail.prev = node
        self.size += 1

    def addAtIndex(self, index: int, val: int) -> None:
        if index > self.size:
            return
        flag = self.head
        for i in range(index):
            flag = flag.next
        node = Node(val, flag, flag.next)
        flag.next.prev = node
        flag.next = node
        self.size += 1
        # self.debug()

    def deleteAtIndex(self, index: int) -> None:
        # If the index is invalid, do nothing
        if index < 0 or index >= self.size:
            return
        flag = self.head
        for i in range(index):
            flag = flag.next
        # Delete pred.next 
        self.size -= 1    
        deleted = flag.next
        deleted.next.prev = flag
        flag.next = deleted.next
        
    def debug(self):
        flag = self.head
        while flag is not None:
            print(flag.value, end=",")
            flag = flag.next  
        print()
