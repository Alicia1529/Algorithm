/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };
 */
class Solution {
public:
    TreeNode* ans = nullptr;

    int dfs(TreeNode* node, TreeNode* p, TreeNode* q) {
        if (node == nullptr) return 0;
        int mid = 0;
        if (node == p || node == q)
            mid = 1;
        int left = dfs(node->left, p, q);
        int right = dfs(node->right, p, q);
        if (left + right + mid == 2 && ans == nullptr)
            ans = node;
        return left + right + mid;
        
    }
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        dfs(root, p, q);
        return ans;
    }
};

            if root is None: return 0
            mid = 0
            if root is p or root is q: mid = 1
            left = findPathToNode(root.left)

            if left + right + mid == 2 and self.ans is None:
                self.ans = root
            return left + right + mid;

        findPathToNode(root)
        return self.ans

// Runtime: 16 ms, faster than 68.08% of C++ online submissions for Lowest Common Ancestor of a Binary Tree.
// Memory Usage: 17.4 MB, less than 11.94% of C++ online submissions for Lowest Common Ancestor of a Binary Tree.
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };
 */
class Solution {
public:
    bool findNode(TreeNode* node, TreeNode* target, vector<TreeNode*>& l) {
        if (node == nullptr) return false;
        if (node == target) {
            l.push_back(node);
            return true;
        };
        bool ans = findNode(node->left, target, l) || findNode(node->right, target, l);
        if (ans == true)
            l.push_back(node);
        return ans;
    }
    TreeNode* lowestCommonAncestor(TreeNode* root, TreeNode* p, TreeNode* q) {
        vector<TreeNode*> l1;
        vector<TreeNode*> l2;
        findNode(root, p, l1);
        findNode(root, q, l2);
        TreeNode* ans = root;
        while (!l1.empty() && !l2.empty() && l1.back() == l2.back()) {
            ans = l1.back();
            l1.pop_back();
            l2.pop_back();
        }
        return ans;
    }
};

class Solution:
    def lowestCommonAncestor(self, root: 'TreeNode', p: 'TreeNode', q: 'TreeNode') -> 'TreeNode':
        # Find path from root to p, q
        def findPathToNode(root, node, track):
            if root is None: return False
            if root == node:
                track.append(root)
                return True
            
            ans = findPathToNode(root.left, node, track) or findPathToNode(root.right, node, track)
            if ans: track.append(root)
            return ans

        path_to_p, path_to_q = [], []
        findPathToNode(root, p, path_to_p)
        findPathToNode(root, q, path_to_q)
        

        # Find first place where the two node diverge
        res = root
        while len(path_to_p) and len(path_to_q) and path_to_p[-1] == path_to_q[-1]:
            res = path_to_p[-1]
            path_to_p.pop()
            path_to_q.pop()
        return res
