// how to consider the last few cases
/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */

// one pass
class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        ListNode* p1 = head;
        ListNode* p2 = head;
        while (n--)
            p1 = p1->next;
        if (!p1)
            return p2->next;
        while (p1->next) {
            p1 = p1->next;
            p2 = p2->next;
        }
        p2->next = p2->next->next;
        return head;
    }
};
// two passes
class Solution {
public:
    ListNode* removeNthFromEnd(ListNode* head, int n) {
        int len = 0;
        ListNode* front = head;
        while (front) {
            len++;
            front = front->next;
        }
        front = head;
        for (int i = 0; i < len - n - 1; i++) {
            front = front->next;
        }
        if (front == head && n == len)
            return head->next;
        if (front == head && front->next == nullptr)
            return head->next; 
        front->next = front->next->next;
        return head;
    }
};

class Solution:
    def removeNthFromEnd(self, head: Optional[ListNode], n: int) -> Optional[ListNode]:
        # Find out the length of the linked list: K
        # Find the distance from the head to the node to be removed: L - N
        # Iterate (L-N) steps from head and remove it. 
        size = 0
        curr = head
        while curr:
            size += 1
            curr = curr.next
        dummy = ListNode(0, head)
        curr = dummy
        for i in range(size-n):
            curr = curr.next
        curr.next = curr.next.next
        return dummy.next
