class Codec {
public:

    // Encodes a tree to a single string.
    string serialize(TreeNode* root) {
        queue<TreeNode*> q;
        q.push(root);
        string ret = "[";
        while (!q.empty()) {
            string line = "";
            bool valid = false;
            int size = q.size();
            
            for (int i = 0; i < size; i++) {
                TreeNode* node = q.front();
                q.pop();

                if (node == nullptr) {
                    line += "null,";
                } else {
                    valid = true;
                    line += to_string(node->val);
                    line += ',';
                    q.push(node->left);
                    q.push(node->right);
                }
            }

            if (valid || ret.size() == 1) {
                ret += line;
            }
        }
        ret = ret.substr(0, ret.size() - 1);
        ret += ']';
        return ret;
    }

    // Decodes your encoded data to tree.
    TreeNode* deserialize(string data) {
        if (data.size() == 2) return nullptr;
        
        TreeNode* node;
        queue<TreeNode*> q;
        int start = 1;
        for (int curr = start; curr < data.size(); curr++) {
            if (data[curr] != ',' && data[curr] != ']') continue;
            if (data.substr(start, curr - start) == "null") {
                node = nullptr;
            } else if (start == curr) {
                continue;
            } else {
                node = new TreeNode(stoi(data.substr(start, curr - start)));
            }
            q.push(node);
            start = curr + 1;
        }
        
        
        TreeNode* root = q.front();
        q.pop();
        queue<TreeNode*> level;
        level.push(root);
        while (!q.empty()) {
            int size = level.size();
            while (size && !q.empty()) {
                node = q.front();
                level.front()->left = node;
                if (node) level.push(node);
                q.pop();
                if (!q.empty()) {
                    node = q.front();
                    level.front()->right = node;
                    if (node) level.push(node);
                    q.pop();
                }
                level.pop();
                size--;
            }
            
        }
        return root;
    }
};
