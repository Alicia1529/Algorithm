// Memory Usage: 23.3 MB, less than 70.20% of C++ online submissions for Max Area of Island.
class Solution {
public:
    int local_area = 0;
    int ans = 0;
    int m;
    int n;
    
    void dfs(vector<vector<int>>& grid, int i, int j) {
        if (i < 0 || i >= m || j < 0 || j >= n) return;
        if (grid[i][j] == 0) return;
        local_area++;
        grid[i][j] = 0;
        dfs(grid, i + 1, j);
        dfs(grid, i - 1, j);
        dfs(grid, i, j + 1);
        dfs(grid, i, j - 1);
    }

    int maxAreaOfIsland(vector<vector<int>>& grid) {
        m = grid.size();
        n = grid[0].size();
        
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 1) {
                    local_area = 0;
                    dfs(grid, i, j);
                    ans = max(ans, local_area);
                }
            }
        }
        return ans;
    }
};

class Solution {
public:
    int m, n;
    int BFS(vector<vector<int>>& grid, int i, int j) {
        int ans = 1;
        grid[i][j] = 0;
        if (0 <= i - 1 && grid[i - 1][j]) ans += BFS(grid, i - 1, j);
        if (i + 1 < m && grid[i + 1][j]) ans += BFS(grid, i + 1, j);
        if (0 <= j - 1 && grid[i][j - 1]) ans += BFS(grid, i, j - 1);
        if (j + 1 < n && grid[i][j + 1]) ans += BFS(grid, i, j + 1);
        return ans;
    }
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        m = grid.size();
        n = grid[0].size();
        int ans = 0;
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] == 0) continue;
                ans = max(ans, BFS(grid, i, j));
            }
        }
        return ans;
    }
};
