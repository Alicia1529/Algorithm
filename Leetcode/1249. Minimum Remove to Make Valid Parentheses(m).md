string minRemoveToMakeValid(string s) {
  stack<int> st;
  for (auto i = 0; i < s.size(); ++i) {
    if (s[i] == '(') st.push(i);
    if (s[i] == ')') {
      if (!st.empty()) st.pop();
      else s[i] = '*';
    }
  }
  while (!st.empty()) {
    s[st.top()] = '*';
    st.pop();
  }
  s.erase(remove(s.begin(), s.end(), '*'), s.end());
  return s;
}

class Solution:
    def minRemoveToMakeValid(self, s: str) -> str:
        total_left = 0
        extra_left = 0
        candidates = []
        for letter in s:
            if letter == "(":
                total_left += 1
                extra_left += 1
            elif letter == ")":
                extra_left -= 1
            if extra_left >= 0:
                candidates.append(letter)
            else:
                extra_left = 0
        ans = []
        valid_left = total_left - extra_left
        for c in candidates:
            if c == "(" and valid_left > 0:
                ans.append(c)
                valid_left -= 1
            elif c != "(":
                ans.append(c)
        return "".join(ans)
