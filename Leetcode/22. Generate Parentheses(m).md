### Python
```python
class Solution:
    def generateParenthesis(self, n: int) -> List[str]:
        result = []
        path = []

        def backtrack(open_left, close_left):
            if open_left == 0 and close_left == 0:
                result.append("".join(path))
                return
            if open_left > 0:
                path.append('(')
                backtrack(open_left - 1, close_left)
                path.pop()
            if close_left > open_left:
                path.append(')')
                backtrack(open_left, close_left - 1)
                path.pop()

        backtrack(n, n)
        return result
```

### C++
```c++
class Solution {
public:
    vector<string> ans;
    int helperfunc(string s, int num_left, int num_right) {
        if (num_left < 0 || num_right < 0) return 0;
        if (num_left == 0 && num_right == 0) {
            ans.push_back(s);
            return 0;
        }
        helperfunc(s + "(", num_left - 1, num_right + 1);
        helperfunc(s + ")", num_left, num_right - 1);
        return 0;
    }
    vector<string> generateParenthesis(int n) {
        string s = "(";
        int num_left = n - 1;
        int num_right = 1;
        helperfunc(s + "(", num_left - 1, num_right + 1);
        helperfunc(s + ")", num_left, num_right - 1);
        return ans;
        
    }
};
```
