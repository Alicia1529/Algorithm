class Solution {
public:
    vector<string> emails;
    string email_id;
    void dfs(unordered_map<string, unordered_set<string>>& graph, unordered_map<string, bool>& visited, string v) {
        if (visited[v]) return;
        visited[v] = true;
        if (v.find('@') == v.npos) {
            email_id = v;
        } else {
            emails.push_back(v);
        }
        for (auto& u: graph[v]) {
            dfs(graph, visited, u);
        }
    }
    vector<vector<string>> accountsMerge(vector<vector<string>>& accounts) {
        vector<vector<string>> ans;
        unordered_map<string, unordered_set<string>> graph;
        unordered_map<string, bool> visited;
        for (int i = 0; i < accounts.size(); i++) {
            string id = to_string(i);
            for (int j = 1; j < accounts[i].size(); j++) {
                graph[id].insert(accounts[i][j]);
                graph[accounts[i][j]].insert(id);
                visited[accounts[i][j]] = false;
            }
            visited[id] = false;
        }
        
        for (auto& node: graph) {
            string v = node.first;
            if (!visited[v]) {
                email_id = "";
                emails.clear();
                dfs(graph, visited, v);
                string name = accounts[stoi(email_id)][0];
                sort(emails.begin(), emails.end());
                emails.insert(emails.begin(), name);
                ans.emplace_back(emails);
            }
        }
        return ans;
    }
};

class Solution:
    def accountsMerge(self, accounts: List[List[str]]) -> List[List[str]]:
        email2name = dict() # key: email, value: name
        graph = defaultdict(set) # key: email, value: all emails that associated with it
        
        for acc in accounts:
            name = acc[0]
            for email in acc[1:]:
                email2name[email] = name
                graph[email] = graph[email].union(set(acc[1:]))
                
        def dfs(email, addresses, graph, visited):
            for sub in graph[email]:
                if sub not in visited:
                    visited.add(sub)
                    addresses.add(sub)
                    dfs(sub, addresses, graph, visited)
            
        
        visited = set()
        ans = []
        for email in graph.keys():
            if email not in visited:
                name = email2name[email]
                addresses = set()
                visited.add(email)
                addresses.add(email)
                dfs(email, addresses, graph, visited)
                addresses = sorted(list(addresses))
                ans.append([name] + addresses)
        return ans
