class Solution {
public:
     TreeNode* bstFromPreorderUtil(vector<int> &preorder, int &i, int bound){
        if(i == preorder.size() || preorder[i] > bound) return NULL;
        TreeNode* root = new TreeNode(preorder[i++]);
        root->left = bstFromPreorderUtil(preorder, i, root->val);
        root->right = bstFromPreorderUtil(preorder, i, bound);
        return root;
    }
    
    TreeNode* bstFromPreorder(vector<int>& preorder) {
        int i = 0;
        return bstFromPreorderUtil(preorder, i, INT_MAX);
    }
};

class Solution {
public:
    TreeNode* helper(vector<int>& preorder, int l, int r) {
        if (r < l) return nullptr;
        if (r == l) return new TreeNode(preorder[l]);
        
        int right_start = l;
        while (right_start <= r && preorder[right_start] <= preorder[l]) right_start++;
        TreeNode* left = helper(preorder, l + 1, right_start - 1);
        TreeNode* right = helper(preorder, right_start, r);
        return new TreeNode(preorder[l], left, right);
    }
    TreeNode* bstFromPreorder(vector<int>& preorder) {
        return helper(preorder, 0, preorder.size() - 1);
    }
};
