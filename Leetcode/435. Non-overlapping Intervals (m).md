### Approach #2 Using DP based on starting point[Accepted]

Time complexity : N^2

Space complexity : N
```c++
// Runtime: 792 ms, faster than 5.00% of C++ online submissions for Non-overlapping Intervals.
// Memory Usage: 10.3 MB, less than 37.79% of C++ online submissions for Non-overlapping Intervals.
class Solution {
public:
    int eraseOverlapIntervals(vector<vector<int>>& intervals) {
        int size = intervals.size();
        vector<int> dp(size, 1);
        int max_intervals = 1;
        
        sort(intervals.begin(), intervals.end());
        
        for (int i = 1; i < size; i++) {
            for (int j = 0; j < i; j++) {
                if (intervals[i][0] >= intervals[j][1])
                    dp[i] = max(dp[i], dp[j] + 1);
            }
            if (dp[i] > max_intervals) max_intervals = dp[i];
        }
        return size - max_intervals;
        
    }
};
```

### Approach #4 Using Greedy Approach based on ending points [Accepted]
Time complexity : NlogN

Space complexity : 1
```c++
// Runtime: 12 ms, faster than 93.54% of C++ online submissions for Non-overlapping Intervals.
// Memory Usage: 10.3 MB, less than 59.09% of C++ online submissions for Non-overlapping Intervals.
class Solution {
public:
    int eraseOverlapIntervals(vector<vector<int>>& intervals) {
        int size = intervals.size();
        int prev = 0, cnt = 1;
        
        sort(intervals.begin(), intervals.end(), [](const vector<int>& v1, const vector<int>& v2) {
            return (v1[1] != v2[1]) ? v1[1] < v2[1]: v1[0] < v2[0];
        });
        
        for (int i = 1; i < size; i++) {
            if (intervals[i][0] >= intervals[prev][1]) {
                prev = i;
                cnt++;
            }
        }
        return size - cnt;
        
    }
};
```
