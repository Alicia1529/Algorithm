// Memory Usage: 17.9 MB, less than 38.50% of C++ online submissions for Asteroid Collision.
class Solution {
public:
    vector<int> asteroidCollision(vector<int>& asteroids) {
        stack<int> stk;
        vector<int> ans;
        for (int i = 0; i < asteroids.size(); i++) {
            if (asteroids[i] > 0) {
                stk.push(asteroids[i]);
            } else {
                while (stk.size() && stk.top() > 0 && stk.top() < abs(asteroids[i])) {
                    stk.pop();
                }
                if (stk.empty() || stk.top() < 0) {
                    stk.push(asteroids[i]);
                } else if (stk.top() == abs(asteroids[i])){
                    stk.pop();
                }
            }
        }
        while (stk.size()) {
            ans.push_back(stk.top());
            stk.pop();
        }
        reverse(ans.begin(), ans.end());
        return ans;
    }
};

class Solution {
public:
    vector<int> asteroidCollision(vector<int>& asteroids) {
        for (int i = 0; i < asteroids.size(); i++) {
            if (asteroids[i] >= 0) continue;
            for (int j = i - 1; j >= 0; j--) {
                if (asteroids[j] <= 0) continue;
                if (asteroids[j] == abs(asteroids[i])) {
                    asteroids[i] = 0;
                    asteroids[j] = 0;
                    break;
                } else if (asteroids[j] > abs(asteroids[i])) {
                    asteroids[i] = 0;
                    break;
                } else {
                    asteroids[j] = 0;
                }
            }
        }
        vector<int> ans;
        for (int i = 0; i < asteroids.size(); i++) {
            if (asteroids[i] != 0) ans.push_back(asteroids[i]);
        }
        return ans;
    }
};
