// Runtime: 36 ms, faster than 79.11% of C++ online submissions for Construct Binary Tree from String.
// Memory Usage: 26.3 MB, less than 60.31% of C++ online submissions for Construct Binary Tree from String.
class Solution {
public:
    TreeNode* str2treeInternal(string& s, int start, int end) {
        if (start >= end) return nullptr;
        
        int l = start;
        int cnt = 0;
        while (l < end && s[l] != '(') l++;
        int val = stoi(s.substr(start, l - start));

        int r = l;
        while (r < end && (r == l || cnt)) {
            if (s[r] == '(') cnt++;
            if (s[r] == ')') cnt--;
            r++;
        }

        TreeNode* left = str2treeInternal(s, l + 1, r - 1);
        TreeNode* right = str2treeInternal(s, r + 1, end - 1);
        return new TreeNode(val, left, right);
    }
    TreeNode* str2tree(string s) {
        return str2treeInternal(s, 0, s.size());
    }
};
