    def topKFrequent(self, nums: List[int], k: int) -> List[int]:

        cnt = defaultdict(int)
        for num in nums:
            cnt[num] += 1
        kv = list(cnt.items())
        ks = len(kv) - k

        def partition(low, high):
            i = low - 1
            pivot = kv[high][1]
            for j in range(low, high):
                if kv[j][1] <= pivot:
                    i += 1
                    kv[i], kv[j] = kv[j], kv[i]
            kv[i + 1], kv[high] = kv[high], kv[i + 1]
            return i + 1
        
        def select(low, high):
            if low > high: return
            p = partition(low, high)
            if p < ks:
                select(p + 1, high)
            else:
                select(low, p - 1)
        
        select(0, len(kv) - 1)
        ans = map(lambda x: x[0], kv[ks:])
        return list(ans)

---

Three heap/quickselect approaches and their trade-offs. All share the O(N) frequency-count
build; let D = number of *distinct* elements (D ≤ N).

| Method | Time | Extra space (beyond counter) | Top-K ordered? | When |
|---|---|---|---|---|
| 1 — size-K min-heap | O(N log K) | **O(K)** | yes | small K, streaming, memory-tight |
| 2 — heapify + pop K | O(N + K log N) | O(N) | yes | K constant-ish; fewer comparisons |
| 3 — quickselect (random pivot) | **O(N) avg**, O(N²) worst | O(N) array + O(log N) recursion | **no** | best average time, K large |

> Note: including the unavoidable counter, all three are O(N) **total** space. The O(K) vs O(N)
> above measures only the *extra* structure each builds. Quickselect's array is O(N) because a
> dict isn't index-swappable — `list(cnt.items())` reuses the counts, just in a swappable container.

```python
from collections import defaultdict
import heapq
import random


class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        hashmap = defaultdict(int)            # shared O(N) frequency count
        for num in nums:
            hashmap[num] += 1
        ans = []

        # Method 1: size-K min-heap -> O(N log K), space O(K)
        pq = []
        for num, v in hashmap.items():
            if len(pq) < k:
                heapq.heappush(pq, (v, num))
            elif pq[0][0] < v:                # new count beats the smallest in top-K
                heapq.heappop(pq)
                heapq.heappush(pq, (v, num))
        while pq:
            ans.append(heapq.heappop(pq)[1])
        return ans
```

        pq = [(-v, num) for num, v in hashmap.items()]   # negate → max-heap
        heapq.heapify(pq)                                 # O(N), NOT N log N
        while pq and k:                                   # K pops × O(log N)
            ans.append(heapq.heappop(pq)[1])
            k -= 1
        return ans

```python
class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        hashmap = defaultdict(int)
        for num in nums:
            hashmap[num] += 1

        # Method 3: quickselect -> O(N) average (O(N^2) worst without random pivot)
        array = list(hashmap.items())         # reuse counts in a swappable list

        def partition(l, r):                  # inclusive
            rand = random.randint(l, r)       # ← random pivot avoids O(N^2) worst case
            array[rand], array[r] = array[r], array[rand]
            idx = l
            pivot = array[r]                  # (number, count)
            for i in range(l, r):
                if array[i][1] < pivot[1]:    # move less-frequent to the left
                    array[idx], array[i] = array[i], array[idx]
                    idx += 1
            array[idx], array[r] = array[r], array[idx]   # pivot to its final place
            return idx

        def quickselect(l, r, k_smallest):    # inclusive
            if l >= r:
                return
            p = partition(l, r)
            if k_smallest == p:
                return
            elif k_smallest < p:
                quickselect(l, p - 1, k_smallest)
            else:
                quickselect(p + 1, r, k_smallest)

        n = len(array)
        quickselect(0, n - 1, n - k)          # the K most frequent end up on the right
        return [num for num, _ in array[n - k:]]
```
