class Solution:
    def maximumSwap(self, num: int) -> int:
        numstr = list(str(num))
        numint = list(map(lambda x: int(x), numstr))
        size = len(numstr)

        def findMax(array, start):
            idx, maxValue = start, array[start]
            for i in range(start + 1, len(array)):
                if array[i] >= maxValue:
                    maxValue = array[i]
                    idx = i
            return idx, maxValue
    
        for i in range(size):
            idx, maxValue = findMax(numint, i)
            if maxValue == numint[i]: continue
            numstr[i], numstr[idx] = numstr[idx], numstr[i]
            break
        return int("".join(numstr))


class Solution:
    def maximumSwap(self, num: int) -> int:
        
        # 1. convert to string, and then split
        num = list(str(num))

        # 2. find the largest number, switch its position with head
        
        # the index of value that is larger than num[i]
        # cannot find one: return i, otherwise j
        def findNextLargest(num, i):
            currMax = int(num[i])
            idx = i
            for j in range(i + 1, len(num)):
                if int(num[j]) >= currMax:
                    idx = j
                    currMax = int(num[j])
            return idx
                
        for i in range(len(num)):
            idx = findNextLargest(num, i)
            # cannot find or equal
            if idx == i or num[idx] == num[i]: continue
            num[i], num[idx] = num[idx], num[i]
            return int("".join(num))
        return int("".join(num))
