// Runtime: 120 ms, faster than 66.61% of C++ online submissions for Count Good Nodes in Binary Tree.
// Memory Usage: 86.3 MB, less than 90.65% of C++ online submissions for Count Good Nodes in Binary Tree.
class Solution {
public:
    int total = 0;
    
    void traverse(TreeNode* node, int value) {
        if (node->val >= value) total++;
        if (node->left != nullptr)
            traverse(node->left, max(value, node->val));
        if (node->right != nullptr)
            traverse(node->right, max(value, node->val));
    }
    int goodNodes(TreeNode* root) {
        traverse(root, root->val);
        return total;
        
    }
};
