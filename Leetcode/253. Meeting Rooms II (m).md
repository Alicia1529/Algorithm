class Solution {
public:
    int minMeetingRooms(vector<vector<int>>& intervals) {
        priority_queue<int, vector<int>, greater<int>> pq;
        int ans = 0;

        std::sort(intervals.begin(), intervals.end());
        // std::sort(intervals.begin(), intervals.end(), [](const vector<int>& v1, const vector<int>& v2) {
        //     return (v1[0] <= v2[0]) ? true: false;
        // });

        for (vector<int>& interval: intervals) {
            while (!pq.empty() && pq.top() <= interval[0]) {
                pq.pop();
            }
            pq.push(interval[1]);
            if (pq.size() > ans) ans = pq.size();
        }
        return ans;
    }
};

import heapq
class Solution:
    def minMeetingRooms(self, intervals: List[List[int]]) -> int:
        if len(intervals) == 0:
            return 0
        intervals.sort()
        heap = [] # stores end times of active meetings
        for start, end in intervals:
            if heap and heap[0] <= start:
                heapq.heappop(heap) # reuse the room
            heapq.heappush(heap, end)
            # print("heap:", heap)
        return len(heap)


        
class Solution:
    def minMeetingRooms(self, intervals: List[List[int]]) -> int:
        #  0     5   10    15   20     25      30
        #  |------------------------------------|

        # 5-10, 15-20, 0-30
        intervals.sort()
        queue = []
        heapq.heapify(queue)
        
        ans = 0
        for interval in intervals:
            while len(queue) and queue[0] <= interval[0]:
                heapq.heappop(queue)
            heapq.heappush(queue, interval[1])
            if len(queue) > ans:
                ans = len(queue)
        return ans
