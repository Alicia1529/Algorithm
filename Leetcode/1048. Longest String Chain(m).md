// Runtime: 192 ms, faster than 21.47% of C++ online submissions for Longest String Chain.
// Memory Usage: 13.1 MB, less than 91.18% of C++ online submissions for Longest String Chain.
class Solution {
public:
    bool isPredecessor(vector<string>& words, int i, int j) {
        if (words[j].size() - words[i].size() != 1) return false;
        int similar = 0;
        for (int k1 = 0, k2 = 0; k1 < words[i].size() && k2 < words[j].size(); k1++, k2++) {
            if (words[i][k1] != words[j][k2]) {
                k1--;
            } else {
                similar++;
            }
        }
        return similar == words[i].size();
    }
    int longestStrChain(vector<string>& words) {
        sort(words.begin(), words.end(),[](const string& w1, const string& w2) {
            return w1.size() < w2.size();
        });
        int ans = 1;
        int n = words.size();
        vector<int> dp(n, 1);
        for (int i = 0; i < n; i++) {
            for (int j = i - 1; j >= 0; j--) {
                if (isPredecessor(words, j, i)) {
                    dp[i] = max(dp[i], dp[j] + 1);
                    ans = max(ans, dp[i]);
                }
            }
        }
        return ans;
    }
};
