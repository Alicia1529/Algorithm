### Recursion
```python
class Solution:
    def minMoves(self, target: int, maxDoubles: int) -> int:
        if target == 1: return 0
        if target % 2 == 1: return self.minMoves(target - 1, maxDoubles) + 1
        if maxDoubles: return self.minMoves(target // 2, maxDoubles - 1) + 1
        return target - 1
```
### Greedy
```python
class Solution:
    def minMoves(self, target: int, maxDoubles: int) -> int:
        res = 0
        while target > 1 and maxDoubles > 0:
            res += target % 2 + 1
            maxDoubles -= 1
            target >>= 1
        return res + target - 1
```
