Key insight: register the new node in the map **before** recursing into / enqueuing its neighbors. That's what stops cycles from looping forever — a node is cloned once, but each edge is wired as it's encountered.

In the BFS version, two jobs are kept separate:
- **Create** `nb`'s clone — once, the first time `nb` is seen (guarded by `if nb not in visited`)
- **Wire** the edge `cur → nb` — every time the edge is seen (outside the `if`)

The `append` is always safe because by the time it runs, `visited[nb]` is guaranteed to exist (the `if` just created it, or it existed already), and `visited[cur]` exists because a node only enters the queue right after being put in `visited`.

1. DFS

```python
"""
# Definition for a Node.
class Node:
    def __init__(self, val = 0, neighbors = None):
        self.val = val
        self.neighbors = neighbors if neighbors is not None else []
"""

class Solution:
    def cloneGraph(self, node: 'Node') -> 'Node':
        hashmap = dict()
        def helper(node):
            if node is None: return None
            if node in hashmap: return hashmap.get(node)
            hashmap[node] = Node(node.val)
            for neighbor in node.neighbors:
                hashmap[node].neighbors.append(helper(neighbor))
            return hashmap[node]
        return helper(node)


    def cloneGraph(self, node: 'Node') -> 'Node':
        if node is None: return None

        visited = dict()
        queue = deque([node])
        visited[node] = Node(node.val, [])

        while queue:
            n = queue.popleft()
            for neighbor in n.neighbors:
                if neighbor not in visited:
                    visited[neighbor] = Node(neighbor.val)
                    queue.append(neighbor)
                visited[n].neighbors.append(visited[neighbor])
        return visited[node]

```

class Solution {
public:
    unordered_map<Node*, Node*> visited;
    Node* cloneGraph(Node* node) {
        if (!node) return nullptr;
        if (visited.find(node) != visited.end())
            return visited[node];
        Node* cloned = new Node(node->val);
        visited[node] = cloned;
        for (auto &child: node->neighbors) {
            cloned->neighbors.push_back(cloneGraph(child));
        }
        return cloned;
    }
};

// Runtime: 8 ms, faster than 80.01% of C++ online submissions for Clone Graph.
// Memory Usage: 8.7 MB, less than 84.45% of C++ online submissions for Clone Graph.
class Solution {
public:

    Node* cloneGraph(Node* node) {
        if (!node) return nullptr;

        unordered_map<Node*, Node*> visited;
        queue<Node*> q;
        q.push(node);
        Node *cloned = new Node(node->val);
        visited[node] = cloned;
        
        while (!q.empty()) {
            auto n = q.front();
            q.pop();
            for (auto &neighbor: n->neighbors) {
                if (visited.find(neighbor) == visited.end()) {
                    visited[neighbor] = new Node(neighbor->val);
                    q.push(neighbor);
                }
                visited[n]->neighbors.push_back(visited[neighbor]);
            }
        }
        return visited[node];
    }
};
