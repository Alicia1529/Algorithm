class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        vector<int> ans(nums.size(), 1);
        for (int i = 1; i < nums.size(); i++) {
            ans[i] = ans[i - 1] * nums[i - 1];
        }
        int product = 1;
        for (int i = nums.size() - 2; i >= 0; i--) {
            product *= nums[i + 1];
            ans[i] *= product;
        }
        return ans;
        
    }
};

## Python

1. Two prefix/suffix arrays (inclusive products) — O(n) time, O(n) space

```python
class Solution:
    def productExceptSelf(self, nums: List[int]) -> List[int]:
        n = len(nums)
        l_array = [0] * n
        r_array = [0] * n
        for i in range(n):
            if i == 0:
                l_array[i] = nums[i]
                r_array[n - i - 1] = nums[n - i - 1]
            else:
                l_array[i] = l_array[i - 1] * nums[i]
                r_array[n - i - 1] = r_array[n - i] * nums[n - i - 1]
        ans = []
        r_array.append(1)                      # sentinel so r_array[i+1] works at the end
        for i in range(n):
            if i == 0:
                ans.append(r_array[i + 1])
            else:
                ans.append(r_array[i + 1] * l_array[i - 1])
        return ans
```

2. Running prefix/suffix scalars (exclusive products) — O(n) time, **O(1) extra space**

- Writing the prefix *before* folding in `nums[i]` makes `ans[i]` naturally exclude the
  current element → no `i±1` juggling, no `i == 0` special cases.
- Only the output array + two scalars; output doesn't count → hits the O(1)-space follow-up.

```python
class Solution:
    def productExceptSelf(self, nums: List[int]) -> List[int]:
        n = len(nums)
        ans = [1] * n

        prefix = 1                     # product of everything LEFT of i
        for i in range(n):
            ans[i] = prefix            # excludes nums[i]
            prefix *= nums[i]          # fold in for next index

        suffix = 1                     # product of everything RIGHT of i
        for i in range(n - 1, -1, -1):
            ans[i] *= suffix           # excludes nums[i]
            suffix *= nums[i]
        return ans
```
