```c++
class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        if (nums.size() == 0) return {-1, -1};

        int l = 0, r = nums.size() - 1;
        while (l <= r) {
            int m = (l + r) / 2;
            if (nums[m] < target) {
                l = m + 1;
            } else {
                r = m - 1;
            }
        }
        if (l >= nums.size() || nums[l] != target)
            return {-1, -1};
        r = l;
        while (r < nums.size() && nums[r] == nums[l])
            r++;
        return {l, r - 1};
    }
};

class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        int left, right;
        vector<int> miss(2, -1);
        if (nums.size() == 0) return miss;
        
        int lo = 0, hi = nums.size() - 1;
        while (lo <= hi) {
            int mid = lo + (hi - lo) / 2;
            if (nums[mid] < target) {
                lo = mid + 1;
            } else {
                hi = mid - 1;
            }
        }
        if (lo > nums.size() - 1 || nums[lo] != target)
            return miss;
        left = lo;
        lo = 0, hi = nums.size() - 1;
        while (lo <= hi) {
            int mid = lo + (hi - lo) / 2;
            if (target < nums[mid]) {
                hi = mid - 1;
            } else {
                lo = mid + 1;
            }
        }
        if (hi < 0 || nums[hi] != target)
            return miss;
        right = hi;
        return {left, right};
    }
};

        vector<int> ans;
        int front = 0, back = nums.size() - 1;
        while (front <= back) {
            int mid = (front +  back) / 2;
            if (nums[mid] == target) {
                int left = mid, right = mid;
                while (left >= 0 && nums[left] == target)
                    left--;
                while (right < nums.size() && nums[right] == target)
                    right++;
                ans.push_back(left + 1);
                ans.push_back(right - 1);
                return ans;
            }
            if (nums[mid] > target) {
                back = mid - 1;
            } else {
                front = mid + 1;
            }
        }
        ans.push_back(-1);
        ans.push_back(-1);
        return ans;
    }
```

```python
# Template A (bisect_left + bisect_right), half-open bounds.
# first = bisect_left  (first index nums[i] >= target)
# last  = bisect_right (first index nums[i] >  target), minus 1
class Solution:
    def searchRange(self, nums: List[int], target: int) -> List[int]:
        n = len(nums)
        # --- first position: first index with nums[i] >= target ---
        left_low, left_high = 0, n - 1
        while left_low < left_high:
            mid = (left_low + left_high) // 2
            if nums[mid] < target:
                left_low = mid + 1
            else:
                left_high = mid
        if left_low == n or nums[left_low] != target:
            return [-1, -1]
        # --- last position: first index with nums[i] > target, then step back ---
        right_low, right_high = left_low, n        # high = n, NOT n-1 (see below)
        while right_low < right_high:
            mid = (right_low + right_high) // 2
            if nums[mid] <= target:
                right_low = mid + 1
            else:
                right_high = mid
        return [left_low, right_low - 1]
```
- **Failing case that caught this:** `nums = [1]`, `target = 1` → returned `[0, -1]`
  instead of `[0, 0]`.
- **Root cause:** the *last-position* search is `bisect_right`, whose answer can be
  `n` (when target is the largest element — index "one past the end"). Initializing
  `right_high = n - 1` caps it below `n`, so `right_low` never advances and
  `right_low - 1` underflows to `-1`.
- **Rule:** when the valid answer includes "past the end" (`bisect_right` / any
  insert-position search), init `high = n`, not `n - 1`. The *first*-position search
  can use `n - 1` here only because the explicit `left_low == n` guard catches the
  overflow case.
