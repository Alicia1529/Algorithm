```c++
class Node {
public:
    char c;
    bool isWord;
    Node* children[26];
    Node(char _c) {
        c = _c;
        isWord = false;
        for (int i = 0; i < 26; i++) children[i] = nullptr;
    }
};

class WordDictionary {
public:
    /** Initialize your data structure here. */
    Node* root;

    WordDictionary() {
        root = new Node('\0');
    }
    
    void addWord(string word) {
        Node* curr = root;
        for (int i = 0; i < word.size(); i++) {
            char c = word[i];
            if (curr->children[c - 'a'] == nullptr) curr->children[c - 'a'] = new Node(c);
            curr = curr->children[c - 'a'];
        }
        curr->isWord = true;
    }
    
    bool searchRecursive(string& word, Node* node, int idx) {
        if (idx == word.size()) {
            if (node->isWord) return true;
            return false;   
        }
        
        char c = word[idx];
        if (c != '.') {
            if (node->children[c - 'a'] == nullptr) return false;
            return searchRecursive(word, node->children[c - 'a'], idx + 1);
        }
        bool isTrue = false;
        for (int i = 0; i < 26; i++) {
            if (node->children[i] == nullptr) continue;
            if (searchRecursive(word, node->children[i], idx + 1)) return true;
        }
        return false;
    }
    
    bool search(string word) {
        Node* curr = root;
        return searchRecursive(word, curr, 0);
    }
};
```

```python
class TrieNode:
    def __init__(self):
        self.children = {}      # char -> TrieNode
        self.is_word = False


class WordDictionary:
    def __init__(self):
        self.root = TrieNode()

    def addWord(self, word: str) -> None:
        node = self.root
        for c in word:
            if c not in node.children:
                node.children[c] = TrieNode()
            node = node.children[c]
        node.is_word = True

    def search(self, word: str) -> bool:
        def helper(idx, node):
            if idx == len(word):
                return node.is_word        # reached the end → must end a word here
            c = word[idx]
            if c == '.':                   # wildcard → try every child branch
                return any(helper(idx + 1, child) for child in node.children.values())
            if c not in node.children:
                return False
            return helper(idx + 1, node.children[c])

        return helper(0, self.root)
```

Iterative-until-dot variant — only recurses when it hits a `.`, so a word with no
wildcards is a plain O(L) walk with no recursion:

```python
class TrieNode:
    def __init__(self):
        self.children = dict()
        self.is_word = False


class WordDictionary:
    def __init__(self):
        self.root = TrieNode()

    def addWord(self, word: str) -> None:
        node = self.root
        for c in word:
            if c not in node.children:
                node.children[c] = TrieNode()
            node = node.children[c]
        node.is_word = True

    def search(self, word: str) -> bool:
        def helper(i, node):
            idx = i                        # cursor stays in sync with c
            for c in word[i:]:
                if c == '.':               # wildcard → fan out into every child
                    return any(helper(idx + 1, child) for child in node.children.values())
                else:
                    if c not in node.children:
                        return False
                    node = node.children[c]
                idx += 1
            return node.is_word
        return helper(0, self.root)
```
