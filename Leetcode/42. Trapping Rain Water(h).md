1. Using Dynamic Programming to find the highest bar size
```c++
// Runtime: 4 ms, faster than 93.89% of C++ online submissions for Trapping Rain Water.
// Memory Usage: 14.5 MB, less than 42.38% of C++ online submissions for Trapping Rain Water.
class Solution {
public:
    int trap(vector<int>& height) {
        int ans = 0;
        int size = height.size();
        vector<int> left_max(size, 0);
        vector<int> right_max(size, 0);
        for (int i = 1; i < size; i++) {
            left_max[i] = max(left_max[i - 1], height[i - 1]);
        } 
        
        for (int i = size - 2; i >= 0; i--) {
            right_max[i] = max(right_max[i + 1], height[i + 1]);
            ans += max(0, min(left_max[i], right_max[i]) - height[i]);
        }  
        return ans;
    }
};
```

2. Using stacks to find the highest bar size
```c++
// Runtime: 4 ms, faster than 93.89% of C++ online submissions for Trapping Rain Water.
// Memory Usage: 14.4 MB, less than 42.38% of C++ online submissions for Trapping Rain Water.
class Solution {
public:
    int trap(vector<int>& height) {
        int ans = 0;
        int size = height.size();
        stack<int> stack;
        for (int i = 0; i < size; i++) {
            while (!stack.empty() && height[i] > height[stack.top()]) {
                int top = stack.top();
                stack.pop();
                if (stack.empty()) break;
                int distance = i - stack.top() - 1;
                int bounded_height = min(height[stack.top()], height[i]) - height[top];
                ans += distance * bounded_height;
            }
            stack.push(i);
        } 
        return ans;
    }
};
```

3. Using 2 pointers
```c++
// Runtime: 4 ms, faster than 93.89% of C++ online submissions for Trapping Rain Water.
// Memory Usage: 14 MB, less than 99.78% of C++ online submissions for Trapping Rain Water.
class Solution {
public:
    int trap(vector<int>& height) {
        int ans = 0;
        int size = height.size();
        int left = 0, right = size - 1;
        int left_max = 0, right_max = 0;
        while (left < right) {
            if (height[left] < height[right]) {
                if (height[left] >= left_max) {
                    left_max = height[left];
                } else {
                    ans += left_max - height[left];
                }
                left++;
            } else {
                if (height[right] < right_max) {
                    ans += right_max - height[right];
                } else {
                    right_max = height[right];
                }
                right--;
            }
        }
        return ans;
    }
};
```

```python
class Solution:
    def trap(self, height: List[int]) -> int:
        size = len(height)
        left_bound = [0 for i in range(size)]
        right_bound = [0 for i in range(size)]
        for i in range(size):
            if i == 0 or height[i] > left_bound[i - 1]:
                left_bound[i] = height[i]
            else:
                left_bound[i] = left_bound[i - 1]
            j = size - i - 1
            if j == size - 1 or height[j] > right_bound[j + 1]:
                right_bound[j] = height[j]
            else:
                right_bound[j] = right_bound[j + 1]
        amt = 0
        for i in range(1, size - 1):
            gap = min(left_bound[i - 1], right_bound[i + 1]) - height[i]
            amt += max(gap, 0)
        return amt
```

4. Python DP — inclusive prefix/suffix maxes (no edge-case guards), O(n) time / O(n) space

- `left_max[i]` / `right_max[i]` are *inclusive* of `i`, so `min(left, right) >= height[i]`
  always holds → no `max(..., 0)` clamp and no asymmetric `i > 0` / `i > 1` index guards.

```python
class Solution:
    def trap(self, height: List[int]) -> int:
        n = len(height)
        if n == 0:
            return 0
        left_max = [0] * n
        right_max = [0] * n

        left_max[0] = height[0]
        for i in range(1, n):
            left_max[i] = max(left_max[i - 1], height[i])

        right_max[n - 1] = height[n - 1]
        for i in range(n - 2, -1, -1):
            right_max[i] = max(right_max[i + 1], height[i])

        res = 0
        for i in range(n):
            res += min(left_max[i], right_max[i]) - height[i]
        return res
```

5. Python two pointers — **O(1) space** (optimal)

- Only need to know which side is *shorter*: water at a position is bounded by the smaller
  side-max. If `height[l] < height[r]`, `left_max` is the limiting wall for `l` (a bar at
  `r >= height[l]` holds the right), so settle `l` and move inward.

```python
class Solution:
    def trap(self, height: List[int]) -> int:
        l, r = 0, len(height) - 1
        left_max = right_max = 0
        res = 0
        while l < r:
            if height[l] < height[r]:
                left_max = max(left_max, height[l])
                res += left_max - height[l]
                l += 1
            else:
                right_max = max(right_max, height[r])
                res += right_max - height[r]
                r -= 1
        return res
```

6. Python monotonic stack — fills water **horizontally** (slab by slab), O(n) time / O(n) space

- Keep indices in *decreasing* height (bars waiting for a taller right wall). When the current
  bar is taller than the stack top, that top is a valley floor that just got its right wall →
  pop it and add `width * bounded` for the slab between the two walls.
- Same machinery as Largest Rectangle in Histogram / Next Greater Element (Day 3).

```python
class Solution:
    def trap(self, height: List[int]) -> int:
        stack = []          # indices, heights decreasing from bottom to top
        res = 0
        for i, h in enumerate(height):
            while stack and height[stack[-1]] < h:
                bottom = stack.pop()          # valley floor
                if not stack:                 # no left wall → water spills off the left
                    break
                left = stack[-1]              # left wall
                width = i - left - 1          # gap between the two walls
                bounded = min(height[left], h) - height[bottom]   # slab height
                res += width * bounded
            stack.append(i)
        return res
```
