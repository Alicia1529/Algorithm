class Solution {
public:
    vector<int> findBuildings(vector<int>& heights) {
        stack<int> walls;
        for (int i = heights.size() - 1; i >= 0; i--) {
            if (walls.empty() || heights[i] > heights[walls.top()]) {
                walls.push(i);
            }
        }
        vector<int> ans;
        while (walls.size()) {
            ans.push_back(walls.top());
            walls.pop();
        }
        return ans;
    }
};

class Solution:
    def findBuildings(self, heights: List[int]) -> List[int]:
        stack = list() # indices with height[idx] in increasing order
        for i in range(len(heights)):
            while len(stack) and heights[stack[-1]] <= heights[i]:
                stack.pop()
            stack.append(i)
        return stack
