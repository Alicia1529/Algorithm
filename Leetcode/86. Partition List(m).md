# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def partition(self, head: Optional[ListNode], x: int) -> Optional[ListNode]:
        before_head = ListNode()
        after_head = ListNode()

        curr_before, curr_after = before_head, after_head
        while head is not None:
            if head.val < x:
                curr_before.next = head
                curr_before = curr_before.next
            else:
                curr_after.next = head
                curr_after = curr_after.next
            # move ahead in the orignal list
            head = head.next
        # Last node of "after" list would also be ending node of the reformed list
        curr_after.next = None
        curr_before.next = after_head.next
        return before_head.next

/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */


// Runtime: 8 ms, faster than 54.11% of C++ online submissions for Partition List.
class Solution {
public:
    ListNode* partition(ListNode* head, int x) {
        ListNode* head1 = new ListNode();
        ListNode* head2 = new ListNode();
        ListNode *curr1 = head1, *curr2 = head2;
        while (head) {
            if (head->val < x) {
                curr1->next = head;
                curr1 = curr1->next;
            } else {
                curr2->next = head;
                curr2 = curr2->next;
            }
            head = head->next;
        }
        curr1->next = head2->next, curr2->next = nullptr;
        return head1->next;
    }
};
