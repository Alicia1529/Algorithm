class Solution {
public:
    int count = 0;
    
    bool issafe(int row, int col, int n, vector<string>& board){
        
        for (int i=0; i < row; i++){
            if (board[i][col] == 'Q'){
                return false;
            }
        }
        
        for (int i=row-1, j = col-1; i >=0 && j >= 0; i--,j--){
            if (board[i][j] == 'Q'){
                return false;
            }
        }
        
        for (int i=row-1, j=col+1; i >=0 && j < n; i--,j++){
            if (board[i][j] == 'Q'){
                return false;
            }
        }
        return true;
    }
    
    void NQueen(int row, int col, int n, vector<string>& board){
        if (row >= n){
            count++;
            return;
        }
        
        for (; col < n; col++){
            
            if (issafe(row, col, n, board)){
                
                board[row][col] = 'Q';
                NQueen(row+1, 0, n, board);
                board[row][col] = '.';
            }
            
        }
    }
    
    int totalNQueens(int n) {
        vector<string> board(n);
        
        string s = "";
        
        for (int i=0; i < n; i++){
            s.push_back('.');
        }
        
        for (int i=0; i < n; i++){
            board[i] = s;
        }
        
        NQueen(0, 0, n, board);
        return count;
    }
};
