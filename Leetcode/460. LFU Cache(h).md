1. OrderedDict — O(1) all operations

- `freq2key`: frequency → OrderedDict of keys (preserves insertion order, O(1) add/remove/peek-oldest)

```python
from collections import defaultdict, OrderedDict

class Node:
    def __init__(self, value, frequency):
        self.value = value
        self.frequency = frequency

class LFUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.key2val = dict()                    # key -> Node
        self.freq2key = defaultdict(OrderedDict) # frequency -> {key: 1} ordered by recency
        self.minFrequency = 0

    def get(self, key: int) -> int:
        if key not in self.key2val:
            return -1
        node = self.key2val[key]
        oldFrequency = node.frequency
        node.frequency += 1
        del self.freq2key[oldFrequency][key]
        if len(self.freq2key[oldFrequency]) == 0:
            del self.freq2key[oldFrequency]
        self.freq2key[node.frequency][key] = 1
        if self.minFrequency not in self.freq2key:
            self.minFrequency += 1
        return node.value

    def put(self, key: int, value: int) -> None:
        if self.capacity <= 0:
            return
        if key in self.key2val:
            self.key2val[key].value = value
            self.get(key)
            return
        if len(self.key2val) == self.capacity:
            candidate, _ = self.freq2key[self.minFrequency].popitem(last=False)
            if len(self.freq2key[self.minFrequency]) == 0:
                del self.freq2key[self.minFrequency]
            del self.key2val[candidate]
        self.key2val[key] = Node(value, 1)
        self.freq2key[1][key] = 1
        self.minFrequency = 1
```

- Time: O(1) for all operations
- Space: O(capacity)

2. deque — O(n) get/put due to `remove(key)`

- `freq2key`: frequency → deque of keys (O(n) to remove from middle)

```python
from collections import defaultdict, deque

class Node:
    def __init__(self, value, frequency):
        self.value = value
        self.frequency = frequency

class LFUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.key2val = dict()               # key -> Node
        self.freq2key = defaultdict(deque)  # frequency -> deque of keys
        self.minFrequency = 0

    def get(self, key: int) -> int:
        if key not in self.key2val:
            return -1
        node = self.key2val[key]
        oldFrequency = node.frequency
        node.frequency += 1
        self.freq2key[oldFrequency].remove(key)   # O(n) — scans entire deque
        if len(self.freq2key[oldFrequency]) == 0:
            del self.freq2key[oldFrequency]
        self.freq2key[node.frequency].append(key)
        if self.minFrequency not in self.freq2key:
            self.minFrequency += 1
        return node.value

    def put(self, key: int, value: int) -> None:
        if self.capacity <= 0:
            return
        if key in self.key2val:
            self.key2val[key].value = value
            self.get(key)
            return
        if len(self.key2val) == self.capacity:
            candidate = self.freq2key[self.minFrequency].popleft()  # O(1)
            if len(self.freq2key[self.minFrequency]) == 0:
                del self.freq2key[self.minFrequency]
            del self.key2val[candidate]
        self.key2val[key] = Node(value, 1)
        self.freq2key[1].append(key)
        self.minFrequency = 1
```

- Time: O(n) for get/put due to `deque.remove(key)`
- Space: O(capacity)
