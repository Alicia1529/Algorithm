# Approach 1: Brute Force
class Solution:
    def nextGreaterElement(self, nums1: List[int], nums2: List[int]) -> List[int]:
        ans = [-1] * len(nums1)
        for i, num1 in enumerate(nums1):
            is_found = False
            for j, num2 in enumerate(nums2):
                if num1 == num2:
                    is_found = True
                if is_found and num2 > num1:
                    ans[i] = num2
                    break
        return ans

# Approach 2: Better Brute Force
class Solution:
    def nextGreaterElement(self, nums1: List[int], nums2: List[int]) -> List[int]:
        ans = [-1] * len(nums1)
        hashmap = {num: i for i, num in enumerate(nums2)}
        for i, num in enumerate(nums1):
            j = hashmap[num] + 1
            while j < len(nums2):
                if num < nums2[j]:
                    ans[i] = nums2[j]
                    break
                j += 1
        return ans


# Approach 3: Using Stack

class Solution:
    def nextGreaterElement(self, nums1: List[int], nums2: List[int]) -> List[int]:
        stack = []
        next_greater = dict()
        for num in nums2:
            while stack and num > stack[-1]:
                val = stack.pop()
                next_greater[val] = num
            stack.append(num)
        ans = []
        for num in nums1:
            if num not in next_greater:
                ans.append(-1)
            else:
                ans.append(next_greater[num])
        return ans

class Solution {
public:
    vector<int> nextGreaterElement(vector<int>& nums1, vector<int>& nums2) {
        vector<int> ans;
        for (auto num: nums1) {
            int j = 0;
            while (nums2[j] != num) j++;
            j++;
            while (j < nums2.size() && nums2[j] < num) j++;
            if (j == nums2.size()) {
                ans.push_back(-1);
            } else {
                ans.push_back(nums2[j]);
            }
        }
        return ans;
    }
};
