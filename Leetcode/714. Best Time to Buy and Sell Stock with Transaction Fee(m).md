// Runtime: 84 ms, faster than 93.05% of C++ online submissions for Best Time to Buy and Sell Stock with Transaction Fee.
// Memory Usage: 55 MB, less than 60.45% of C++ online submissions for Best Time to Buy and Sell Stock with Transaction Fee.
class Solution {
public:
    int maxProfit(vector<int>& prices, int fee) {
        if (prices.size() <= 1) return 0;
        int cash = 0, hold = -prices[0];
        for (int i = 1; i < prices.size(); i++) {
            cash = max(cash, hold + prices[i] - fee);
            hold = max(hold, cash - prices[i]);
        }
        return cash;
    }
};
