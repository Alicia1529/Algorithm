class Solution {
public:
    bool CheckIpv6Address(string& ip)
    {
        int cnt = 0, i = 0;
        string tmp = "";
        
        while (i < ip.size()) {
            if ((tolower(ip[i]) <= 'f' && tolower(ip[i]) >= 'a') || 
                (ip[i] <= '9' && ip[i] >= '0')) {
                tmp += ip[i];
            } else if (ip[i] == ':') {
                if (tmp.size() > 4 || tmp == "") return false;
                tmp = "";
                ++cnt;
            } else {
                return false;
            }
            ++i;
        }
        
        if (tmp.size() > 4 || tmp == "") return false;
        if (cnt < 7 || cnt > 7) return false;
        
        return true;
    }
    
    bool CheckIpv4Address(string& ip)
    {
        int cnt = 0, i = 0, val;
        string tmp = "";
        
        while (i < ip.size()) {
            if (ip[i] <= '9' && ip[i] >= '0') {
                tmp += ip[i];
            } else if (ip[i] == '.') {
                if (tmp == "" || (tmp[0] == '0' && tmp.size() > 1) || tmp.size() > 3) return false;
                val = stoi(tmp);
                if (val > 255) return false;
                tmp = "";
                ++cnt;
            } else {
                return false;
            }
            ++i;
        }
        
        if (tmp == "" || (tmp[0] == '0' && tmp.size() > 1) || tmp.size() > 3) return false;
        val = stoi(tmp);
        if (val > 255) return false;
        
        if (cnt < 3 || cnt > 3) return false;
        
        return true;
    }
    
    string validIPAddress(string IP) 
    {
        string result = "Neither", ipv4 = "IPv4", ipv6 = "IPv6", tmpResult = "Neither";
        bool valid = false;;
        
        int i = 0;
        
        while (i < IP.size() && (IP[i] != '.' && IP[i] != ':')) {
            ++i;
        }
        
        if (i != IP.size()) {
            if (IP[i] == '.') tmpResult = ipv4;
            else tmpResult = ipv6;
        }
        
        if (tmpResult == ipv4) {
            valid = CheckIpv4Address(IP);
        } else if (tmpResult == ipv6) {
            valid = CheckIpv6Address(IP);
        }
        
        if (valid) {
            result = tmpResult;
        }
            
        return result;
    }
};
