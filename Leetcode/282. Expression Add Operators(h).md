// Runtime: 357 ms, faster than 69.84% of C++ online submissions for Expression Add Operators.
// Memory Usage: 87 MB, less than 81.87% of C++ online submissions for Expression Add Operators.

class Solution {
public:
    unordered_set<string> ans;
    
    void backtrack(string& num, int target, int idx, string& track, char op, long curr, long last, long cumu) {
        if (idx == num.size()) {
            if (op == '+' && cumu + curr == target) ans.insert(track);
            if (op == '-' && cumu - curr == target) ans.insert(track);
            if (op == '*' && cumu - last + last * curr == target) ans.insert(track);
            return;
        }

        // neg: op == "" && 0 < idx && num[idx - 1] == '0'
        if (curr != 0) {
            track += num[idx];
            curr = curr * 10 + (int) (num[idx] - '0');
            backtrack(num, target, idx + 1, track, op, curr, last, cumu);
            curr /= 10;
            track = track.substr(0, track.size() - 1); 
        }

        if (op == '+') {
            cumu += curr;
        } else if (op == '-') {
            curr = -curr;
            cumu += curr;
        } else {
            cumu -= last;
            curr = last * curr;
            cumu += curr;
        }
        
        last = curr;
        curr = num[idx] - '0';
    
        track += '+';
        track += num[idx];
        backtrack(num, target, idx + 1, track, '+', curr, last, cumu);
        track = track.substr(0, track.size() - 2); 

        track += '-';
        track += num[idx];
        backtrack(num, target, idx + 1, track, '-', curr, last, cumu);
        track = track.substr(0, track.size() - 2); 

        track += '*';
        track += num[idx];
        backtrack(num, target, idx + 1, track, '*', curr, last, cumu);
        track = track.substr(0, track.size() - 2); 
    }

    vector<string> addOperators(string num, int target) {
        string track = num.substr(0, 1);
        int curr = num[0] - '0';
        backtrack(num, target, 1, track, '+', curr, 0, 0);
        return vector<string>(ans.begin(), ans.end());
    }
};
