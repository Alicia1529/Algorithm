def backtrack(pos, neg, i, j):
    if i == len(pos):
        return 0
    if pos[i] == 0:
        return backtrack(pos, neg, i + 1, 0)

    min_step = float("inf")
    for jj in range(j, len(neg)):
        if neg[jj] == 0:
            continue
        reduce = min(pos[i], -neg[jj])
        pos[i] -= reduce
        neg[jj] += reduce
        min_step = min(min_step, 1 + backtrack(pos, neg, i, jj + 1))
        pos[i] += reduce
        neg[jj] -= reduce
    return min_step

class Solution:
    def minTransfers(self, transactions: List[List[int]]) -> int:
        from collections import defaultdict
        balances = defaultdict(int)
        for s, d, a in transactions:
            balances[s] -= a
            balances[d] += a
            
        pos = list(filter(lambda x: x > 0, balances.values()))
        neg = list(filter(lambda x: x < 0, balances.values()))
        return backtrack(pos, neg, 0, 0)
