```c++
class Solution {
public:
    int countSpecialSubsequences(vector<int>& A) {
        vector<int> dp = {0, 0, 0};
        int mod = 1e9 + 7;
        for (int& a: A)
            dp[a] = ((dp[a] + dp[a]) % mod + (a > 0 ? dp[a-1] : 1)) % mod;
        return dp[2];
    }
};
```

```python
class Solution:
    def countSpecialSubsequences(self, nums: List[int]) -> int:
        start = 0
        end = len(nums) - 1
        while start < end and nums[start] != 0: start += 1
        while 0 < end and nums[end] != 2: end -= 1
            
        if start >= end - 1: return 0
        
        dp = [[0, 0, 0] for i in range(end - start + 2)]
        
        for i in range(start, end + 1):
            val = nums[i]
            i -= start
            for j in range(0, 3):
                if val != j: dp[i + 1][j] = dp[i][j]
            if (val == 0):
                dp[i + 1][0] = 2 * dp[i][0] + 1;        
            elif (val == 1):
                dp[i + 1][1] = 2 * dp[i][1] + dp[i + 1][0];   
            else:
                dp[i + 1][2] = 2 * dp[i][2] + dp[i + 1][1];   
        return dp[end - start + 1][2] % (10**9 + 7)
```
