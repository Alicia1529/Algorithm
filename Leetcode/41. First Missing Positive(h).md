class Solution {
public:
    int firstMissingPositive(vector<int>& nums) {
        int n = nums.size();
        for (int i = 0;i < n;i++) {
            if (nums[i] <= 0 || nums[i] > n) {
                nums[i] = n + 1;
            }
        }
        
        for (int i = 0;i < n;i++) {
            int index = abs(nums[i]) - 1;
            if (index >= 0 && index < n)
                nums[index] = -1 * abs(nums[index]);
        }
        
        for (int i = 0;i < n;i++) {
            if (nums[i] >= 0) {
                return i + 1;
            }
        }
        return n + 1;
    }
};

class Solution {
public:
    bool isvalid(int i, int n) {
        if (i <= 0 || i > n) return false;
        return true;
    }
    int firstMissingPositive(vector<int>& nums) {
        int n = nums.size();
        for (int i = 0; i < n; i++) {
            if (!isvalid(nums[i], n)) continue;
            int idx = nums[i] - 1;
            if (idx == i) continue;
            if (idx < i) {
                nums[idx] = nums[i];
                continue;
            }
            if (idx < n && nums[idx] == nums[i]) continue;
            swap(nums[i], nums[idx]);
            i--;
        }
        for (int i = 0; i < n; i++) {
            if (!isvalid(nums[i], n) || nums[i] != i + 1) return i + 1;
        }
        return n + 1;
    }
};
