# Memory Usage: 14.4 MB, less than 53.85% of Python3 online submissions for Maximum Difference Between Increasing Elements.
class Solution:
    def maximumDifference(self, nums: List[int]) -> int:
        n = len(nums)
        
        smallest = nums[0]
        ans = float("-inf")
        
        for i in range(1, n):
            if nums[i] < smallest:
                smallest = nums[i]
            ans = max(ans, nums[i] - smallest)
        return ans if ans > 0 else -1

class Solution:
    def maximumDifference(self, nums: List[int]) -> int:
        ans = float("-inf")
        n = len(nums)
        for i in range(n):
            for j in range(i + 1, n):
                if nums[j] - nums[i] > ans:
                    ans = nums[j] - nums[i] 
        return ans if ans > 0 else -1
