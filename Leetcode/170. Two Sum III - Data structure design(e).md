Approach 1: Sorted List
Time Complexity:
    Add: O(1), Find: O(NLogN)
Space Complexity: O(N)
```python
class TwoSum:
    def __init__(self):
        self.nums = []
        self.is_sorted = False
        

    def add(self, number: int) -> None:
        self.nums.append(number)
        self.is_sorted = False
        

    def find(self, value: int) -> bool:
        if not self.is_sorted:
            self.nums.sort()
            self.is_sorted = True
        low, high = 0, len(self.nums) - 1
        while low < high:
            currSum = self.nums[low] + self.nums[high]
            if currSum < value:
                low += 1
            elif currSum > value:
                high -= 1
            else:
                return True
        return False
```
Approach 2: HashTable
Time Complexity:
    Add: O(1), Find: O(N)
Space Complexity: O(N)
```python
class TwoSum:

    def __init__(self):
        self.num_counts = defaultdict(int)
        

    def add(self, number: int) -> None:
        self.num_counts[number] += 1

    def find(self, value: int) -> bool:
        for num in self.num_counts.keys():
            comple = value - num
            if num != comple:
                if comple in self.num_counts: return True
            elif self.num_counts[num] > 1:
                return True
        return False
        
```
