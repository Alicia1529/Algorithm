class Solution {

        int ans = 0;
        int m = matrix.size();
        int n = matrix[0].size();
        vector<vector<int>> dp(m, vector<int>(n));
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (matrix[i][j] == '0') continue;
                dp[i][j] = 1;
                if (i != 0 and j != 0) {
                    dp[i][j] = min(dp[i - 1][j - 1], min(dp[i - 1][j], dp[i][j - 1])) + 1;

                }
                ans = max(ans, dp[i][j]);
            }
        }
        return ans * ans;
    }

public:

        int m = matrix.size();
        int n = matrix[0].size();
        vector<vector<int>> dp(m, vector<int>(n));
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                if (matrix[i][j] == '0') continue;
                dp[i][j] = 1;
                if (i != 0 and j != 0) {
                    int size = dp[i - 1][j - 1];
                    for (int k = 1; k <= size; k++) {
                        if (dp[i - k][j] != 0 && dp[i][j - k] != 0) {
                            dp[i][j] = k + 1;
                        } else {
                            break;
                        }
                            
                    }

                }
                ans = max(ans, dp[i][j]);
            }
        }
        return ans * ans;
    }
};
