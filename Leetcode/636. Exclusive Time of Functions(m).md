```python
class Solution:
    def exclusiveTime(self, n: int, logs: List[str]) -> List[int]:
        ans = [0 for i in range(n)]
        stack = list()
        for log in logs:
            task, event, tm = log.split(":")
            task = int(task)
            tm = int(tm)
            if event == "start":
                if stack:
                    prev_task, start_tm = stack[-1]
                    ans[prev_task] += tm - start_tm
                stack.append([task, tm])
            else:
                _, start_tm = stack.pop()
                interval = tm - start_tm + 1
                ans[task] += interval
                if stack:
                    stack[-1][1] = tm + 1
        return ans
```
