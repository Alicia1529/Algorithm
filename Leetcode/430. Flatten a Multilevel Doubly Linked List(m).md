class Solution {
public:
    Node* flatten(Node* head) {
        for(Node *curr = head; curr; curr=curr->next)
            if(curr->child)
            {
                Node *next = curr->next;
                curr->next = curr->child;
                curr->next->prev = curr;
                curr->child = NULL;
                Node *temp = curr;
                while(temp->next)
                    temp = temp->next;
                temp->next = next;
                if(next) next->prev = temp;
            }
        return head;
    }
};

class Solution {
public:
    pair<Node*, Node*> flatten_impl(Node* head) {
        Node* front = head;
        Node* tail = head;
        while (front) {
            Node* tmp = front->next;
            if (front->child) {
                pair<Node*, Node*> p = flatten_impl(front->child);
                p.first->prev = front;
                if (front->next) front->next->prev = p.second;
                p.second->next = front->next;
                front->next = p.first;
                front->child = nullptr;
                tail = p.second;
            } else {
                tail = front;
            }
            front = tmp;
        }
        return make_pair(head, tail);
    }
    Node* flatten(Node* head) {
        return flatten_impl(head).first;
    }
