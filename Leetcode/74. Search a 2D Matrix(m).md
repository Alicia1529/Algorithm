### Python
```python
class Solution:
    def searchMatrix(self, matrix: List[List[int]], target: int) -> bool:
        m, n = len(matrix), len(matrix[0])

        # Treat the m×n grid as ONE sorted array of length m*n (rows are sorted
        # and each row's first > previous row's last). Map a flat index back to
        # 2D:  i = idx // n,  j = idx % n.  Then it's a plain exact-match search.
        l, r = 0, m * n - 1
        while l <= r:                      # Template B: <= pairs with mid-1
            mid = (l + r) // 2
            i, j = mid // n, mid % n
            if matrix[i][j] == target:
                return True
            if matrix[i][j] < target:
                l = mid + 1
            else:
                r = mid - 1
        return False
```
- **Key idea:** the flatten trick works only because the whole matrix is globally
  sorted in row-major order — divide the index by `n` for the row, mod by `n` for
  the column. `n` (columns), *not* `m`, is the divisor.
- O(log(m·n)) time, O(1) space — one binary search, not two.
- Contrast with **240. Search a 2D Matrix II** (rows/cols sorted but *not* globally):
  there you can't flatten — walk from the top-right corner in O(m+n) instead.

### C++
```c++
class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int R = matrix.size();
        int C = matrix[0].size();
        int row_lo = 0, row_hi = R - 1;
        while (row_lo < row_hi) {
            int mid = (row_lo + row_hi) / 2;
            if (matrix[mid][C - 1] == target)
                return true;
            if (matrix[mid][C - 1] > target) {
                row_hi = mid;
            } else {
                row_lo = mid + 1;  // take care!
            }
        }
        int col_left = 0, col_right = C - 1;
        while (col_left <= col_right) {   // take care!
            int mid = (col_left + col_right) / 2;
            if (matrix[row_lo][mid] == target)
                return true;
            if (matrix[row_lo][mid] > target) {
                col_right = mid - 1;
            } else {
                col_left = mid + 1;
            }
        }
        return false;
        
    }
};


class Solution {

        int start=0, end=r*c;
        
        while(start < end){
            int mid = (start + end) >> 1;// divide by 2
            if ( matrix[mid / c] [mid % c] == target) return true;
            else if( matrix[ mid / c][mid % c] < target){
                start = mid + 1;
            }
            else{
                end = mid;
            }
        }
        return false;
    }
};
```
