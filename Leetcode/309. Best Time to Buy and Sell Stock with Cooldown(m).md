// Runtime: 0 ms, faster than 100.00% of C++ online submissions for Best Time to Buy and Sell Stock with Cooldown.
// Memory Usage: 11.3 MB, less than 31.23% of C++ online submissions for Best Time to Buy and Sell Stock with Cooldown.
class Solution {
public:
    int maxProfit(vector<int>& prices) {
        if (prices.size() <= 1) return 0;
        
        int n = prices.size();
        vector<int> buy(n), sell(n);
        
        // base case
        buy[0] = -prices[0];
    
        for (int i = 1; i < n; i++) {
            if (i - 2 >= 0) {
                buy[i] = max(buy[i - 1], sell[i - 2] - prices[i]);
            } else {
                buy[i] = max(buy[i - 1], 0 - prices[i]);
            }
            sell[i] = max(sell[i - 1], buy[i - 1] + prices[i]);
        }
        return sell[n - 1];
    }
};

class Solution {
public:
    int maxProfit(vector<int>& prices) {
        if (prices.size() <= 1) return 0;
        
        int sold = INT_MIN, held = INT_MIN, reset = 0;
        
        for (int price: prices) {
            int preSold = sold;
            
            sold = held + price;
            held = max(held, reset - price);
            reset = max(reset, preSold);
        }
        return max(sold, reset);
    }
};
