class Solution {
private:
// prev: upper bound of left
// head: lower bound of right
    void inorder(Node* n, Node*& prev, Node*& head) {
        if (!head && !n->left) {
            head = n;
            prev = n;
        }

        Node* left = n->left;
        Node* right = n->right;

        // left
        if (left)
            inorder(left, prev, head);

        // current
        prev->right = n;
        n->left = prev;

        // right
        prev = n;
        if (right)
            inorder(right, prev, head);
    }
public:
    Node* treeToDoublyList(Node* root) {
       if (!root)
           return nullptr;

        Node* prev = nullptr;
        Node* head = nullptr;

        inorder(root, prev, head);

        prev->right = head;
        head->left = prev;

        return head;                
    }
};

class Solution {
public:
    pair<Node*, Node*> dfs(Node* root) {
        if (root == nullptr) return {nullptr, nullptr};
        auto left = dfs(root->left);
        auto right = dfs(root->right);
        if (left.second == nullptr && right.first == nullptr)
            return {root, root};
        if (left.second == nullptr) {
            root->right = right.first;
            right.first->left = root;
            return {root, right.second};
        }
        if (right.first == nullptr) {
            left.second->right = root;
            root->left = left.second;
            return {left.first, root};
        }
        root->left = left.second;
        left.second->right = root;
        root->right = right.first;
        right.first->left = root;
        return {left.first, right.second};
    }
    Node* treeToDoublyList(Node* root) {
        if (root == nullptr) return nullptr;

        auto pair = dfs(root);
        pair.first->left = pair.second;
        pair.second->right = pair.first;
        return pair.first;   
    }
};

class Solution:
    def dfs(self, root):
        if root.left is None and root.right is None:
            return root, root
        if root.left is None:
            left, right = self.dfs(root.right)
            root.right = left
            left.left = root
            return root, right
        if root.right is None:
            left, right = self.dfs(root.left)
            right.right = root
            root.left = right
            return left, root
        ll, lr = self.dfs(root.left)
        rl, rr = self.dfs(root.right)
        lr.right = root
        root.left = lr
        rl.left = root
        root.right = rl
        return ll, rr
        
    def treeToDoublyList(self, root: 'Node') -> 'Node':
        if root is None:
            return None
        left, right = self.dfs(root)
        left.left = right
        right.right = left
        return left
