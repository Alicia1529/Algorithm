// Runtime: 88 ms, faster than 11.55% of C++ online submissions for Strobogrammatic Number II.
// Memory Usage: 38.9 MB, less than 17.99% of C++ online submissions for Strobogrammatic Number II.

class Solution {
public:
    vector<string> odd = {"0", "1", "8"};
    vector<string> even_outer = {"11", "69", "88", "96"};
    vector<string> even_inner = {"00", "11", "69", "88", "96"};

    vector<string> findStrobogrammatic(int n) {
        if (n == 1) return odd;
        if (n == 2) return even_outer;

        
        if (n % 2) {
            vector<string> ans;
            vector<string> recursion = findStrobogrammatic(n - 1);
            int mid = (n - 1) / 2;
            for (int i = 0; i < recursion.size(); i++) {
                string first = recursion[i].substr(0, mid);
                string second = recursion[i].substr(mid);
                for (int j = 0; j < odd.size(); j++) {
                    ans.push_back(first + odd[j] + second);
                }
            }
            return ans;
        } 
        
        unordered_set<string> ans;
        vector<string> recursion = findStrobogrammatic(n - 2);
        int mid = (n - 2) / 2;
        for (int i = 0; i < recursion.size(); i++) {
            string first = recursion[i].substr(0, mid);
            string second = recursion[i].substr(mid);
            for (int j = 0; j < even_inner.size(); j++) {
                ans.insert(first + even_inner[j] + second);
            }
            for (int j = 0; j < even_outer.size(); j++) {
                ans.insert(even_outer[j][0] + recursion[i] + even_outer[j][1]);
            }
        }
        return vector<string>(ans.begin(), ans.end());
    }
};
