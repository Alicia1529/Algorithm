class Solution {
class Solution {
public:
    int totalFruit(vector<int>& fruits) {
        
        int maxFruits = 0;
        unordered_map<int, int> fruitMap;
        
        int windowStart = 0;
        for(int windowEnd = 0; windowEnd < fruits.size(); windowEnd++){
            
            int fruit = fruits[windowEnd];
            fruitMap[fruit]++;
            
            while(fruitMap.size() > 2){
                fruitMap[fruits[windowStart]]--;
                if(fruitMap[fruits[windowStart]] == 0){
                    fruitMap.erase(fruits[windowStart]);
                }
                windowStart++;
            }
            
            maxFruits = max(maxFruits, windowEnd - windowStart + 1);
        }
        
        return maxFruits;
    }
};



public:
    int findVictim(unordered_map<int, int>& map) {
        int pos = INT_MAX;
        int victim;
        
        for (auto& p: map) {
            if (p.second < pos) {
                pos = p.second;
                victim = p.first;
            }
        }
        return victim;
    }
    int totalFruit(vector<int>& fruits) {
        unordered_map<int, int> map;
        int l = 0, maxfruit = 0;
        for (int r = 0; r < fruits.size(); r++) {
            if (map.size() < 2) {
                map[fruits[r]] = r;
            } else if (map.find(fruits[r]) != map.end()) {
                map[fruits[r]] = r;
            } else {
                if (r - l > maxfruit) maxfruit = r - l;
                int v = findVictim(map);
                l = map[v] + 1;
                map.erase(v);
                map[fruits[r]] = r;
            }
        }
        maxfruit = max(maxfruit, (int) fruits.size() - l);
        return maxfruit;
        
    }
};
