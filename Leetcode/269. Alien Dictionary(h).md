// Runtime: 15 ms, faster than 5.13% of C++ online submissions for Alien Dictionary.
// Memory Usage: 10.3 MB, less than 12.22% of C++ online submissions for Alien Dictionary.

class Solution {
public:
    string alienOrder(vector<string>& words) {
        // get vertices
        unordered_set<char> vertices;
        for (auto& word: words) {
            for (auto chr: word) vertices.insert(chr);
        }
        // get edges
        vector<pair<char, char>> orders;
        for (int i = 1; i < words.size(); i++) {
            int n = min(words[i - 1].size(), words[i].size());
            int j;
            for (j = 0; j < n; j++) {
                if (words[i - 1][j] != words[i][j]) {
                    orders.emplace_back(make_pair(words[i - 1][j], words[i][j]));
                    break;
                }
            }
            if (j == n && words[i - 1].size() > words[i].size()) return "";
        }
        
        // construct graph
        unordered_map<char, unordered_set<char>> inedges;
        unordered_map<char, unordered_set<char>> outedges;
        
        for (auto& itor: orders) {
            inedges[itor.second].insert(itor.first);
            outedges[itor.first].insert(itor.second);
        }
        
        string ans;
        queue<char> zeros;
        for (auto itor: vertices) {
            if (inedges.find(itor) == inedges.end()) zeros.push(itor);
        }            
        while (!zeros.empty()) {
            char v = zeros.front();
            zeros.pop();
            ans += v;
            for (auto u:outedges[v]) {
                inedges[u].erase(v);
                if (inedges[u].empty()) {
                    zeros.push(u);
                    inedges.erase(u);
                }
            }
        }
        return inedges.empty() ? ans: "";
    }
};
