```python
class Solution:
    def largestRectangleArea(self, heights: List[int]) -> int:
        stack = []  # monotonic increasing stack of indices
        max_area = 0
        heights.append(0)  # sentinel: forces all bars to be popped at the end

        for i, h in enumerate(heights):
            while stack and heights[stack[-1]] >= h:
                height = heights[stack.pop()]
                width = i if not stack else i - stack[-1] - 1
                max_area = max(max_area, height * width)
            stack.append(i)

        return max_area
```

>
> **Key insight:** For each bar, the largest rectangle extends left and right as far as all bars are ≥ its height. So when a shorter bar arrives, every taller bar in the stack has found its right boundary.
>
> **Width formula:** `i - stack[-1] - 1` — after popping, `stack[-1]` is the nearest shorter bar to the left. The rectangle spans between that left boundary and the current right boundary `i`, exclusive on both ends.
> If the stack is empty after popping, there's no shorter bar to the left, so width extends all the way: `width = i`.
>
> **Sentinel `0`:** Appending a height of `0` at the end guarantees every remaining bar in the stack gets popped and evaluated — without it you'd need a second loop (like the C++ solution below).

```c++
class Solution {
public:
    int largestRectangleArea(vector<int>& heights) {
        int max_area = 0;
        stack<int> stk;
        stk.push(-1);
        for (int i = 0; i < heights.size(); i++) {
            while (stk.top() != -1 && heights[i] < heights[stk.top()]) {
                int h = heights[stk.top()];
                stk.pop();
                int w = i - stk.top() - 1;
                if (h * w > max_area) max_area = h * w;
            }
            stk.push(i);
        }
        while (stk.top() != -1) {
            int h = heights[stk.top()];
            stk.pop();
            int w = heights.size() - stk.top() - 1;
            if (h * w > max_area) max_area = h * w;
        }
        return max_area;
    }
};
```
