class Solution {
public:
    vector<int> rightSideView(TreeNode* root) {
        if (root == nullptr) return {};

        vector<int> ans;
        
        queue<TreeNode*> q;
        q.push(root);
        ans.push_back(root->val);
        TreeNode* node;
    
        while (!q.empty()) {
            int size = q.size();
            for (int i = 0; i < size; i++) {
                node = q.front();
                q.pop();
                if (node->left) q.push(node->left);
                if (node->right) q.push(node->right);
            }
            if (!q.empty())
                ans.push_back(q.back()->val);
        }
        return ans;
        
    }
};

class Solution:
    def rightSideView(self, root: TreeNode) -> List[int]:
        if root is None:
            return []
        
        queue = [root] # nodes in level i
        ans = [root.val] # rigth side view in level i
        while len(queue):
            size = len(queue)
            for i in range(size):
                node = queue.pop(0)
                if node.left:
                    queue.append(node.left)
                if node.right:
                    queue.append(node.right)
            if len(queue):
                ans.append(queue[-1].val)
        return ans
