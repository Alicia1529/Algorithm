1. Brute Force — Check all point pairs

```python
class Solution:
    def minAreaRect(self, points: List[List[int]]) -> int:
        points_set = set()
        ans = float('inf')
        for x, y in points:
            points_set.add((x, y))
        points.sort()
        for idx, p1 in enumerate(points):
            for j in range(idx + 1, len(points)):
                x1, y1 = p1
                x2, y2 = points[j]
                if x2 > x1 and y2 > y1 and (x1, y2) in points_set and (x2, y1) in points_set:
                    ans = min(ans, (x2-x1)*(y2-y1))
        return 0 if ans == float('inf') else ans
```

- Time: O(N²), Space: O(N)

2. Group by x-coordinate — check y-pairs across columns

```python
from collections import defaultdict

class Solution:
    def minAreaRect(self, points: List[List[int]]) -> int:
        cols = defaultdict(set)
        for x, y in points:
            cols[x].add(y)

        ans = float('inf')
        last_x = {}  # (y1, y2) -> last x where both appeared

        for x in sorted(cols):
            ys = sorted(cols[x])
            for i in range(len(ys)):
                for j in range(i + 1, len(ys)):
                    y1, y2 = ys[i], ys[j]
                    if (y1, y2) in last_x:
                        ans = min(ans, (x - last_x[(y1, y2)]) * (y2 - y1))
                    last_x[(y1, y2)] = x

        return 0 if ans == float('inf') else ans
```

- Time: O(N²/C) in practice where C = number of distinct x-values, Space: O(N)
