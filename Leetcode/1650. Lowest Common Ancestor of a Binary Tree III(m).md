class Solution {
public:
    Node* helper(Node* p, Node* q, unordered_set<Node*>& pPath, unordered_set<Node*>& qPath) {
        if (p == q) {
            return p;
        }
        if (pPath.find(q) != pPath.end()) {
            return q;
        }
        if (qPath.find(p) != qPath.end()) {
            return p;
        }
        pPath.insert(p);
        qPath.insert(q);
        if (!p) {
            return helper(p, q->parent, pPath, qPath);
        } else if (!q) {
            return helper(p->parent, q, pPath, qPath);
        }
        return helper(p->parent, q->parent, pPath, qPath);
    }

        unordered_set<Node*> pPath;
        unordered_set<Node*> qPath;
        return helper(p, q, pPath, qPath);
    }
};

        vector<Node*> vp, vq;
        while (p) {
            vp.push_back(p);
            p = p->parent;
        }
        while (q) {
            vq.push_back(q);
            q = q->parent;
        }
        reverse(vp.begin(), vp.end());
        reverse(vq.begin(), vq.end());
        int l = 0, r = 0;
        while (l < vp.size() && r < vq.size()) {
            if (vp[l] == vq[r]) {
                l++, r++;
            } else {
                break;
            }
        }
        if (l == vp.size()) return vp[vp.size() - 1];
        if (r == vq.size()) return vq[vq.size() - 1];
        return vp[l - 1];
    }
