class Solution {
public:
    bool checkSubarraySum(vector<int>& nums, int k) {
        if (nums.size() == 1) return false;
        
        unordered_map<int, int> map;
        int sum = 0;
        int remainder;
        for (int i = 0; i < nums.size(); i++) {
            sum += nums[i];
            remainder = sum % k;
            if (0 < i && remainder == 0) return true;
            if (map.find(remainder) != map.end() && i - map[remainder] > 1) return true;
            if (map.find(remainder) == map.end())
                map[remainder] = i;
        }
        return false;
    }
};

class Solution:
    def checkSubarraySum(self, nums: List[int], k: int) -> bool:
        remainders = dict() # key: remainder, values = index of its first appearance
        size = len(nums)
        total = 0
        for i in range(size):
            total += nums[i]
            total %= k
            if total == 0 and i != 0: return True
            if i - remainders.get(total, i) > 1:
                return True
            remainders[total] = remainders.get(total, i)
        return False
