        int l=0,r=0;

        }

        if(cur==s.size()){
             if(!cnt)ans.push_back(path);

        if(s[cur]!='('&&s[cur]!=')')dfs(s,cur+1,path+s[cur],cnt,l,r);

        }  
    }

class Solution {
public:
    vector<string> removeInvalidParentheses(string s) {
        int l = 0, r = 0;
        for (int i = l; i < s.size(); i++) {
            if (s[i] == '(') l++;
            if (s[i] == ')') r++;
            // num of right parentheses cannot exceed the num of left parentheses
            r = min(l, r);
        }
        
        int size = min(l, r);
        unordered_set<string> ans;
        string track = "";
        dfs(s, ans, size, 0, 0, 0, track);
        
        return vector<string>(ans.begin(), ans.end());
    }
    void dfs(string& s, unordered_set<string>& ans, int size, int idx, int num_l, int num_r, string& track) {
        // 如果走到了字符串末尾，那么就进行判断是否已经是合法字符串了，如果是的话，就放到返回值中
        if (idx == s.size()) {
            if (num_l == num_r && num_l == size) ans.insert(track);
            return;
        }
    
        char c = s[idx];
        if (c == '(') {
            dfs(s, ans, size, idx + 1, num_l, num_r, track);
            track += c;
            dfs(s, ans, size, idx + 1, num_l + 1, num_r, track);
            track = track.substr(0, track.size() - 1);
            
        } else if (c == ')') {
            dfs(s, ans, size, idx + 1, num_l, num_r, track);
            if (num_r >= num_l) return;
            track += c;
            dfs(s, ans, size, idx + 1, num_l, num_r + 1, track);
            track = track.substr(0, track.size() - 1);
            
        } else {
            track += c;
            dfs(s, ans, size, idx + 1, num_l, num_r, track);
            track = track.substr(0, track.size() - 1);
        }
    }
};

class Solution {
private:
	bool isValid(string s) {
		int sum = 0;
		for (char& c : s) {
			if (c == '(') ++sum;
            else if (c == ')') --sum;
			if (sum < 0) return false;
		}
		return sum == 0;
	}
public:
	std::vector results;
	vector removeInvalidParentheses(string s) {
		int left = 0, right = 0;
		for (char& c : s) {
			if (c == '(') ++left;
            else if (c == ')') {
				if (left == 0) ++right;
                else if (left > 0) {
					--left;
				}
			}
		}
		dfs(s, 0, left, right);
		return results;
	}
	void dfs(string s, int beg, int left, int right) {
		if (left == 0 && right == 0) {
			if (isValid(s))
			results.push_back(s);
		} else {
			for (int i = beg; i < s.size(); ++i) {
				string tmp = s;
				if (right == 0 && left > 0 && tmp[i] == '(') {
					if (i == beg || tmp[i] != tmp[i - 1]) {
						tmp.erase(i, 1);
						dfs(tmp, i, left - 1, right);
					}
				}
				if (right > 0 && tmp[i] == ')') {
					if (i == beg || tmp[i] != tmp[i - 1]) {
						tmp.erase(i, 1);
						dfs(tmp, i, left, right - 1);
					}
				}
			}
		}
	}
}
;

class Solution:
    def __init__(self):
        self.ans = set()
        
    def removeInvalidParentheses(self, s: str) -> List[str]:
        l = 0
        r = 0
        for e in s:
            if e == "(":
                l += 1
            elif e == ")":
                r += 1
            # num of right parentheses cannot exceed the num of left parentheses
            r = min(l , r)
                
        def backtrack(i, numLeft, numRight, track):
            if i == len(s) and numLeft == 0 and numRight == 0:
                self.ans.add("".join(track))
            if i == len(s) or numLeft < 0 or numRight < 0:
                return
            if s[i] == "(":
                track.append(s[i])
                backtrack(i + 1, numLeft - 1, numRight, track)
                track.pop()
                backtrack(i + 1, numLeft, numRight, track)
            elif s[i] == ")":
                backtrack(i + 1, numLeft, numRight, track)
                # cannot have more right parentheses than right parentheses
                if numRight <= numLeft: return
                track.append(s[i])
                backtrack(i + 1, numLeft, numRight - 1, track)
                track.pop()
            else:
                track.append(s[i])
                backtrack(i + 1, numLeft, numRight, track)
                track.pop()
                
        track = []
        backtrack(0, r, r, track)
        return list(self.ans)
