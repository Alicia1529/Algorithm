class Solution:
    def groupStrings(self, strings: List[str]) -> List[List[str]]:
        # same length
        # same relative ord
        
        # 97, 99, 101, 102; 98, 100, 102, 103
        # "acef" "bdfg"
        # 97, 
        # "adef"\
        
        # dict   key: tuple of relative diff to first letter, value: [] for all strings
        # 0, 25; 0, -1
        # ["az"],["ba"]
        
        
        diff_map = defaultdict(list)
        for string in strings:
            base = ord(string[0])
            diff = []
            for letter in string:
                # take care!!! the mod sign
                diff.append(str((ord(letter) - base)%26))
            key = ",".join(diff)
            diff_map[key].append(string)
            
        ans = []
        for vals in diff_map.values():
            ans.append(vals)
        return ans
