class Solution {
public:
    string decodeString(string s) {
        stack<int> nums;
        stack<string> values;
        
        int num = 0;
        string value = "";
        for (int i = 0; i < s.size(); i++) {
            if (isdigit(s[i])) {
                if (num == 0) {
                    values.push(value);
                    value = "";
                }
                num = num * 10 + (int) (s[i] - '0');
            } else if (isalpha(s[i])) {
                value += s[i];
            } else if (s[i] == '[') {
                nums.push(num);
                num = 0;
                value = "";
            } else {
                string rep;
                for (int j = 0; j < nums.top(); j++) {
                    rep += value;
                }
                nums.pop();
                value = values.top() + rep;
                values.pop();
            }
        }
        return value;
    }
};

class Solution:   
    def decodeString(self, s: str) -> str:
        stack = []
        num = 0
        for i in range(len(s)):
            if s[i].isdigit():
                num = num * 10 + int(s[i])
            elif s[i] == '[':
                stack.append(num)
                num = 0
            elif s[i] == ']':
                substring = ""
                while isinstance(stack[-1], str):
                    substring = stack.pop() + substring
                substring = stack.pop() * substring
                stack.append(substring)
            else: # letter
                stack.append(s[i])
        return "".join(stack)

class Solution:
    @staticmethod
    def findRight(s, l): # left brackets
        balance = 0
        for r in range(l, len(s)):
            if s[r] == '[':
                balance += 1
            elif s[r] == ']':
                balance -= 1
            if balance == 0:
                return r
        raise Exception("Invalid string %s: no matching for %d".format(s, l))
            
    def decodeString(self, s: str) -> str:
        # base case: no bracket, then directly return 
        # recursion: k * returned value
        if "[" not in s:
            return s

        # flat case
        ret = ""
        l = 0
        num = 0
        while l < len(s):
            if s[l] == '[':
                r = self.findRight(s, l)
                ret += num * self.decodeString(s[l+1:r])
                l = r + 1
                num = 0
            elif s[l].isdigit():
                num = num * 10 + int(s[l])
                l += 1
            else:
                ret += s[l]
                l += 1
        return ret
