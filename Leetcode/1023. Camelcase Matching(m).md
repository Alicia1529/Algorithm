class Solution {
public:
    bool check(string s,string pattern){
        int i=0,j=0;
        int m=s.length(),n=pattern.length();
        while(i<m){
            if(s[i]==pattern[j]){
                i++;
                j++;
            }
            else if(s[i]>'A' && s[i]<'Z'){
                return 0;
            }
            else
                i++;
        }
        return j==n;
    }
    vector<bool> camelMatch(vector<string>& queries, string pattern) {
        vector<bool>ans;
        for(string s:queries){
            ans.push_back(check(s,pattern));       
        }
        return ans;
    }
};

class Solution {
public:
    vector<bool> camelMatch(vector<string>& queries, string pattern) {
        vector<bool> ans;
        for (auto& query: queries) {
            int i = 0, j = 0;
            for (; i < query.size(), j < pattern.size(); i++) {
                if (query[i] == pattern[j]) {
                    j++;
                } else if (query[i] <= 'Z') {
                    break;
                }
            }
            bool extra = false;
            for (; i < query.size(); i++) {
                if (query[i] <= 'Z') {
                    extra = true;
                    break;
                }
            }
            if (extra == false && j == pattern.size()) {
                ans.push_back(true);
            } else {
                ans.push_back(false);
            }
        }
        return ans;
    }
};
