
### Approach 2: Memoise Count
```c++
class Solution {
public:
    string removeDuplicates(string s, int k) {
        if (s.size() < k) return s;
        vector<int> count(s.size());
        for (int i = 0; i < s.size(); i++) {
            if (i == 0 || s[i] != s[i - 1]) {
                count[i] = 1;
            } else {
                count[i] = count[i - 1] + 1;
                if (count[i] == k) {
                    s.erase(i - k + 1, k);
                    i = i - k;
                }
            }
        }
        return s;
    }
};
```

### Approach 3: Stack
```c++
class Solution {
public:
    string removeDuplicates(string s, int k) {
        if (s.size() < k) return s;
        stack<int> counts;
        for (int i = 0; i < s.size(); i++) {
            if (i == 0 || s[i] != s[i - 1]) {
                counts.push(1);
            } else if (++counts.top() == k) {
                counts.pop();
                s.erase(i - k + 1, k);
                i -= k;
            }
        }
        return s;
    }
};
```

```python
class Solution:
    def removeDuplicates(self, s: str, k: int) -> str:
        stack = []
        for l in s:
            if len(stack) == 0 or stack[-1][0] != l:
                stack.append([l, 1])
            else:
                stack[-1][1] += 1
            if stack[-1][1] == k:
                stack.pop()
        for i in range(len(stack)):
            stack[i] = stack[i][1] * stack[i][0]
        return "".join(stack)
```
