class Solution {
public:
    unordered_map<Node*, Node*> mappings;
    Node* copyRandomList(Node* head) {
        if (!head) return nullptr;
        if (mappings.find(head) != mappings.end())
            return mappings[head];
        Node* cloned = new Node(head->val);
        mappings[head] = cloned;
        if (head->next)
            cloned->next = copyRandomList(head->next);
        if (head->random)
            cloned->random = copyRandomList(head->random);
        return cloned;
    }
};

        unordered_map<Node*, Node*> mappings;
        Node *front = head;
        while (front) {
            Node *cloned = new Node(front->val);
            mappings[front] = cloned;
            front = front->next;
        }
        front = head;
        while (front) {
            mappings[front]->next = mappings[front->next];
            mappings[front]->random = mappings[front->random];
            front = front->next;
        }
        return mappings[head];
    }
};

class Solution {
public:
    unordered_map<Node*, Node*> mappings;
    Node* getClonedNode(Node* head, Node* cloned_head, Node* curr) {
        Node* front = head;
        Node* cloned_front = cloned_head;
        while (front != curr) {
            front = front->next;
            cloned_front = cloned_front->next;
        }
        return cloned_front;
    }
    Node* copyRandomList(Node* head) {
        if (!head) return nullptr;
        Node *front = head;
        Node *cloned_head = new Node(head->val);
        Node *cloned_front = cloned_head;
        while (front->next) {
            cloned_front->next = new Node(front->next->val);
            front = front->next;
            cloned_front = cloned_front->next;
        }
        front = head, cloned_front = cloned_head;
        while (front) {
            cloned_front->random = getClonedNode(head, cloned_head, front->random);
            front = front->next;
            cloned_front = cloned_front->next;
        }
        return cloned_head;
    }
};

class Solution {
public:
    Node* copyRandomList(Node* head) {
        if (head == nullptr) return nullptr;

        Node* new_head = new Node(head->val);
        
        int index = 0;
        unordered_map<Node*, int> node_2_index; // for the old list
        unordered_map<int, Node*> index_2_node; // for the new list
        node_2_index[head] = index;
        index_2_node[index] = new_head;
        
        Node* front = head;
        Node* new_front = new_head;
        while (front->next) {
            new_front->next = new Node(front->next->val);
            front = front->next;
            new_front = new_front->next;

            index++;
            node_2_index[front] = index;
            index_2_node[index] = new_front;
        }
        
        front = head;
        new_front = new_head;
        while (front) {
            if (front->random != nullptr) {
                int index = node_2_index[front->random];
                new_front->random = index_2_node[index];
            }
            front = front->next;
            new_front = new_front->next;
        }
        return new_head;
    }
};
