class MyCircularQueue {
public:
    vector<int> queue;
    int size;
    int capacity;
    int front, rear;
    
    MyCircularQueue(int k) {
        queue = vector<int>(k, -1);
        capacity = k;
        size = 0;
        front = 0;
        rear = -1;
    }
    
    bool enQueue(int value) {
        if (isFull()) return false;
        rear = _nextIdx(rear);
        queue[rear] = value;
        size++;
        return true;
    }
    
    bool deQueue() {
        if (isEmpty()) return false;
        front = _nextIdx(front);
        size--;
        return true;
    }
    
    int Front() {
        if (isEmpty()) return -1;
        return queue[front];
    }
    
    int Rear() {
        if (isEmpty()) return -1;
        return queue[rear];
    }
    
    bool isEmpty() {
        return size == 0;
    }
    
    bool isFull() {
        return size == capacity;
    }
    
    int _nextIdx(int idx) {
        return (idx + 1) % capacity;
    }
};
