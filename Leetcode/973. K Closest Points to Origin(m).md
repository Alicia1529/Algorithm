class Solution {
public:
    vector<vector<int>> p;

    struct point {
        int x;
        int y;
        bool operator() (point& a, point& b) {
            return pow(a.x, 2) + pow(a.y, 2) >  pow(b.x, 2) + pow(b.y, 2);
        }
    };

    vector<vector<int>> kClosest(vector<vector<int>>& points, int k) {
        priority_queue<point, vector<point>, point> pq;
        
        for (auto& cord: points) {
            point p = {cord[0], cord[1]};
            pq.push(p);
        }
        
        vector<vector<int>> ans;
        while (k--) {
            point p = pq.top();
            pq.pop();
            ans.push_back({p.x, p.y});
        }
        return ans;
    }
};

class Solution {
public:
    vector<vector<int>> kClosest(vector<vector<int>>& points, int k) {
        priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>> > pq;
        for (int i = 0; i < points.size(); i++) {
            int dis = points[i][0] * points[i][0] + points[i][1] * points[i][1];
            pq.push(make_pair(dis, i));
        }
        vector<vector<int>> ret;
        while (ret.size() < k) {
            auto pair = pq.top();
            pq.pop();
            ret.emplace_back(points[pair.second]);
        }
        return ret;
    }
};

        for x, y in points:
            dis = (x**2 + y**2)**0.5
