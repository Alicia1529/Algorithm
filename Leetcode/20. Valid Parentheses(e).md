### Python
```python
class Solution:
    def isValid(self, s: str) -> bool:
        stack = []
        mapping = {
            "}": "{",
            ")": "(",
            "]": "[",
        }
        for element in s:
            if element in "([{":
                stack.append(element)
            else:
                pairing = mapping[element]
                if not stack or stack[-1] != pairing:
                    return False
                stack.pop()
        return len(stack) == 0
```

### C++
```c++
class Solution {
public:
    unordered_map<char, char> mappings= {
        {'(', ')'}, {'{', '}'}, {'[', ']'} 
    };
    bool isValid(string s) {
        stack<char> memo;
        for (int i = 0; i < s.size(); i++) {
            if (memo.empty()) {
                memo.push(s[i]);
            } else if (mappings[memo.top()] == s[i]) {
                memo.pop();
            } else {
                memo.push(s[i]);
            }
        }
        return memo.empty();
    }
};
```
