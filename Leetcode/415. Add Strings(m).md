class Solution {
public:
    string addStrings(string num1, string num2) {
        if (num1.empty()) return num2;
        if (num2.empty()) return num1;

        int carryover = 0;
        string ret;

        for (int i = (int) num1.size() - 1, j = (int) num2.size() - 1;
             i >= 0 || j >=0; i--, j--)
        {
            if (i >= 0) carryover += num1[i] - '0';
            if (j >= 0) carryover += num2[j] - '0';
            string digit(1, '0' + carryover % 10);
            ret = digit + ret;
            carryover = carryover / 10;
        }
        if (carryover) ret = "1" + ret;
        return ret;
        
    }
};



class Solution {
public:
    string addStrings(string num1, string num2) {
         string result;

    int indexOne = num1.length() - 1;
    int indexTwo = num2.length() - 1;
    int carry = 0;

    while (indexOne >= 0 || indexTwo >= 0 || carry) {
        int current = carry;

        if (indexOne >= 0) {
            current += num1[indexOne] - '0';
        }

        if (indexTwo >= 0) {
            current += num2[indexTwo] - '0';
        }

        carry = current / 10;
        current = current % 10;

        result = (char)(current + '0') + result;

        indexOne--;
        indexTwo--;
    }

    return result;
    }
};
