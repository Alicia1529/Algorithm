class Solution {
public:
    int shape;
    vector<vector<string>> ans;
    
    bool is_valid(vector<string>& board, int r, int c) {
        // check col
        for (int i = 0; i < r; i++) {
            if (board[i][c] == 'Q') return false;
        }
        // check diagonal
        for (int i = r - 1, j = c - 1; i >= 0 && j >= 0; i--, j--) {
            if (board[i][j] == 'Q') return false;
        }
        for (int i = r - 1, j = c + 1; i >= 0 && j < shape; i--, j++) {
            if (board[i][j] == 'Q') return false;
        }
        return true;
    }
    
    void backtrack(vector<string>& board, int row, int col) {
        if (row >= shape) {
            ans.emplace_back(board);
            return;
        }
        for (int i = col; i < shape; i++) {
            if (is_valid(board, row, i)) {
                board[row][i] = 'Q';
                backtrack(board, row + 1, 0);
                board[row][i] = '.';
            }
        }
        
    }
    
    vector<vector<string>> solveNQueens(int n) {
        shape = n;
        string s(n, '.');
        vector<string> board(shape, s);
        backtrack(board, 0, 0);
        return ans;
        
    }
};

class Solution {
public:
    int shape;
    vector<vector<string>> ans;
    
    vector<string> borad_2_string(vector<vector<bool>>& board) {
        vector<string> ret;
        for (auto& row: board) {
            string s;
            for (auto v: row) {
                if (v) {
                    s += "Q";
                } else {
                    s += ".";
                }
            }
            ret.emplace_back(s);
        }
        return ret;
        
    }
    
    bool is_valid(vector<vector<bool>>& board, int r, int c) {
        // check row
        for (int i = 0; i < c; i++) {
            if (board[r][i]) return false;
        }
        // check col
        for (int i = 0; i < r; i++) {
            if (board[i][c]) return false;
        }
        // check diagonal
        for (int i = r - 1, j = c - 1; i >= 0 && j >= 0; i--, j--) {
            if (board[i][j]) return false;
        }
        for (int i = r - 1, j = c + 1; i >= 0 && j < shape; i--, j++) {
            if (board[i][j]) return false;
        }
        return true;
    }
    
    void backtrack(vector<vector<bool>>& board, int remaining, int idx) {
        if (remaining == 0) {
            ans.push_back(borad_2_string(board));
            return;
        }
        for (int i = idx; i < shape * shape; i++) {
            int r = i / shape, c = i % shape;
            if (is_valid(board, r, c)) {
                board[r][c] = true;
                backtrack(board, remaining - 1, i + 1);
                board[r][c] = false;
            }
        }
        
    }
    
    
    vector<vector<string>> solveNQueens(int n) {
        shape = n;
        vector<vector<bool>> board(shape, vector<bool>(shape, false));
        backtrack(board, n, 0);
        return ans;
        
    }
};
