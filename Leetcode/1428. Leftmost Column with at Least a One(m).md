/**
 * // This is the BinaryMatrix's API interface.
 * // You should not implement it, or speculate about its implementation
 * class BinaryMatrix {
 *   public:
 *     int get(int row, int col);
 *     vector<int> dimensions();
 * };
 */

class Solution {
public:
    int helper(BinaryMatrix &binaryMatrix, int row, int n) {
        int l = 0, r = n - 1;
        while (l < r) {
            int m = l + (r - l) / 2;
            int v = binaryMatrix.get(row, m);
            if (v == 0) {
                l = m + 1;
            } else {
                r = m;
            }
        }
        return binaryMatrix.get(row, l) == 1 ? l : n;
    }
    int leftMostColumnWithOne(BinaryMatrix &binaryMatrix) {
        auto dim = binaryMatrix.dimensions();
        int m = dim[0], n = dim[1];
        int ans = n;
        for (int i = 0; i < m; i++) {
            ans = min(ans, helper(binaryMatrix, i, n));
        }
        return ans == n ? -1: ans;
    }
};

# """

#    def dimensions(self) -> list[]:

class Solution:
    def leftMostColumnWithOne(self, binaryMatrix: 'BinaryMatrix') -> int:
        m, n = binaryMatrix.dimensions()
        leftMostColumn = n
        
        # each row: use binary search to find the first index where it's 1
        def binarySearch(row, l, r):
            r -= 1
            
            while l < r:
                mid = l + (r - l) // 2
                if binaryMatrix.get(row, mid) == 0:
                    l = mid + 1
                else:
                    r = mid
            return l if binaryMatrix.get(row, l) == 1 else l + 1 
            
        for r in range(m):
            if r != 0 and leftMostColumn < n and binaryMatrix.get(r, leftMostColumn) == 0:
                continue
            c = binarySearch(r, 0, leftMostColumn)
            if c < leftMostColumn:
                leftMostColumn = c
            
        
        return -1 if leftMostColumn == n else leftMostColumn
