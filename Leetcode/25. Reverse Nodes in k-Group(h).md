/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* findTail(ListNode* head, int k) {
        ListNode* tail = head;
        while (--k) {
            if (tail == nullptr)
                return nullptr;
            tail = tail->next;
        }
        return tail;
    }

    ListNode* reverseList(ListNode* head, ListNode* tail) {
        if (tail == nullptr) return head;
        ListNode* rest = tail->next;
        ListNode* curr = head;
        ListNode* new_head = rest;
        while (curr != rest) {
            ListNode* tmp = curr->next;
            curr->next = new_head;
            new_head = curr;
            curr = tmp;
        }
        return new_head;
    }

    ListNode* reverseKGroup(ListNode* head, int k) {
        ListNode* dummy = new ListNode(-1, head);
        ListNode* prev = dummy;
        ListNode* tail = findTail(head, k);
        while (tail != nullptr) {
            reverseList(head, tail);
            prev->next = tail;
            prev = head;
            tail = findTail(head->next, k);
            head = head->next;
        }
        return dummy->next;
        
    }
};
