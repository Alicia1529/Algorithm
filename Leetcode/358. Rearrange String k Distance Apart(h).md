class Solution {
public:
    string rearrangeString(string s, int k) {
        priority_queue<pair<int, char>> pq;
        unordered_map<char, int> map;
        deque<pair<int, char>> dq;
        string ret;
        
        for (auto c: s) map[c]++;
        for (auto &[c, i]: map) pq.emplace(i, c);
        
        while (!pq.empty()) {
            auto [i, c] = pq.top();
            pq.pop();
            ret += c;
            
            dq.push_back({i, c});
            if (dq.size() >= k) {
                auto [i, c] = dq.front();
                dq.pop_front();
                if (i - 1 > 0) pq.emplace(i - 1, c);
            }
        }
        while (!dq.empty()) {
            auto [i, c] = dq.front();
            dq.pop_front();
            if (i > 1) return "";
        }
        return ret;
    }
};
