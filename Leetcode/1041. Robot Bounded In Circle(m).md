class Solution {
public:
    bool isRobotBounded(string instructions) {
        vector<vector<int>> direction = {{0, 1}, {1, 0}, {0, -1}, {-1, 0}};
        int x = 0, y = 0;
        // facing north
        int idx = 0;
        for (char chr: instructions) {
            if (chr == 'L') {
                idx = (idx + 3) % 4;
            } else if (chr == 'R') {
                idx = (idx + 1) % 4;
            } else {
                x += direction[idx][0];
                y += direction[idx][1];
            }
        }
        // after one cycle:
        // robot returns into initial position
        // or robot doesn't face north
        return (x == 0 and y == 0) or idx != 0;
    }
};
