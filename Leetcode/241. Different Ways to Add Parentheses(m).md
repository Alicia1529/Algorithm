class Solution {
public:
    unordered_map<string, vector<int>> mem;

    vector<int> diffWaysToCompute(string expression) {
        vector<int> ans;
        for (int i = 0; i < expression.size(); i++) {
            if (expression[i] != '*' && expression[i] != '+' && expression[i] != '-')
                continue;
            string left = expression.substr(0, i);
            string right = expression.substr(i + 1);
            if (mem.find(left) == mem.end()) mem[left] = diffWaysToCompute(left);
            if (mem.find(right) == mem.end()) mem[right] = diffWaysToCompute(right);
            for (auto l: mem[left]) {
                for (auto r: mem[right]) {
                    if (expression[i] == '*') {
                        ans.push_back(l * r);
                    } else if (expression[i] == '+') {
                        ans.push_back(l + r);
                    } else {
                        ans.push_back(l - r);
                    }
                }
            }
        }
        if (ans.empty()) {
            ans.push_back(stoi(expression));
        }
        return ans;
    }
};
