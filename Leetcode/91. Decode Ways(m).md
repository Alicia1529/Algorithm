class Solution:
    def numDecodings(self, decInput: str) -> int:
        n = len(decInput)
        if n == 0: return 0
        dp = [0 for i in range(n + 1)]
        dp[0] = 1 
        dp[1] = 0 if decInput[0] == '0' else 1 
        for i in range(1, n):
            one = int(decInput[i])
            two = int(decInput[i - 1: i + 1])
            if one == 0:
                if two == 10 or two == 20:
                    dp[i + 1] = dp[i - 1]
                else:
                    dp[i + 1] = 0 
            else:
                dp[i + 1] = dp[i]
                if two >= 10 and two <= 26: dp[i + 1] += dp[i - 1]
        return dp[n]

class Solution {

        vector<int> dp(size + 1);

        dp[1] = (s[0] == '0') ? 0: 1;

            int one_digit = s[i] - '0';

                    dp[i + 1] = 0;

class Solution {
public:
    int numDecodings(string s) {
        int dp[s.size() + 1];
        dp[0] = 1;
        dp[1] = (s[0] == '0') ? 0: 1;
        for (int i = 1; i < s.size(); i++) {
            int num = stoi(s.substr(i - 1, 2));
            if (num == 10 || num == 20) {
                dp[i + 1] = dp[i - 1];
            } else if (num % 10 == 0) {
                return 0;
            } else if (10 < num && num <= 26) {
                dp[i + 1] = dp[i] + dp[i - 1];
            } else {
                dp[i + 1] = dp[i];
            }
        }
        return dp[s.size()];
    }
};


class Solution {
public:
    int numDecoding(string s,int in,vector<int>& dp){
        int res=0;
        if(in==s.size())
            return 1;
        if(dp[in]!=-1)
            return dp[in];
        if(s[in]!='0'){
            res+=numDecoding(s,in+1,dp);
            if(in+1<s.size()){
                int x=(s[in]-'0')*10+s[in+1]-'0';
                if(x<=26)
                    res+=numDecoding(s,in+2,dp);
            }
        }
        return dp[in]=res;
    }
    int numDecodings(string s) {
        vector<int>dp(s.size(),-1);
        return numDecoding(s,0,dp);
    }
};
