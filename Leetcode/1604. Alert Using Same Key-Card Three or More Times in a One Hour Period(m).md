class Solution {
public:
    bool isInOneHour(string& t1, string& t2) {
        int tm1 = stoi(t1.substr(0, 2)) * 60 + stoi(t1.substr(3));
        int tm2 = stoi(t2.substr(0, 2)) * 60 + stoi(t2.substr(3));
        return tm2 - tm1 <= 60;
    }
    vector<string> alertNames(vector<string>& keyName, vector<string>& keyTime) {
        // group by name, and then sort
        // for individual users
        unordered_map<string, vector<string>> map; // name, vector of time
        for (int i = 0; i < keyName.size(); i++) {
            map[keyName[i]].push_back(keyTime[i]);
        }
        
        vector<string> ret;
        for (auto &[name, array]: map) {
            sort(array.begin(), array.end());
            for (int i = 0; i < array.size() - 2; i++) {
                if (isInOneHour(array[i], array[i + 2])) {
                    ret.push_back(name);
                    break;
                }                
            }
        }
        sort(ret.begin(), ret.end());
        return ret;
    }
};

// ["10:00", "11:00"]
