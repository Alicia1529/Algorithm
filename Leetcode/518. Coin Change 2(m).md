// Memory Usage: 6.9 MB, less than 93.57% of C++ online submissions for Coin Change 2.
class Solution {
public:
    int change(int amount, vector<int>& coins) {
        int t[amount + 1];
        memset(t, 0, sizeof(t));
        t[0] = 1;

        for (int coin : coins) {
            for (int x = coin; x < amount + 1; ++x) {
                t[x] += t[x - coin];                   
            }
        }
        return t[amount];
    }
};

// Runtime: 20 ms, faster than 56.01% of C++ online submissions for Coin Change 2.
// Memory Usage: 12 MB, less than 39.91% of C++ online submissions for Coin Change 2.

class Solution {
public:
    int change(int m, vector<int>& coins) {
        int n = coins.size();
        
        int t[n + 1][m + 1];
        for (int j = 0; j <= m; j++) t[0][j] = 0;
        for (int i = 0; i <= n; i++) t[i][0] = 1;

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (j >= coins[i - 1]) t[i][j] = t[i - 1][j] + t[i][j - coins[i - 1]];
                else t[i][j] = t[i - 1][j];                             
            }
        }
        return t[n][m];
    }
};
