class Solution {
public:

    string reorganizeString(string s) {
        priority_queue<pair<int, char>> pq;
        unordered_map<char, int> map;
        string ret;

        for (auto chr: s) map[chr]++;
        for (auto &[chr, num]: map) pq.emplace(num, chr);
        pair<int, char> last = {-1, 0};
        while (!pq.empty()) {
            auto [num, chr] = pq.top();
            pq.pop();
            ret += chr;
            
            if (last.first > 0) pq.push(last);
            last = {num - 1, chr};
        }
        return last.first > 0 ? "":ret;
    }
};

class Solution {
public:
    vector<string> ans;
    void backtracking(unordered_map<char, int>& map, string tmp, int n) {
        if (!ans.empty()) return;

        if (tmp.size() == n) {
            ans.push_back(tmp);
            return;
        };
        
        for (auto p = map.begin(); p != map.end(); p++) {
            if (p->second != 0 && (tmp.empty() || tmp[tmp.size() - 1] != p->first)) {
                map[p->first]--;
                string next = tmp;
                next += p->first;
                backtracking(map, next, n);
                map[p->first]++;
            }
        }
    }
    string reorganizeString(string s) {
        unordered_map<char, int> map;
        for (auto chr: s) map[chr]++;
        backtracking(map, "", s.size());
        return ans.empty() ? "": ans.back();
    }
};
