class Solution {
public:
    int dfs(vector<vector<int>>& grid, int n, int i, int j, int group_id) {
        int total = 1;
        grid[i][j] = group_id;
        if (0 <= i - 1 && grid[i - 1][j] == 1) {
            total += dfs(grid, n, i - 1, j, group_id);
        }
        if (i + 1 < n && grid[i + 1][j] == 1) {
            total += dfs(grid, n, i + 1, j, group_id);
        }
        if (0 <= j - 1 && grid[i][j - 1] == 1) {
            total += dfs(grid, n, i, j - 1, group_id);
        }

        if (j + 1 < n && grid[i][j + 1] == 1) {
            total += dfs(grid, n, i, j + 1, group_id);
        }
        return total;
    }
    int updateIslands(vector<vector<int>>& grid, unordered_map<int, int>& islands, int n, int i, int j) {
        set<int> neighbors;
        if (0 <= i - 1 && grid[i - 1][j] != 0) {
            neighbors.insert(grid[i - 1][j]);
        }
        if (i + 1 < n && grid[i + 1][j] != 0) {
            neighbors.insert(grid[i + 1][j]);
        }
        if (0 <= j - 1 && grid[i][j - 1] != 0) {
            neighbors.insert(grid[i][j - 1]);
        }
        if (j + 1 < n && grid[i][j + 1] != 0) {
            neighbors.insert(grid[i][j + 1]);
        }
        int total = 1;
        for (auto itor: neighbors) {
            total += islands[itor];
        }
        return total;
    }
    
    int largestIsland(vector<vector<int>>& grid) {
        int n = grid.size();
        int group_id = 2;
        int ans = 0;
        unordered_map<int, int> islands;
        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] != 1) continue;
                islands[group_id] = dfs(grid, n, i, j, group_id);
                ans = max(ans, islands[group_id]);
                group_id++;
            }
        }
        
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (grid[i][j] != 0) continue;
                ans = max(ans, updateIslands(grid, islands, n, i, j));
            }
        }
        return ans;
    }
};

class Solution:
    def dfs(self, grid, i, j, n, island_id):
        if i < 0 or j < 0 or i >= n or j >= n or grid[i][j] != 1:
            return 0
        grid[i][j] = island_id
        area = 1
        area += self.dfs(grid, i + 1, j, n, island_id)
        area += self.dfs(grid, i - 1, j, n, island_id)
        area += self.dfs(grid, i, j + 1, n, island_id)
        area += self.dfs(grid, i, j - 1, n, island_id)
        return area
    
    def connectLand(self, grid, area_map, i, j, n):
        area = 1
        island_set = set()
        if i - 1 >= 0 and grid[i - 1][j] != 0:
            island_set.add(grid[i - 1][j])
        if j - 1 >= 0 and grid[i][j - 1] != 0:
            island_set.add(grid[i][j - 1])    
        if i + 1 < n and grid[i + 1][j] != 0:
            island_set.add(grid[i + 1][j])
        if j + 1 < n and grid[i][j + 1] != 0:
            island_set.add(grid[i][j + 1])
        for island_id in island_set:
            area += area_map[island_id]
        return area
    
    def largestIsland(self, grid: List[List[int]]) -> int:
        # run dfs: area of each island
        # change one and calculate the max area
        island_id = 2
        area_map = dict() # key: island_id, value: size
        largestIsland = 0
        
        n = len(grid)
        
        for i in range(n):
            for j in range(n):
                if grid[i][j] == 1:
                    area_map[island_id] = self.dfs(grid, i, j, n, island_id)
                    largestIsland = max(largestIsland, area_map[island_id])
                    island_id += 1
        
        for i in range(n):
            for j in range(n):
                if grid[i][j] == 0:
                    largestIsland = max(largestIsland, self.connectLand(grid, area_map, i, j, n))
                    
        return largestIsland
