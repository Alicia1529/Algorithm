class Solution {
public:
    string originalDigits(string s) {
        vector<int> a(10, 0);
        vector<int> alpha(128, 0);
        for (char c : s)
            alpha[c]++;
        a[0] = alpha['z'];
        a[2] = alpha['w'];
        a[4] = alpha['u'];
        a[6] = alpha['x'];
        a[8] = alpha['g'];
        a[1] = alpha['o'] - a[0] - a[2] - a[4];
        a[3] = alpha['h'] - a[8];
        a[5] = alpha['f'] - a[4];
        a[7] = alpha['v'] - a[5];
        
        a[9] = alpha['i'] - a[5] - a[6] - a[8];
        string ans;
        for (int i = 0; i < 10; i++) {
            if (a[i] > 0)
                ans += string(a[i], '0' + i);
        }
        return ans;
    }
};

class Solution {
public:
    vector<int> nums;
    vector<string> representations = {"zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine"};

    int removeNum(unordered_map<char, int>& cnt, char chr, int i) {
        int num = 0;
        while (cnt[chr]) {
            for (char c: representations[i]) cnt[c]--;
            nums.push_back(i);
        }
        return num;
    }

    string originalDigits(string s) {
        unordered_map<char, int> cnt;
        for (char chr: s) cnt[chr]++;
        
        removeNum(cnt, 'z', 0);
        removeNum(cnt, 'w', 2);
        removeNum(cnt, 'u', 4);
        removeNum(cnt, 'x', 6);
        removeNum(cnt, 'g', 8);
        
        removeNum(cnt, 'o', 1);
        removeNum(cnt, 't', 3);
        removeNum(cnt, 'f', 5);
        removeNum(cnt, 's', 7);
        removeNum(cnt, 'i', 9);
        
        
        sort(nums.begin(), nums.end());
        string ans;
        for (int num: nums) {
            ans += (char) num + '0';
        }
        return ans;
    }
};


class Solution {
public:
    vector<int> nums;

    vector<string> representations = {"zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine"};

    bool check(unordered_map<char, int>& cnt) {
        for (auto& itor: cnt) {
            if (itor.second != 0) return false;
        }
        return true;
    }
    
    bool substract(unordered_map<char, int>& cnt, string& word) {
        bool valid = true;
        for (char chr: word) {
            cnt[chr]--;
            if (cnt[chr] < 0) valid = false;
        }
        return valid;
    }
    void add(unordered_map<char, int>& cnt, string& word) {
        for (char chr: word) {
            cnt[chr]++;
        }
    }
    void backtrack(unordered_map<char, int>& cnt, vector<int>& track) {
        if (nums.size()) return;
        if (check(cnt)) {
            nums = track;
            return;
        }
        for (int i = 0; i < representations.size(); i++) {
            string word = representations[i];
            if (substract(cnt, word)) {
                track.push_back(i);
                backtrack(cnt, track);
                track.pop_back();
            }
            add(cnt, word);
        }
    }
    
    
    string originalDigits(string s) {
        unordered_map<char, int> cnt;
        for (char chr: s) cnt[chr]++;
        
        vector<int> track;
        backtrack(cnt, track);
        sort(nums.begin(), nums.end());
        string ans;
        for (int num: nums) {
            ans += (char) num + '0';
        }
        return ans;
    }
