class Solution:
    def isNumber(self, s: str) -> bool:
        def checkInteger(string):
            # check sign
            size = len(string)
            string = string.lstrip("+-")

            # check type and One or more digits.
            if size - len(string) > 1 or len(string) == 0:
                return False
            for l in string:
                if not l.isdigit(): return False
            return True
    
        def checkDecimal(string):
            # check sign
            size = len(string)
            string = string.lstrip("+-")

            # check type and One or more digits.
            if size - len(string) > 1 or len(string) <= 1:
                return False
            if string.count(".") != 1:
                return False
            
            for l in string:
                if not l.isdigit() and l != ".": return False
            return True
    
        s = s.upper().split("E")
        if len(s) > 2: return False
        if len(s) == 1:
            return checkInteger(s[0]) or checkDecimal(s[0])
        return (checkInteger(s[0]) or checkDecimal(s[0])) and checkInteger(s[1])
