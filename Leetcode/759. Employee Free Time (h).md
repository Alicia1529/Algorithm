class Solution {
public:
    vector<Interval> employeeFreeTime(vector<vector<Interval>> schedule) {
        vector<Interval> flattened;
        for (auto& worker: schedule) {
            for (auto& interval: worker) {
                flattened.emplace_back(interval);
            }
        }
        sort(flattened.begin(), flattened.end(), [](const Interval& i1, const Interval& i2) {
            return i1.start <= i2.start;
        });
        vector<Interval> working_hours;
        for (auto& interval: flattened) {
            if (!working_hours.empty() && interval.start <= working_hours.back().end) {
                working_hours.back().end = max(working_hours.back().end, interval.end);
            } else {
                working_hours.emplace_back(interval);
            }
        }
        vector<Interval> ans;
        for (int i = 0; i < (int) working_hours.size() - 1; i++) {
            Interval free_period(working_hours[i].end, working_hours[i + 1].start);
            ans.emplace_back(free_period);
        }
        return ans;
        
    }
};
