### dynamic programming
```c++
class Solution {
public:
    int longestValidParentheses(string s) {
        int n = s.size();
        vector<int> dp(n, 0);
        int ans = 0;

        for (int i = 1; i < n; i++) {
            if (s[i - 1] == '(' && s[i] == ')') {
                dp[i] = (i - 2 >= 0)? dp[i - 2] + 2: 2;
            } else if (s[i - 1] == ')' && s[i] == ')') {
                int j = i - dp[i - 1] - 1;
                if (j >= 0 && s[j] == '(') {
                    int prev = (j - 1 >= 0) ? dp[j - 1]: 0;
                    dp[i] = prev + 2 + dp[i - 1];
                }
                
            }
            if (dp[i] > ans) ans = dp[i];
        }
        return ans;
    } 
};
```
### stack
```c++
class Solution {
public:
    int longestValidParentheses(string s) {
        int n = s.size();
        stack<int> stk;
        int ans = 0;
        
        stk.push(-1);
        
        for (int i = 0; i < n; i++) {
            if (s[i] == '(') {
                stk.push(i);
            } else {
                // 每一个)都对应消耗一个(,但是如果没有(则消耗自己
                stk.pop();
                if (stk.empty()) {
                    // 如果是消耗的自己，则重置一下
                    stk.push(i);
                } else {
                    ans = max(ans, i - stk.top());
                }
            }
        }
        return ans;
    }
};
```
