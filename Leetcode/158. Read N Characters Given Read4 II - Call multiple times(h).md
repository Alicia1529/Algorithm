class Solution {
public:
    /**
     * @param buf Destination buffer
     * @param n   Number of characters to read
     * @return    The number of actual characters read
     */
    queue<char> queue;
    char buf4[4];
    int read(char *buf, int n) {
        int read_size = 4;
        while (queue.size() < n && read_size == 4) {
            read_size = read4(buf4);
            for (int i = 0; i < read_size; i++) {
                queue.push(buf4[i]);
            }
        }
        int q_size = queue.size();
        for (int i = 0; i < n && i < q_size; i++) {
            buf[i] = queue.front();
            queue.pop();
        }
        return min(q_size, n);
    }
};

class Solution {
public:
    /**
     * @param buf Destination buffer
     * @param n   Number of characters to read
     * @return    The number of actual characters read
     */
    queue<int> cache;

    int read(char *buf, int n) {
        int offset = 0;
        int cache_size = cache.size();
        for (int i = 0; i < min(cache_size, n); i++) {
            buf[offset++] = cache.front();
            cache.pop();
        }
        int m = 4;
        while (m != 0 && offset < n) {
            m = read4(buf + offset);
            offset += m;
        }
        for (int i = n; i < offset; i++) {
            cache.push(buf[i]);
            buf[i] = '\0';
        }
        return min(offset, n);
        
    }
};
