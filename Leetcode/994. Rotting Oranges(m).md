        dirs = [(0,1),(0,-1),(1,0),(-1,0)]
        while queue and fresh > 0:
            minutes += 1
            for _ in range(len(queue)):
                i, j = queue.popleft()
                for di, dj in dirs:
                    ni, nj = i + di, j + dj
                    if 0 <= ni < m and 0 <= nj < n and grid[ni][nj] == 1:
                        grid[ni][nj] = 2
                        fresh -= 1
                        queue.append((ni, nj))
        return minutes if fresh == 0 else -1

class Solution:
    def rottenForSides(self, grid, i, j, next_rotten, fresh):
        if grid[i][j] == -1:
            return fresh
        # left
        if j - 1 >= 0 and grid[i][j-1] == 1:
            grid[i][j-1] = 2
            next_rotten.add((i, j-1))
            fresh -= 1
        # right
        if j + 1 < len(grid[0]) and grid[i][j+1] == 1:
            grid[i][j+1] = 2
            next_rotten.add((i, j+1))
            fresh -= 1
        # up
        if i - 1 >= 0 and grid[i-1][j] == 1:
            grid[i-1][j] = 2
            next_rotten.add((i-1, j))
            fresh -= 1
        # down
        if i + 1 < len(grid) and grid[i+1][j] == 1:
            grid[i+1][j] = 2
            next_rotten.add((i+1, j))
            fresh -= 1
        return fresh
    def orangesRotting(self, grid: List[List[int]]) -> int:
        # identify all rotten oranges -> start spreading
        # identify all fresh oranges -> when to stop
        # when to end: Max(M,N)
        m = len(grid)
        n = len(grid[0])
        # identify all rotten oranges -> start spreading
        # identify all fresh oranges -> when to stop
        rotten = set()
        fresh = 0
        for i in range(m):
            for j in range(n):
                if grid[i][j] == 2:
                    rotten.add((i,j))
                if grid[i][j] == 1:
                    fresh += 1
        for k in range(m*n):
            if fresh == 0:
                return k
            next_rotten = set()
            while rotten:
                i, j = rotten.pop()
                fresh = self.rottenForSides(grid, i, j, next_rotten, fresh)
            rotten = next_rotten
        if fresh == 0:
                return m+n
        return -1
