// Runtime: 20 ms, faster than 93.46% of C++ online submissions for Count Binary Substrings.
// Memory Usage: 10.6 MB, less than 41.32% of C++ online submissions for Count Binary Substrings.

class Solution {
public:
    int countBinarySubstrings(string s) {
        int ans = 0, prev = 0, curr = 1;
        for (int i = 1; i < s.size(); i++) {
            if (s[i] != s[i - 1]) {
                ans += min(prev, curr);
                prev = curr;
                curr = 1;
            } else {
                curr++;
            }
        }
        return ans + min(prev, curr);
    }
};

class Solution {
public:
    int countBinarySubstrings(string s) {
        int n = s.size();
        vector<int> cnt = {0, 0};
        
        int ans = 0;
        for (int i = 0; i < n; i++) {
            int val = (int) (s[i] - '0');
            cnt[val]++;
            
            for (int j = i + 1; j < n && j <= i + cnt[val]; j++) {
                if (s[j] == (char) ('0' + 1 - val)) {
                    ans++;
                } else {
                    break;
                }
            }
            if (i + 1 < n && s[i] != s[i + 1]) cnt = {0, 0};
        }
        return ans;
    }
};
