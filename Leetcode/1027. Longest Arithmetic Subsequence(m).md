class Solution {
public:
    int longestArithSeqLength(vector<int>& nums) {
        int n = nums.size();
        int result = 2;
        int dp[n][1001];
        memset(dp, 0, sizeof(dp));
        for (int i = 0; i < nums.size(); i++) {
            for (int j = i - 1; j >= 0; j--) {
                int diff = nums[i] - nums[j] + 500;
                dp[i][diff] = max(dp[i][diff], dp[j][diff] == 0? 2: dp[j][diff] + 1);
                result = max(result, dp[i][diff]);
            }
            
        }
        return result;
    }
};
