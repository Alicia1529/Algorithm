// Runtime: 15 ms, faster than 51.08% of C++ online submissions for High Five.
// Memory Usage: 10.5 MB, less than 80.51% of C++ online submissions for High Five.

class Solution {
public:
    vector<vector<int>> highFive(vector<vector<int>>& items) {
        vector<vector<int>> ans;
        unordered_map<int, vector<int>> scores;
        for (auto& item: items) {
            scores[item[0]].push_back(item[1]);
        }
        for (auto& [id, v]: scores) {
            sort(v.begin(), v.end(), greater<int>() );
            ans.push_back({id, (v[0] + v[1] + v[2] + v[3] + v[4]) / 5});
        }
        sort(ans.begin(), ans.end());
        return ans;
    }
};
