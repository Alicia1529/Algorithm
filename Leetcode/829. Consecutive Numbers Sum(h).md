class Solution {
public:
    int consecutiveNumbersSum(int N) {
        int ans = 0;
        int end = ceil((float) N / 2);
        for (int i = 1; i <= end; i++) {
            if (i % 2 == 1 && N % i == 0 && N / i - i / 2 > 0) {
                ans++;
            }
            if (i % 2 == 0) {
                float quotient = (float) N / i;
                if (quotient - N / i == 0.5 && N / i - i / 2 + 1 > 0)
                    ans++;
            }
        }
        return ans;
        
    }
};
