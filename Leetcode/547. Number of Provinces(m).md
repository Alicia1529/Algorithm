class Solution:
    def findCircleNum(self, isConnected: List[List[int]]) -> int:
        n = len(isConnected)
        result = 0
        seen = set()
        
        def dfs(start):
            seen.add(start)
            for end in range(n):
                if isConnected[start][end] and end not in seen:
                    dfs(end)
        
        for i in range(n):
            if i not in seen:
                result += 1
                dfs(i)
        
        return result



class Solution:
    def findCircleNum(self, isConnected: List[List[int]]) -> int:
        n = len(isConnected)
        self.size = n
        parent = [i for i in range(n)]
        
        def union(i, j):
            root_i = find(i)
            root_j = find(j)
            if root_i == root_j:
                return
            parent[root_j] = root_i
            self.size -= 1

        def find(i):
            if parent[i] != i:
                parent[i] = find(parent[i])
            return parent[i]
        
        for i in range(n):
            for j in range(i + 1, n):
                if isConnected[i][j]:
                    union(i, j)
        return self.size
