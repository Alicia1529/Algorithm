// Runtime: 0 ms, faster than 100.00% of C++ online submissions for Integer Break.
// Memory Usage: 6.3 MB, less than 23.36% of C++ online submissions for Integer Break.
class Solution {
public:
    int integerBreak(int n) {
        vector<int> dp(n, 0);
        dp[0] = 1; // max product of 1
        dp[1] = 1; // max product of 2
        for (int i = 3; i <= n; i++) {
            int tmp = 0;
            for (int j = 1; j < i; j++) {
                if (max(dp[i - j - 1], i - j) * j > tmp) tmp = max(dp[i - j - 1], i - j) * j;
            }
            dp[i - 1] = tmp;
        }
        return dp[n - 1];
    }
};
