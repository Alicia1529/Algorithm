```c++
class Solution {
public:
    void addDir(string path, int start, int end, stack<string>& s) {
        if (start >= end) return;
        string sub = path.substr(start, end - start);
        if (sub == ".")
            return;
        if (sub == "..") {
            if (!s.empty()) s.pop();
        } else {
            s.push("/" + sub);
        }
    }
    string simplifyPath(string path) {
        path.append("/");
        stack<string> s;
        int start = 0;
        for (int i = 0; i < path.size(); i++) {
            if (path[i] == '/') {
                addDir(path, start, i, s);
                start = i + 1;
            }
        }
        string ret = "";
        while (!s.empty()) {
            ret = s.top() + ret;
            s.pop();
        }
        return (ret == "") ? "/" : ret;
    }
};
```

```python
class Solution:
    def simplifyPath(self, path: str) -> str:
        # "//.//an/".split("/") -> ['', '', '.', '', 'an', '']
        path = path.split("/")
        
        # empty space -> skip
        # . ->skip
        # ..     /home/../next ->/next, /../next
        
        ans = []
        for p in path:
            if p == "." or p == "": continue
            if p == ".." and len(ans):
                ans.pop()
            elif p != "..":
                ans.append(p)
        return "/" + "/".join(ans)
```
