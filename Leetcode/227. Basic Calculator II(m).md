```c++
// Runtime: 12 ms, faster than 61.58% of C++ online submissions for Basic Calculator II.
// Memory Usage: 8.6 MB, less than 45.68% of C++ online submissions for Basic Calculator II.
class Solution {
public:
    int calculate(string s) {
        stack<int> stack;
        char op = '+';
        int num = 0, cumu = 0;
        for (int i = 0; i < s.size(); i++) {
            if (isdigit(s[i])) {
                num = num * 10 + (int) (s[i] - '0');
            }
            if (!isdigit(s[i]) && !iswspace(s[i]) || i == s.size() - 1) {
                if (op == '+') {
                    cumu += num;
                } else if (op == '-') {
                    cumu -= num;
                } else if (op == '*') {
                    num *= stack.top();
                    stack.pop();
                    stack.push(num);
                } else {
                    num  = stack.top() / num;
                    stack.pop();
                    stack.push(num);
                }
                num = 0;
                op = s[i];
            }
        }
        int ans = 0;
        while (!stack.empty()) {
            ans += stack.top();
            stack.pop();
        }
        return ans;
    }
};
```

```C++
                    cumu += num;
                } else if (op == '-') {
                    num = -num;
                    cumu += num;
                } else if (op == '*') {
                    cumu -= last_num;
                    num = last_num * num;
                    cumu += num;
                } else {
                    cumu -= last_num;
                    num = last_num / num;
                    cumu += num;
                }
                last_num = num;
                num = 0;
                op = s[i];
            }
        }
        return cumu;
    }
};
```
