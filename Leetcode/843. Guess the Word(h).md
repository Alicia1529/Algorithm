/**
 * // This is the Master's API interface.
 * // You should not implement it, or speculate about its implementation
 * class Master {
 *   public:
 *     int guess(string word);
 * };
 */
class Solution {
public:
    unordered_map<string, unordered_map<int, unordered_set<string>>> graph;
    unordered_set<string> candidates;
    
    int getDistance(string& s1, string& s2) {
        int cnt = 0;
        for (int i = 0; i < s1.size(); i++) {
            if (s1[i] != s2[i]) cnt++;
        }
        return cnt;
    }
    void dfs(Master& master) {
        if (candidates.empty()) return;
        string word = *candidates.begin();
        candidates.erase(word);

        int matches = master.guess(word);
        if (matches == 6) return;
    
        auto complement = graph[word][6 - matches];
        
        unordered_set<string> next;
        for (auto itor = candidates.begin(); itor != candidates.end(); itor++) {
            if (complement.find(*itor) != complement.end()) next.insert(*itor);
        }
        candidates = next;
        dfs(master);
    }
    
    void findSecretWord(vector<string>& wordlist, Master& master) {   
        int n = wordlist.size();
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (i == j) continue;
                string w1 = wordlist[i], w2 = wordlist[j];
                int d = getDistance(w1, w2);
                graph[w1][d].insert(w2);
            }
        }
        candidates = unordered_set(wordlist.begin(), wordlist.end());
        dfs(master);
    }
};
