 /**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<vector<int>> verticalOrder(TreeNode* root) {
        if (root == nullptr) return {};
        // offset (horizontal), level (vertical), order(within a line), value
        unordered_map<int, vector<int>> columns;
        queue<pair<TreeNode*, int>> q;
        q.push({root, 0});
        
        TreeNode* node;
        int level = 0;
        while (q.size()) {
            int size = q.size();
            for (int i = 0; i < size; i++) {
                auto pair = q.front();
                q.pop();
                node = pair.first;
                int offset = pair.second;
                columns[offset].push_back(node->val);
                if (node->left) q.push({node->left, offset - 1});
                if (node->right) q.push({node->right, offset + 1});
            }
            level++;
        }

        vector<vector<int>> ans;
        for (int i = -level; i < level; i++) {
            if (columns.find(i) != columns.end()) ans.emplace_back(columns[i]);
        }
        return ans;
    }
};


class Solution {
public:
    vector<vector<int>> verticalOrder(TreeNode* root) {
        if (root == nullptr) return {};
        // offset (horizontal), level (vertical), order(within a line), value
        vector<vector<int>> positions;
        queue<pair<TreeNode*, int>> q;
        q.push({root, 0});
        
        TreeNode* node;
        int level = 0;
        while (q.size()) {
            int size = q.size();
            for (int i = 0; i < size; i++) {
                auto pair = q.front();
                q.pop();
                node = pair.first;
                int offset = pair.second;
                positions.push_back({offset, level, i, node->val});
                if (node->left) q.push({node->left, offset - 1});
                if (node->right) q.push({node->right, offset + 1});
            }
            level++;
        }
        sort(positions.begin(), positions.end(), [](vector<int>& v1, vector<int>& v2) {
            if (v1[0] != v2[0]) return v1[0] < v2[0];
            if (v1[1] != v2[1]) return v1[1] < v2[1];
            if (v1[2] != v2[2]) return v1[2] < v2[2];
            return true;
        });
        
        vector<vector<int>> ans;
        vector<int> column;
        int c = INT_MIN;
        for (auto p: positions) {
            if (!column.empty() && c != p[0]) {
                ans.emplace_back(column);
                column.clear();
            }
            column.push_back(p[3]);
            c = p[0];
        }
        ans.emplace_back(column);
        return ans;
    }
};

class Solution:
    def verticalOrder(self, root: Optional[TreeNode]) -> List[List[int]]:
        if root is None: return []
        # do a traversal and give a label to each node(position at current layer, depth)
        queue = [(0, 1, root)]
        nodes = defaultdict(list) # key: position, val = []
        while len(queue):
            size = len(queue)
            for i in range(size):
                p, d, n = queue.pop(0)
                nodes[p].append(n.val)
                if n.left:
                    queue.append((p - 1, d + 1, n.left))
                if n.right:
                    queue.append((p + 1, d + 1, n.right))
                    
        ans = []
        for k in sorted(nodes.keys()):
            ans.append(nodes[k])
        return ans
