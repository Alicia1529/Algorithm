class Solution {
public:
    int ans = INT_MIN;
    int helper(TreeNode* root) {
        if (root == nullptr) return 0;
        
        int left = helper(root->left);
        int right = helper(root->right);
        
        if (left + right + root->val > ans)
            ans = left + right + root->val;
        return max(max(left, right) + root->val, 0);
        
    }
    int maxPathSum(TreeNode* root) {
        if (root == nullptr) return 0;
        
        helper(root);
        return ans;        
        
    }
};

        self.maxSum = float("-inf")
        
        def pathSumImpl(root):
            if root is None:
                return 0
            left = max(pathSumImpl(root.left), 0)
            right = max(pathSumImpl(root.right), 0)
            curr = root.val
            if left + right + curr > self.maxSum:
                self.maxSum = left + right + curr
            return curr + max(left, right)
            
        pathSumImpl(root)

Explicit-leaf variant — handles the leaf as its own base case and enumerates the
fork cases at each node:

```python
class Solution:
    def maxPathSum(self, root: Optional[TreeNode]) -> int:
        self.max_sum = float('-inf')

        def helper(node):
            if node.left is None and node.right is None:
                self.max_sum = max(self.max_sum, node.val)
                return node.val
            sum_left, sum_right = 0, 0
            if node.left is not None:
                sum_left = helper(node.left)
            if node.right is not None:
                sum_right = helper(node.right)
            # When finding the maximum value, below cases are counted
            #   X  |   X      |      X
            #      |    X     |   X    X
            self.max_sum = max(self.max_sum, node.val + max([0, sum_left, sum_right, sum_left + sum_right]))
            # When returning value to its parent, only below cases are counted
            #   X  |   X      |
            #      |    X     |
            return node.val + max([0, sum_left, sum_right])

        helper(root)
        return self.max_sum
```

Clean variant — `None` is the base case, clamp each branch at 0 to drop negative gains:

```python
class Solution:
    def maxPathSum(self, root: Optional[TreeNode]) -> int:
        self.max_sum = float('-inf')

        def helper(node):
            if node is None:
                return 0
            # clamp at 0: drop a branch whose gain is negative
            sum_left  = max(0, helper(node.left))
            sum_right = max(0, helper(node.right))
            # answer candidate: path may fork through this node
            self.max_sum = max(self.max_sum, node.val + sum_left + sum_right)
            # gain returned upward: parent can extend only one branch
            return node.val + max(sum_left, sum_right)

        helper(root)
        return self.max_sum
```
