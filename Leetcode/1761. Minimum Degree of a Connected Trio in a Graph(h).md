class Solution:

    def minTrioDegree(self, n: int, edges: List[List[int]]) -> int:
        import numpy as np
        mininum_degree = float("inf")
        graph = np.zeros((n, n))
        degree = defaultdict(int)
        for n1, n2 in edges:
            graph[n1 - 1][n2 - 1] = 1
            graph[n2 - 1][n1 - 1] = 1
            degree[n1 - 1] += 1
            degree[n2 - 1] += 1
        for i in range(n):
            for j in range(i, n):
                for k in range(j, n):
                    if graph[i][j] == 1 and graph[j][k] == 1 and graph[i][k] == 1:
                        mininum_degree = min(mininum_degree, degree[i] + degree[j] + degree[k] - 6)
        return -1 if mininum_degree == float("inf") else mininum_degree
