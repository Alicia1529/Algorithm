**Two bugs to watch for** (both about single-row / single-column rings):
1. Base case must stop at `height <= 0 or width <= 0` (empty ring), not `< 0`.
2. Only walk the bottom row + left column when `height > 1 and width > 1` — otherwise
   a single-row ring re-walks the top row (duplicates), a single-col ring re-walks
   the left column.

### Python

```python
# Recursive: peel one ring at a time.
class Solution:
    def spiralOrder(self, matrix: List[List[int]]) -> List[int]:
        m, n = len(matrix), len(matrix[0])
        output = []
        def helper(x, y, height, width):
            if height <= 0 or width <= 0:      # empty ring → stop
                return
            # top row: left → right
            for i in range(width):
                output.append(matrix[x][y+i])
            # right col: top → bottom (skip the corner already taken)
            for j in range(1, height):
                output.append(matrix[x+j][y+width - 1])
            if height > 1 and width > 1:       # only a real 2D ring has bottom + left
                # bottom row: right → left
                for i in range(width - 2, -1, -1):
                    output.append(matrix[x + height - 1][y + i])
                # left col: bottom → top
                for j in range(height - 2, 0, -1):
                    output.append(matrix[x + j][y])
            helper(x+1, y+1, height - 2, width - 2)

        helper(0, 0, m, n)
        return output

# Iterative: shrink four boundaries. The two `if` guards are the same fix as above.
#
# KEY INSIGHT: top/bottom/left/right bound the UNVISITED rectangle. Walk one side,
# then retract that edge inward. Invariant: after each ±=, [top..bottom]×[left..right]
# is exactly the not-yet-added cells; the loop stops when the box empties (edges cross).
# The two guards matter because after the top+right passes the box may have collapsed
# to a single row/col — the bottom/left passes must re-check before running.
class Solution:
    def spiralOrder(self, matrix: List[List[int]]) -> List[int]:
        res = []
        top, bottom = 0, len(matrix) - 1
        left, right = 0, len(matrix[0]) - 1
        while top <= bottom and left <= right:
            for c in range(left, right + 1):         # top row →
                res.append(matrix[top][c])
            top += 1
            for r in range(top, bottom + 1):         # right col ↓
                res.append(matrix[r][right])
            right -= 1
            if top <= bottom:                         # still a row left?
                for c in range(right, left - 1, -1):  # bottom row ←
                    res.append(matrix[bottom][c])
                bottom -= 1
            if left <= right:                         # still a col left?
                for r in range(bottom, top - 1, -1):  # left col ↑
                    res.append(matrix[left][r])
                left += 1
        return res

# Iterative variant: shrink m/n and flip a single `direction` instead of tracking
# four boundaries. Each loop does one horizontal + one vertical pass; flipping
# direction turns the spiral. The len(output) < total guard stops the final pass.
class Solution:
    def spiralOrder(self, matrix: List[List[int]]) -> List[int]:
        m, n = len(matrix), len(matrix[0])
        total = m * n
        output = []
        i, j = 0, -1
        direction = 1
        while len(output) < total:
            for _ in range(n):                # horizontal pass (n columns)
                j += direction
                output.append(matrix[i][j])
            m -= 1                             # one fewer row to walk vertically
            for _ in range(m):                # vertical pass (m rows)
                i += direction
                output.append(matrix[i][j])
            n -= 1                             # one fewer column next time
            direction *= -1                    # flip: next passes go the other way
        return output
```

- All O(m·n) time, O(1) extra space (besides output). The 4-boundary version is
  usually easiest to get right in an interview — no recursion depth, and the "did I
  already consume this row/col?" guards are right where you look. The direction-flip
  variant is the most compact.

### C++
```c++
// Runtime 0ms, 100%
class Solution {
public:
    vector<int> spiralOrder(vector<vector<int>>& matrix) {
        vector<int> res;
        int m = matrix.size();
        int n = matrix[0].size();
        int i = 0, j = -1;
        int rows = m, cols = n;
        int row_start = 0, col_start=0;
        while (res.size() < m * n) {
            while (res.size() < m * n && j < cols - 1)
                res.push_back(matrix[i][++j]);
            while (res.size() < m * n && i < rows - 1)
                res.push_back(matrix[++i][j]);
            while (res.size() < m * n && j > col_start)
                res.push_back(matrix[i][--j]);
            while (res.size() < m * n && i > row_start + 1)
                res.push_back(matrix[--i][j]);
            rows--, cols--;
            row_start++, col_start++;
        }
        return res;
    }
};
```
