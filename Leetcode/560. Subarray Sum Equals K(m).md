class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        ret = 0
        total = 0
        cache = {0: 1}
        for num in nums:
            total += num
            ret += cache.get(total - k, 0)

        return ret

// Memory Usage: 36.2 MB, less than 60.91% of C++ online submissions for Subarray Sum Equals K.
class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        unordered_map<int, int> cache;
        cache.emplace(0, 1);
        int ret = 0;
        int sum = 0;
        for (auto num: nums)
        {
            sum += num;
            if (cache.find(sum - k) != cache.end())
                ret += cache[sum - k];
            cache[sum]++;
        }
        return ret;
    }
};
