```c++
class Solution {
public:
    int ans = 0;
    
    int dfs(TreeNode* root) {
        if (root == nullptr) return -1;
        int left = dfs(root->left);
        int right = dfs(root->right);
        if (left + right + 2 > ans) ans = left + right + 2;
        return max(left, right) + 1;
        
    }
    int diameterOfBinaryTree(TreeNode* root) {
        dfs(root);
        return ans;
    }
};
```
