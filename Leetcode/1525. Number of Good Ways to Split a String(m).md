// Memory Usage: 8.6 MB, less than 47.63% of C++ online submissions for Number of Good Ways to Split a String.

class Solution {
public:
    int numSplits(string s) {
        if (s.size() <= 1) return 0;
        int ans = 0;
        unordered_map<char, int> right;
        for (char c: s) right[c]++;
        unordered_set<char> left;
        for (char c: s) {
            left.insert(c);
            right[c]--;
            if (right[c] == 0) right.erase(c);
            if (left.size() == right.size()) ans++;
        }
        return ans;
        
    }
};
