class Solution {
public:
    bool isvalid(unordered_map<char, int>& map) {
        for (auto itor: map) {
            if (itor.second > 0) return false;
        }
        return true;
    }
    string minWindow(string s, string t) {
        unordered_map<char, int> map;
        for (char chr: t) map[chr]++;
        
        
        string ans(s);
        ans += "0";
        
        int l = 0;
        for (int r = 0; r < s.size(); r++) {
            char chr = s[r];

            if (map.find(chr) == map.end()) continue;
            map[chr]--;
            
            if (map[chr] != 0 || !isvalid(map)) continue;
            
            while (l < s.size()) {
                if (map.find(s[l]) == map.end()) {
                    l++;
                } else if (map.at(s[l]) < 0) {
                    map[s[l++]]++;
                } else {
                    break;
                }
            }
    
            if (r - l + 1 < ans.size()) ans = s.substr(l, r - l + 1);
            map[s[l++]]++;
        }

        return ans.size() == s.size() + 1 ? "": ans;
    }
};
