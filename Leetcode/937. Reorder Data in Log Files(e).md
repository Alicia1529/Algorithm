// Runtime: 12 ms, faster than 69.22% of C++ online submissions for Reorder Data in Log Files.
// Memory Usage: 12.4 MB, less than 41.45% of C++ online submissions for Reorder Data in Log Files.
class Solution {
public:
    vector<string> reorderLogFiles(vector<string>& logs) {
        if (logs.size() <= 1) return logs;
        vector<string> letter_logs;
        vector<string> digit_logs;
        for (auto& log: logs) {
            auto idx = log.find(" ");
            if (log.npos == idx || (int) idx == (int) log.size() - 1)
                exit(-1);
            if (isdigit(log[idx + 1]))
                digit_logs.push_back(log);
            else
                letter_logs.push_back(log);
        }
        sort(letter_logs.begin(), letter_logs.end(), [](const string& s1, const string &s2) {
            auto idx1 = s1.find(" ");
            auto id1 = s1.substr(0, idx1);
            auto string1 = s1.substr(idx1);

            auto idx2 = s2.find(" ");
            auto id2 = s2.substr(0, idx2);
            auto string2 = s2.substr(idx2);
            return (string1 == string2)? id1 < id2: string1 < string2;
        });
        for (auto& log: digit_logs) {
            letter_logs.emplace_back(log);
        }
        return letter_logs;
    }
};
