class Solution:
    def evalRPN(self, tokens: List[str]) -> int:
        stack = []
        for token in tokens:
            if token in ('+', "-", '*', '/'):
                ele2 = stack.pop()
                ele1 = stack.pop()
                if token == '+':
                    num = ele1 + ele2
                elif token == '-':
                    num = ele1 - ele2
                elif token == '*':
                    num = ele1 * ele2
                elif token == '/':
                    num = int(ele1 / ele2)
                stack.append(num)
            else:
                stack.append(int(token))
        return stack[-1]

class Solution {
public:
    int evalRPN(vector<string>& tokens) {
        stack<int> values;
        for (auto s: tokens) {
            if (s == "+" || s == "-" || s == "*" || s == "/") {
                int right = values.top(); values.pop();
                int left = values.top(); values.pop();
                if (s == "+") values.push(left + right);
                if (s == "-") values.push(left - right);
                if (s == "*") values.push(left * right);
                if (s == "/") values.push(left / right);
            } else {
                values.push(stoi(s));
            }
        }
        return values.top();
    }
};
