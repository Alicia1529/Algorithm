// Runtime: 8 ms, faster than 67.66% of C++ online submissions for Delete and Earn.
// Memory Usage: 10 MB, less than 50.22% of C++ online submissions for Delete and Earn.
class Solution {
public:
    int deleteAndEarn(vector<int>& nums) {
        unordered_map<int, int> m;
        for (auto num: nums) m[num]++;

        vector<int> keys;
        for (auto& p: m) keys.push_back(p.first);
        sort(keys.begin(), keys.end());
        
        int ans = 0;
        vector<int> dp(1, 0);
        for (int i = 0; i < keys.size(); i++) {
            int key = keys[i];
            int curr = m[key] * key;
            if (i > 0 && keys[i] - keys[i - 1] == 1) {
                curr = max(curr + dp[i - 1], dp[i]);
            } else {
                curr += dp[i];
            }
            ans = max(ans, curr);
            dp.push_back(curr);
        }
        
        return ans;
    }
};
