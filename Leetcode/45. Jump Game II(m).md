Greedy Algorithm
```c++
// Runtime: 12 ms, faster than 83.74% of C++ online submissions for Jump Game II.
// Memory Usage: 16.3 MB, less than 56.63% of C++ online submissions for Jump Game II.
class Solution {
public:
    int jump(vector<int>& nums) {
        int jumps = 0, curr = 0, farthest = 0;
        // take care of the -1 here
        for (int i = 0; i < nums.size() - 1; i++) {
            farthest = max(farthest, i + nums[i]);
            if (curr == i) {
                jumps++;
                curr = farthest;
            }
        }
        return jumps;
        
    }
};
```

Dynamic Programming
```c++
// Runtime: 404 ms, faster than 16.52% of C++ online submissions for Jump Game II.
// Memory Usage: 17 MB, less than 11.38% of C++ online submissions for Jump Game II.
class Solution {
public:
    int jump(vector<int>& nums) {
        vector<int> memo(nums.size(), INT_MAX);
        memo[0] = 0;
        
        for (int i = 0; i < nums.size(); i++) {
            if (memo[i] == INT_MAX) continue;
            for (int j = 1; j <= nums[i] && i + j < nums.size(); j++) {
                memo[i + j] = min(memo[i + j], memo[i] + 1);
            }
        }
        return memo[nums.size() - 1];
        
    }
};
```
