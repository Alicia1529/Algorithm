class Solution {
public:
    int calculate(string s) {
        stack<int> nums, ops;
        int res = 0;
        int sign = 1;
        int num = 0;
        for (char ch: s)
        {
            if (isdigit(ch)) {
                num = num * 10 + (int) (ch - '0');
            } else {
                res += sign * num;
                num = 0;
                if (ch == '+') sign = 1;
                if (ch == '-') sign = -1;
                if (ch == '(') {
                    nums.push(res);
                    ops.push(sign);
                    res = 0;
                    sign = 1;
                }
                if (ch == ')' && ops.size()) {
                    res = ops.top() * res + nums.top();
                    ops.pop(); nums.pop();
                }
            }
        
        res += sign * num;
        return res;
        
    }
};

class Solution {
public:
    int calculateImpl(string& s, int start, int end) {
        int cum = 0, num = 0, sign = 1;
        int i = start;
        while (i <= end) {
            if (s[i] == '(') {
                int cnt = 1;
                i++;
                int recursion_start = i;
                while (i <= end && cnt != 0) {
                    if (s[i] == '(') cnt++;
                    if (s[i] == ')') cnt--;
                    i++;
                }
                num = calculateImpl(s, recursion_start, i - 2);
                continue;
            };
            if (isdigit(s[i])) {
                int val = (int) (s[i] - '0');
                num = num * 10 + val;
            } else if (s[i] == '+') {
                cum += num * sign;
                num = 0;
                sign = 1;
            } else if (s[i] == '-') {
                cum += num * sign;
                num = 0;
                sign = -1;
            }
            // s[i] == " ", do nothing
            i++;
        }
        cum += sign * num;
        return cum;
    }
    int calculate(string s) {
        return calculateImpl(s, 0, s.size() - 1);
    }
};
