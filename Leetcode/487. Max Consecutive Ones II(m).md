class Solution:
    def findMaxConsecutiveOnes(self, nums: List[int]) -> int:
        longestSequence = 0

        n = len(nums)
        numZeros = 0
        l = 0
        for r in range(n):
            if nums[r] == 0:
                numZeros += 1
            while numZeros == 2:
                if nums[l] == 0:
                    numZeros -= 1
                l += 1
            longestSequence = max(longestSequence, r - l + 1)
        return longestSequence





class Solution:
    def findMaxConsecutiveOnes(self, nums: List[int]) -> int:
        if len(nums) == 1:
            return 1
        ones = []
        n = len(nums)
        cnt = 0
        for i in range(n):
            if nums[i] == 1:
                cnt += 1
            else:
                ones.append(cnt)
                cnt = 0
        ones.append(cnt)
        if len(ones) == 1:
            return ones[0]
        ans = ones[0] + ones[1]
        for i in range(len(ones) - 1):
            ans = max(ans, ones[i] + ones[i + 1])
        return ans + 1
