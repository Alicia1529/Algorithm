```python
    def postorderTraversal(self, root: Optional[TreeNode]) -> List[int]:
        output, stack = [], []
        prev, curr = None, root
        while curr or stack:
            # Dive left — push every node going left, same as inorder.
            while curr:
                stack.append(curr)
                curr = curr.left
            # Pop a node — but don't visit yet. Check the right subtree
            curr = stack.pop()
            # If there is no right child or the right child was already processed
            if curr.right is None or curr.right == prev:
                output.append(curr.val)
                prev = curr
                curr = None # Ensure we don’t traverse again from this node
            # After visiting, set prev = curr and curr = None so the outer while doesn't dive left again — 
            # it goes back to popping from the stack.
            else:
                stack.append(curr)
                curr = curr.right
        return output
```

```c++
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:

    vector<int> postorderTraversal(TreeNode* root) {
        vector<int> ans;
        stack<TreeNode*> stk;
        TreeNode *prev = nullptr;
        while (root != nullptr || !stk.empty()) {
            while (root) {
                stk.push(root);
                root = root->left;
            }
            root = stk.top();
            stk.pop();
            if (root->right == nullptr || root->right == prev) {
                ans.push_back(root->val);
                prev = root;
                root = nullptr;
            } else {
                stk.push(root);
                root = root->right;
            }
        }
        return ans;
    }
};
```
