class BSTIterator {
public:
    TreeNode* curr;
    stack<TreeNode*> stk;
    BSTIterator(TreeNode* root) {
        curr = root;
        while (curr != nullptr) {
            stk.push(curr);
            curr = curr->left;
        }
    }
    
    int next() {
        curr = stk.top();
        stk.pop();
        int val = curr->val;
        curr = curr->right;
        while (curr != nullptr) {
            stk.push(curr);
            curr = curr->left;
        }
        return val;
        
    }
    
    bool hasNext() {
        return !stk.empty();
    }
};

class BSTIterator {
private:
    int idx;
    vector<int> nodesSorted;
    void _inorder_traversal(TreeNode* root) {
        if (root->left)  _inorder_traversal(root->left);
        nodesSorted.push_back(root->val);
        if (root->right)  _inorder_traversal(root->right);
    }
public:
    BSTIterator(TreeNode* root) {
        idx = -1;
        _inorder_traversal(root);
    }
    
    int next() {
        idx++;
        return nodesSorted[idx];
        
    }
    
    bool hasNext() {
        return idx + 1 < nodesSorted.size();

};

            self.stack.append(root)
