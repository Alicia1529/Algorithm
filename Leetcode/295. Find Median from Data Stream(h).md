Approach 2: Insertion Sort: binary search + insertion sort
    Time complexity: O(N) + O(logN) = O(N)


Approach 3: Two Heaps(Priority Queue)
    Time Complexity: O(logN)
```c++
class MedianFinder {
public:
    /** initialize your data structure here. */
    priority_queue<int, vector<int>, greater<int> > pq_r;
    priority_queue<int> pq_l;

    MedianFinder() {        
    }
    
    void addNum(int num) {
        if (pq_l.empty() && pq_r.empty()) {
            pq_l.push(num);
        } else if (num > pq_l.top()) {
            pq_r.push(num);
        } else {
            pq_l.push(num);
        }
        int r = pq_r.size();
        int l = pq_l.size();
        if (r - l >= 2) {
            int top = pq_r.top();
            pq_r.pop();
            pq_l.push(top);
        }
        if (l - r >= 2) {
            int top = pq_l.top();
            pq_l.pop();
            pq_r.push(top);
        }
        
    }
    
    double findMedian() {
        if (pq_r.size() == pq_l.size()) {
            int sum = pq_r.top() + pq_l.top();
            return (double) sum / 2;
        }
        if (pq_r.size() > pq_l.size())
            return (double) pq_r.top();
        return (double) pq_l.top();
    }
};

/**
 * Your MedianFinder object will be instantiated and called as such:
 * MedianFinder* obj = new MedianFinder();
 * obj->addNum(num);
 * double param_2 = obj->findMedian();
```
