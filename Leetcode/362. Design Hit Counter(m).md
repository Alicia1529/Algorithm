import bisect
class HitCounter:

    def __init__(self):
        self.interval = 300
        self.history = []
        
    # O(1) — just append
    def hit(self, timestamp: int) -> None:
        self.history.append(timestamp)
        
    # O(log n)
    def getHits(self, timestamp: int) -> int:
        # find leftmost index where timestamp >= current - 299
        earliest = max(0, timestamp-self.interval + 1)
        left = bisect.bisect_left(self.history, earliest)
        return len(self.history) - left




class HitCounter:

    def __init__(self):
        self.counter = []
        

    def hit(self, timestamp: int) -> None:
        while len(self.counter) > 0 and timestamp - self.counter[0] >= 300:
            self.counter.pop(0)
        self.counter.append(timestamp)
        

    def getHits(self, timestamp: int) -> int:
        while len(self.counter) > 0 and timestamp - self.counter[0] >= 300:
            self.counter.pop(0)
        return len(self.counter)
