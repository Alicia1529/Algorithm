```c++
class SparseVector {
public:
    vector<int> indices;
    vector<int> values;
    
    SparseVector(vector<int> &nums) {
        for (int i = 0; i < nums.size(); i++) {
            if (nums[i] != 0) {
                indices.push_back(i);
                values.push_back(nums[i]);
            }
        }
    }
    
    // Return the dotProduct of two sparse vectors
    int dotProduct(SparseVector& vec) {
        int l = 0, r = 0, product = 0;
        while (l < indices.size() && r < vec.indices.size()) {
            if (indices[l] < vec.indices[r]) {
                l++;
            } else if (indices[l] > vec.indices[r]) {
                r++;
            } else {
                product += values[l] * vec.values[r];
                l++, r++;
            }
        }
        return product;
    }
};
```

```python
class SparseVector:
    def __init__(self, nums: List[int]):
        self.memo = dict()
        self.n = None
        self.store(nums)
        
    def store(self, nums):
        for i in range(len(nums)):
            if nums[i] != 0:
                self.memo[i] = nums[i]
        self.n = len(nums)

    # Return the dotProduct of two sparse vectors
    def dotProduct(self, vec: 'SparseVector') -> int:
        if self.n != vec.n:
            raise ValueErorr("The size should be the same")
            
        product = 0
        for k in self.memo:
            if k in vec.memo:
                product += self.memo[k] * vec.memo[k]
        return product
```
