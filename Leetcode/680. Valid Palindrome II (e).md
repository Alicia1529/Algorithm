class Solution:
    def helper(self, s, l, r):
        while l <= r:
            if s[l] != s[r]:
                return False
            l += 1
            r -= 1      
        return True

            if s[l] != s[r]:

            l += 1
            r -= 1

    def validPalindrome(self, s: str) -> bool:
        l = 0
        r = len(s) - 1
        while l <= r:
            if s[l] != s[r]:
                left= s[l+1:r+1]
                right = s[l:r]

            r -= 1
