class Solution:
    def sortedSquares(self, nums: List[int]) -> List[int]:
        # find the first non-negative value
        size = len(nums)
        l = 0
        r = size - 1
        ans = [0 for i in range(size)]
        for idx in range(size - 1, -1, -1):
            if nums[r] >= abs(nums[l]):
                val = nums[r]
                r -= 1
            else:
                val = nums[l]
                l += 1
            ans[idx] = val * val
        return ans

    def sortedSquares(self, nums: List[int]) -> List[int]:
        # find the first non-negative value
        l = 0
        r = len(nums) - 1
        idx = r
        ans = [0 for i in range(r + 1)]
        while l <= r:
            if nums[r] >= abs(nums[l]):
                ans[idx] = nums[r]
                r -= 1
            else:
                ans[idx] = nums[l]
                l += 1
            idx -= 1
        for i in range(len(ans)):
            ans[i] *= ans[i]
        return ans
