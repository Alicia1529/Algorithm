class Solution {
public:
    bool isAlienSorted(vector<string>& words, string order) {
        for (int i = 0; i < words.size() - 1; i++) {
            for (int j = 0; j < words[i].size(); j++) {
                if (j > words[i + 1].size()) return false;
                int idx1 = order.find(words[i][j]);
                int idx2 = order.find(words[i + 1][j]);
                if (idx1 > idx2) return false;
                if (idx1 < idx2) break;
            }
        }
        return true;
    }
};

class Solution:
    def helper(self, word1, word2, order):
        size = min(len(word1), len(word2))
        for i in range(size):
            if order.index(word1[i]) < order.index(word2[i]):
                return True
            if order.index(word1[i]) > order.index(word2[i]):
                return False
        return len(word1) <= len(word2)

    def isAlienSorted(self, words: List[str], order: str) -> bool:
        size = len(words)

        for i in range(size - 1):
            if not self.helper(words[i], words[i + 1], order):
                return False
        return True
