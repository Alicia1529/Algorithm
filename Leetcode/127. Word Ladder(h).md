class Solution {
public:
    unordered_map<string, unordered_set<string>> constructGraph(string& beginWord, string& endWord, vector<string>& wordList) {
        unordered_map<string, unordered_set<string>> successor;
        wordList.push_back(beginWord);
        int size = beginWord.size();

        for (int i = 0; i < wordList.size(); i++) {
            string w1 = wordList[i];
            for (int j = 0; j < wordList.size(); j++) {
                if (i == j) continue;
    
                int cnt = 0;
                string w2 = wordList[j];
                for (int k = 0; k < size; k++) {
                    if (w1[k] != w2[k]) cnt++;
                }
                if (cnt == 1)
                    successor[w1].insert(w2);
            }
        }
        wordList.pop_back();
        return successor;
    }
    int ladderLength(string beginWord, string endWord, vector<string>& wordList) {
        if (find(wordList.begin(), wordList.end(), endWord) == wordList.end())
            return 0;
        unordered_map<string, unordered_set<string>> graph = constructGraph(beginWord, endWord, wordList);
        
        queue<string> q;
        q.push(beginWord);

        unordered_set<string> visited;
        unordered_map<string, int> d;
        for (int i = 0; i < wordList.size(); i++)
            d[wordList[i]] = INT_MAX;
        d[beginWord] = 0;
        
        while (!q.empty()) {
            string word = q.front();
            q.pop();
            for (auto v: graph[word]) {
                if (visited.find(v) != visited.end()) continue;
                d[v] = d[word] + 1;
                q.push(v);
                visited.insert(v);
            }
        }        
        return d[endWord] == INT_MAX ? 0: d[endWord] + 1;
        
    }
};
