class Solution {
public:
    bool isvalid(unordered_map<char, int>& map) {
        for (auto itor: map) {
            if (itor.second != 0) return false; 
        }
        return true;
    }
    bool checkInclusion(string s1, string s2) {
        int m = s1.size(), n = s2.size();
        if (m > n) return false;
        
        unordered_map<char, int> map;
        for (auto chr: s1) map[chr]++;
        
        for (int i = 0; i < m; i++)
            if (map.find(s2[i]) != map.end()) map[s2[i]]--;
        if (isvalid(map)) return true;
        
        for (int i = m; i < n; i++) {
            if (map.find(s2[i - m]) != map.end()) map[s2[i - m]]++;
            if (map.find(s2[i]) != map.end()) map[s2[i]]--;
            if (isvalid(map)) return true;
        }
        return false;
    }
};
