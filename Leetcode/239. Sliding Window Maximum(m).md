class Solution:
    def maxSlidingWindow(self, nums: List[int], k: int) -> List[int]:
        n = len(nums)
        left_max = []
        right_max = []
        for i in range(n):
            if i % k == 0:
                left_max.append(nums[i])
            else:
                left_max.append(max(nums[i], left_max[-1]))
                
        for i in range(n - 1, -1, -1):
            if i == n - 1 or i % k == k - 1:
                right_max.append(nums[i])
            else:
                right_max.append(max(right_max[-1], nums[i]))
        
        ans = []
        for i in range(n - k + 1):
            ans.append(max(left_max[i + k - 1], right_max[n - i - 1]))
        return ans

            while(!dq.empty() && nums[dq.back()]<=nums[i])dq.pop_back();

        }

class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        int n = nums.size();
        int left[n];
        int right[n];
        vector<int> ans;
        for (int i = 0; i < n; i++) {
            if (i % k == 0) {
                left[i] = nums[i];
            } else {
                left[i] = max(left[i - 1], nums[i]);
            }
        }
        for (int i = n - 1; i >= 0; i--) {
            if (i == n - 1 || i % k == k - 1) {
                right[i] = nums[i];
            } else {
                right[i] = max(right[i + 1], nums[i]);
            }
        }
        for (int i = 0; i + k - 1< n; i++) {
            ans.push_back(max(left[i + k - 1], right[i]));
        }
        return ans;
    }
};
