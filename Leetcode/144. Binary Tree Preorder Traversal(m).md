class Solution:
    def preorderTraversal(self, root: Optional[TreeNode]) -> List[int]:
        if root is None:
            return []
        ret = [root.val]
        if root.left is not None:
            ret.extend(self.preorderTraversal(root.left))
        if root.right is not None:
            ret.extend(self.preorderTraversal(root.right))
        return ret

class Solution {
public:
    vector<int> ans;
    void traversal(TreeNode* root) {
        if (!root) return;
        ans.push_back(root->val);
        if (root->left) traversal(root->left);
        if (root->right) traversal(root->right);
    }
    vector<int> preorderTraversal(TreeNode* root) {
        traversal(root);
        return ans;
    }
};

// Runtime: 4 ms, faster than 49.90% of C++ online submissions for Binary Tree Preorder Traversal.
// Memory Usage: 8.5 MB, less than 67.85% of C++ online submissions for Binary Tree Preorder Traversal.
class Solution {
public:
    vector<int> preorderTraversal(TreeNode* root) {
        vector<int> ans;
        if (!root) return ans;
        stack<TreeNode*> st;
        TreeNode *node = root;
        while (node != nullptr || !st.empty()) {
            while (node) {
                ans.push_back(node->val);
                st.push(node);
                node = node->left;
            }
            node = st.top();
            st.pop();
            node = node->right;
        }
        return ans;
    }
};
