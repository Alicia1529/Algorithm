Cycle Detection

1. Kahn's algorithm (BFS)
```python
class Solution:
    def canFinish(self, numCourses: int, prerequisites: List[List[int]]) -> bool:
        indegrees = defaultdict(int)
        adjacency = defaultdict(list)
        remaining = numCourses
        for i, j in prerequisites:
            indegrees[i] += 1
            adjacency[j].append(i)

        zeros = []

        for i in range(numCourses):
            if indegrees[i] == 0:
                zeros.append(i)

        while zeros:
            top = zeros.pop()
            remaining -= 1
            for j in adjacency[top]:
                indegrees[j] -= 1
                if indegrees[j] == 0:
                    zeros.append(j)
        
        return remaining == 0
```

```c++
// Runtime: 20 ms, faster than 94.46% of C++ online submissions for Course Schedule.
class Solution {
public:
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        vector<vector<int>> adjacents(numCourses);
        vector<int> degree(numCourses);
        queue<int> zeros;
        int remaining = numCourses;
        for (int i = 0; i < prerequisites.size(); i++) {
            adjacents[prerequisites[i][1]].push_back(prerequisites[i][0]);
            degree[prerequisites[i][0]]++;
        }
        for (int i = 0; i < numCourses; i++) {
            if (degree[i] == 0) {
                zeros.push(i);
            }
        }
        while (!zeros.empty()) {
            int v = zeros.front();
            remaining--;
            zeros.pop();
            for (int i = 0; i < adjacents[v].size(); i++) {
                int w = adjacents[v][i];
                degree[w]--;
                if (degree[w] == 0)
                    zeros.push(w);
            }
        }
        return remaining == 0;
    }
};
```

```c++
// Runtime: 32 ms, faster than 77.55% of C++ online submissions for Course Schedule.
// 1 if still detecting, -1 for done, 0 for unvisited
class Solution {
public:
    // return true if no cycle is detected
    bool dfs(int u, vector<int>& visited, vector<vector<int>>& adjacencies) {
        if (visited[u] == 1) return false;  // back edgee
        if (visited[u] == -1) return true;  // forward edge
        visited[u] = 1;
        for (int v: adjacencies[u]) {
            if (!dfs(v, visited, adjacencies)) return false;
        }
        visited[u] = -1;
        return true;
    }
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        vector<vector<int>> adjacencies(numCourses);
        vector<int> visited(numCourses);
        fill(visited.begin(), visited.end(), 0);
        for (const auto& info: prerequisites) {
            adjacencies[info[1]].push_back(info[0]);
        }
        for (int i = 0; i < numCourses; i++) {
            if (!dfs(i, visited, adjacencies)) return false;
        }
        return true;
    }
};
```
```python
class Solution:
    def canFinish(self, numCourses: int, prerequisites: List[List[int]]) -> bool:
        # build graph -> dict
        graph = defaultdict(list)
        for a, b in prerequisites:
            graph[a].append(b)

        # find cycle: dfs
        state = [0] * numCourses  # 0=unvisited, 1=visiting, 2=done
        def dfs(i):
            if state[i] == 1:  # back edge → cycle
                return False
            if state[i] == 2:  # already fully processed → safe
                return True
            state[i] = 1       # mark as currently visiting
            for j in graph[i]:
                if not dfs(j):
                    return False
            state[i] = 2       # mark as done
            return True

        return all(dfs(i) for i in range(numCourses))
```
