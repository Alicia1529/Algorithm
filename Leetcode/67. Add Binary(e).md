class Solution {
public:
    string addBinary(string a, string b) {
        if (a == "0") return b;
        if (b == "0") return a;
        
        int carryover = 0;
        int idx = 0;
        int na = a.size() - 1, nb = b.size() - 1, nc = na + nb + 1;
        vector<char> ans(nc + 1);
        while (carryover || 0 <= na - idx || 0 <= nb - idx) {
            if (0 <= na - idx) carryover += (int) (a[na - idx] - '0');
            if (0 <= nb - idx) carryover += (int) (b[nb - idx] - '0');
            ans[nc - idx] = (carryover % 2) + '0';
            carryover /= 2;
            idx++;
        }
        string s(ans.begin() + nc - idx + 1, ans.end());
        return s;
    }
};

class Solution {
public:
    string addBinary(string a, string b) {
        string ret = "";
        int carryover = 0;
        for (int i = a.size() - 1, j = b.size() - 1; i >= 0 || j >= 0; i--, j--) {
            int x = (i >= 0) ? a[i] - '0': 0;
            int y = (j >= 0) ? b[j] - '0': 0;
            ret = char((x + y + carryover) % 2 + '0') + ret;
            carryover = (x + y + carryover) / 2;
        }
        return carryover ? '1' + ret : ret;
    }
};

            ret = t + ret;

        }

    }
