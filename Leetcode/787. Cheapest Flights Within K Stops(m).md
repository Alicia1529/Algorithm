### Bellman-Ford
```C++
class Solution {
public:
    int findCheapestPrice(int n, vector<vector<int>>& flights, int src, int dst, int k) {
        vector<int> prices(n, INT_MAX); // key: city, value: distance to source
        prices[src] = 0;

        for (int i = 0; i < k + 1; i++) {
            vector<int> prices_copy(prices);
            
            for (auto& flight: flights) {
                int source = flight[0], dest = flight[1], price = flight[2];
                if (prices[source] == INT_MAX) continue;
                if (prices[source] + price < prices_copy[dest]) {
                    prices_copy[dest] = prices[source] + price;
                }
            }
            prices = prices_copy;
        }
        return prices[dst] == INT_MAX ? -1: prices[dst];

    }
};
```


### BFS
```c++
class Solution {
public:
    int findCheapestPrice(int n, vector<vector<int>>& flights, int src, int dst, int k) {
        vector<int> distance(n, INT_MAX); // key: city, value: distance to source
        distance[src] = 0;
        unordered_map<int, vector<pair<int, int>>> children; // key: parent, value: children {idx, distance}
        unordered_set<int> cities;
        
        for (auto& flight: flights) {
            children[flight[0]].push_back(make_pair(flight[1], flight[2]));
            cities.insert(flight[0]);
            cities.insert(flight[1]);
        }
        
        for (int i = 0; i < k + 1; i++) {
            vector<int> copy(distance);

            for (auto city: cities) {
                for (auto& [child, price]: children[city]) {
                    if (distance[city] == INT_MAX) continue;
                    if (copy[child] > distance[city] + price) copy[child] = distance[city] + price;
                }
            }
            distance = copy;
        }
        return distance[dst] == INT_MAX ? -1: distance[dst];

    }
};
```
