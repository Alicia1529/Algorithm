1. Expand Around Center
```c++
// 20 ms
class Solution {
public:
    string longestPalindrome(string s) {
        int size = s.size();
        string ans = "";
        
        for (int i = 0; i < size; i++) {
            int j = 1;
            while (0 <= i - j && i + j < size && s[i - j] == s[i + j])
                j++;
            if (2 * j - 1 > ans.size()) ans = s.substr(i - j + 1, 2 * j - 1);

            if (i == 0 || s[i - 1] != s[i]) continue;
            int l = i - 1, r = i;
            while (0 <= l && r < size && s[l] == s[r]) {
                l--;
                r++;
            }
            if (r - l - 1 > ans.size()) ans = s.substr(l + 1, r - l - 1);
        }
        return ans;
    }
};


// slightly better: 144 ms
class Solution {
public:
    int expandAroundCenter(string s, int start, int end) {
        while (start >= 0 && end < s.size() && s[start] == s[end]) {
            start--, end++;
        } 
        return end - start - 1;
    }
    string longestPalindrome(string s) {
        if (s.size() <= 1) return s;
        int idx = 0, length = 1;
        for (int i = 0; i < s.size(); i++) {
            int len1 = expandAroundCenter(s, i, i);
            int len2 = expandAroundCenter(s, i, i + 1);
            int len = max(len1, len2);
            if (len > length) {
                length = len;
                // *** important: take care of the - 1
                idx = i - (len - 1) / 2;
            }
        }
        return s.substr(idx, length);
    }
};

// much better: 56 ms
class Solution {
public:
    string longestPalindrome(string s) {
        if (s.size() <= 1) return s;
        int idx = 0, length = 1;
        for (int i = 0; i < s.size(); i++) {
            int left = i - 1, right = i + 1;
            int len = 1;
            while (left >= 0 && s[left] == s[i]) {
                left--, len++;
            }
            while (right < s.size() && s[right] == s[i]) {
                right++, len++;
            }
            while (left >= 0 && right <= s.size() && s[left] == s[right]) {
                left--, right++;
                len += 2;
            }
            if (len > length) {
                length = len;
                idx = left + 1;
            }
        }
        return s.substr(idx, length);
    }
};
```

Complexity Analysis
- Time complexity : O(n^2)
- Space complexity : O(1)


2. Dynamic Programming
```c++
class Solution {
public:
    string longestPalindrome(string s) {
        int n = s.size();
        bool dp[n][n];
        memset(dp, 0, sizeof(dp));
        
        string ans = s.substr(0, 1);
        for (int i = 0; i < n; i++) {
            dp[i][i] = true;
        }
        
        for (int L = 2; L <= n; L++) {
            for (int i = 0; i < n; i++) {
                int j = i + L - 1; // right bound
                if (j >= n) break;
                
                if (j - i == 1) {
                    dp[i][j] = (s[i] == s[j]) ? true: false;
                } else {
                    dp[i][j] = (dp[i + 1][j - 1] && s[i] == s[j]);
                }
                if (dp[i][j] && L > ans.size()) ans = s.substr(i, j - i + 1);
            }
        }
        return ans;
    }
};
```
