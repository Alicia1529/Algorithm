struct ROOM {

    bool avail;

class ExamRoom {
public:
    int num {0};
    vector<ROOM> seats;

    ExamRoom(int n) {
        num = 0;
        for (int i = 0; i < n; i++) {
            ROOM room;
            if (i == 0) {
                room = {-n, n - 1, true};
            } else if (i == n - 1) {
                room = {0, 2 * n, true};
            } else {
                room = {0, n - 1, true};
            }
            seats.emplace_back(room);
        }
    }
    
    int seat() {
        int candidate = 0;
        while (!seats[candidate].avail) candidate++; // find the first available seat
        
        int distance = min(candidate - seats[candidate].left, seats[candidate].right - candidate);
        
        for (int i = candidate + 1; i < seats.size(); i++) {
            if (!seats[i].avail) continue;
            int tmp = min(i - seats[i].left, seats[i].right - i);
            if (tmp > distance) {
                distance = tmp;
                candidate = i;
            }
        }
        seats[candidate].avail = false;
        for (int i = 0; i < candidate; i++) {
            if (seats[i].right > candidate) seats[i].right = candidate;
        }
        for (int i = candidate + 1; i < seats.size(); i++) {
            if (seats[i].left < candidate) seats[i].left = candidate;
        }
        return candidate;
    }

        seats[p].avail = true;
        int left = p, right = p;
        while (seats[left].avail && left > 0) left--;
        while (seats[right].avail && right < seats.size() - 1) right++;
        for (int i = left + 1; i < right; i++) {
            seats[i].left = left;
            seats[i].right = right;
        }
    }
