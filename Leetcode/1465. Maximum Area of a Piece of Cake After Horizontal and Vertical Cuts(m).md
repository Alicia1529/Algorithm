class Solution {
public:
    int maxArea(int h, int w, vector<int>& horizontalCuts, vector<int>& verticalCuts) {
        sort(horizontalCuts.begin(), horizontalCuts.end());
        sort(verticalCuts.begin(), verticalCuts.end());
        int a = 0, maxH = 0, maxV = 0;
        maxH = horizontalCuts[0] - 0;
        for(int i = 1; i < horizontalCuts.size(); ++i){
            a = horizontalCuts[i] - horizontalCuts[i - 1];
            if(a > maxH)
                maxH = a;
        }
        a = h - horizontalCuts[horizontalCuts.size() - 1];
        if(a > maxH)
            maxH = a;
        maxV = verticalCuts[0] - 0;
        for(int i = 1; i < verticalCuts.size(); ++i){
            a = verticalCuts[i] - verticalCuts[i - 1];
            if(a > maxV)
                maxV = a;
        }
        a = w - verticalCuts[verticalCuts.size() - 1];
        if(a > maxV)
            maxV = a;
        return (long) maxV * maxH %1000000007;
    }
};
