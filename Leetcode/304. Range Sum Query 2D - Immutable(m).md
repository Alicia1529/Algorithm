            pp[m][0] = pp[m-1][0]+matrix[m][0];
        
        for(int n = 1; n <matrix[0].size(); n++)
            pp[0][n] = pp[0][n-1]+matrix[0][n];

};

class NumMatrix {
public:
    int m;
    int n;
    unordered_map<int, int> cache;

    NumMatrix(vector<vector<int>>& matrix) {
        m = matrix.size();
        n = matrix[0].size();
        for (int i = 0; i < m; i++) {
            int sum = 0;
            for (int j = 0; j < n; j++) {
                int key = getKey(i, j);
                sum += matrix[i][j];
                cache[key] = sum;
            }
        }
    }
    
    int getKey(int i, int j) {
        return i * n + j;
    }
    
    int sumRegion(int row1, int col1, int row2, int col2) {
        int sum = 0;
        for (int i = row1; i <= row2; i++) {
            int v1 = (col1 - 1 >= 0) ? cache[getKey(i, col1 - 1)] : 0;
            int v2 = cache[getKey(i, col2)];
            sum += (v2 - v1);
        }
        return sum;
        
    }
};
