/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode(int x) : val(x), left(NULL), right(NULL) {}
 * };
 */
class Solution {
public:
    unordered_map<TreeNode*, TreeNode*> parent;
    

    void dfs(TreeNode* c, TreeNode* p) {
        if (c == nullptr) return;
        parent[c] = p;
        dfs(c->left, c);
        dfs(c->right, c);
    }

    vector<int> distanceK(TreeNode* root, TreeNode* target, int k) {
        dfs(root, nullptr);

        vector<int> ans;
        queue<TreeNode*> q;
        unordered_set<TreeNode*> s;

        q.push(target);
        s.insert(target);
        TreeNode* node;
        int level = 0;
        while (!q.empty()) {
            int size = q.size();
            for (int i = 0; i < size; i++) {
                node = q.front();
                q.pop();
                if (node->left && s.find(node->left) == s.end()) {
                    q.push(node->left);
                    s.insert(node->left);
                }
                if (node->right && s.find(node->right) == s.end()) {
                    q.push(node->right);
                    s.insert(node->right);
                }
                if (parent[node] && s.find(parent[node]) == s.end()) {
                    q.push(parent[node]);
                    s.insert(parent[node]);
                }
                if (k == level) ans.push_back(node->val);
            }
            if (level == k) break;
            level++;
        }

        return ans;
        
    }
};
