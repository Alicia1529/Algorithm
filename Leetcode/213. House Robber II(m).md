class Solution:
    def rob(self, nums: List[int]) -> int:
        length = len(nums);
        if (length == 0): return 0
        if (length == 1): return nums[0]
        pre = 0
        prepre = 0
        for i in range(length - 1):
            curr = max(pre, nums[i] + prepre)
            prepre = pre
            pre = curr
        ret = pre
        pre = 0
        prepre = 0
        for i in range(1, length):
            curr = max(pre, nums[i] + prepre)
            prepre = pre
            pre = curr
        return max(pre, ret)
