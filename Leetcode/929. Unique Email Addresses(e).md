class Solution {
public:
    string parseLocalName(string& email, int end) {
        vector<char> name;
        for (int i = 0; i < end; i++) {
            if (email[i] == '+') break;
            if (email[i] != '.') name.push_back(email[i]);
        }
        return string(name.begin(), name.end());
    }
    int numUniqueEmails(vector<string>& emails) {
        unordered_set<string> s;
        for (auto& email: emails) {
            auto pos = email.find('@');
            string e = parseLocalName(email, pos) + email.substr(pos);
            s.insert(e);
        }
        return s.size();
    }
};
