class UndergroundSystem {
private:
    unordered_map<int, pair<string, int>> map_checkin;
    unordered_map<string, pair<int, int>> stats;
public:
    UndergroundSystem() {
        
    }
    
    void checkIn(int id, string stationName, int t) {
        map_checkin[id] = make_pair(stationName, t);
        
    }
    
    void checkOut(int id, string stationName, int t) {
        string startStation = map_checkin[id].first;
        string key = startStation + " " + stationName;
        int enterTime = map_checkin[id].second;
        if (stats.find(key) != stats.end()) {
            stats[key].first += t - enterTime;
            stats[key].second++;
        } else {
            stats[key] = make_pair(t - enterTime, 1);
        }
    }
    
    double getAverageTime(string startStation, string endStation) {
        string key = startStation + " " + endStation;
        double ans = (double) stats[key].first / stats[key].second;
        return ans;
    }
};
