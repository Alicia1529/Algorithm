        if (i == nums.size()) {
            if (sum == target) count++;
        } else {
            calculate(nums, target, i + 1, sum + nums[i]);
            calculate(nums, target, i + 1, sum - nums[i]);
        }
    }
    int findTargetSumWays(vector<int>& nums, int target) {
        calculate(nums, target, 0, 0);
        return count;
    }
};

// Memory Usage: 9.1 MB, less than 73.53% of C++ online submissions for Target Sum.

class Solution {
public:
    int findTargetSumWays(vector<int>& nums, int target) {
        int neg = 0;
        for (auto num: nums) neg -= num;
        target = target - neg;
        
        int dp[target + 1];
        memset(dp, 0, sizeof(dp));
        dp[0] = 1;
        for (int i = 0; i < nums.size(); i++) {
            int num = nums[i];
            for (int k = target; k >= 2 * num; k--) {
                dp[k] += dp[k - 2 * num];
            }
        }
        return dp[target];
    }
};
