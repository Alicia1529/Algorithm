class Solution {
public:
    vector<vector<char>> rotateTheBox(vector<vector<char>>& box) {
        int m = box.size();
        int n = box[0].size();
        vector<vector<char>> ret(n, vector<char>(m, '.'));
        
        for (int c = m - 1; c >= 0; c--) {
            int i = n - 1;
            for (int r = n - 1; r >= 0; r--) {
                if (box[m - c - 1][r] == '*') {
                    ret[r][c] = '*';
                    i = r - 1;
                    
                } else if (box[m - c - 1][r] == '#') {
                    ret[i][c] = '#';
                    i--;
                }
            }
        }
        return ret;
        
    }
};



class Solution {
public:
    vector<vector<char>> rotateTheBox(vector<vector<char>>& box) {
        int m = box.size();
        int n = box[0].size();
        vector<vector<int>> stones;
        vector<vector<int>> obs;
        for (auto& row: box) {
            vector<int> s, o;
            int num = 0;
            for (int i = 0; i < row.size(); i++) {
                if (row[i] == '*') {
                    o.push_back(i);
                    s.push_back(num);
                    num = 0;
                } else if (row[i] == '#') {
                    num++;
                }
            }
            s.push_back(num);
            stones.push_back(s);
            obs.push_back(o);
        }
        vector<vector<char>> ret(n, vector<char>(m, '.'));
        
        for (int c = m - 1; c >= 0; c--) {
            for (int r = n - 1; r >= 0; r--) {
                if (obs[m - c - 1].size() && obs[m - c - 1].back() == r) {
                    ret[r][c] = '*';
                    obs[m - c - 1].pop_back();
                    stones[m - c - 1].pop_back();
                } else if (stones[m - c - 1].back()) {
                    ret[r][c] = '#';
                    stones[m - c - 1].back()--;
                }
            }
        }
        return ret;
        
    }
};
